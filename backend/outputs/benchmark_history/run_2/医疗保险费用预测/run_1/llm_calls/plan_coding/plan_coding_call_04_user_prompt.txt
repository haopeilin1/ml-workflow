import pandas as pd
import numpy as np

import numpy as np

from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from lightgbm import LGBMRegressor

# 全局状态变量，用于存储预处理拟合的参数
PREPROCESS_STATE = {}

def preprocess(df, mode='train'):
    """
    数据清洗和预处理。
    mode='train' 时拟合参数，保存到 PREPROCESS_STATE。
    mode='test' 时应用已保存的参数。
    返回处理后的 DataFrame（仍包含目标列）。
    """
    df = df.copy()
    
    # 丢弃 id 列（如果存在）
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    
    # 定义类别列和数值列
    cat_cols = ['sex', 'smoker', 'region']
    num_cols = ['age', 'bmi', 'children']
    
    # 只保留存在的列
    cat_cols = [c for c in cat_cols if c in df.columns]
    num_cols = [c for c in num_cols if c in df.columns]
    
    if mode == 'train':
        # 构建预处理管道
        preprocessor = ColumnTransformer(
            transformers=[
                ('cat', Pipeline([
                    ('imputer', SimpleImputer(strategy='most_frequent')),
                    ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
                ]), cat_cols),
                ('num', Pipeline([
                    ('imputer', SimpleImputer(strategy='median')),
                    ('scaler', StandardScaler())
                ]), num_cols)
            ],
            remainder='passthrough'
        )
        
        # 分离特征和目标
        target_col = 'charges'
        if target_col in df.columns:
            y = df[target_col].values
            X = df.drop(columns=[target_col])
        else:
            y = None
            X = df
        
        # 拟合预处理器
        preprocessor.fit(X)
        
        # 保存状态
        PREPROCESS_STATE['preprocessor'] = preprocessor
        PREPROCESS_STATE['cat_cols'] = cat_cols
        PREPROCESS_STATE['num_cols'] = num_cols
        
        # 获取转换后的列名
        transformed_X = preprocessor.transform(X)
        
        # 获取 OneHotEncoder 的列名
        ohe = preprocessor.named_transformers_['cat'].named_steps['encoder']
        cat_feature_names = []
        for i, col in enumerate(cat_cols):
            categories = ohe.categories_[i]
            for cat in categories:
                cat_feature_names.append(f"{col}_{cat}")
        
        num_feature_names = num_cols
        all_feature_names = cat_feature_names + num_feature_names
        
        PREPROCESS_STATE['feature_names'] = all_feature_names
        
        # 构建处理后的 DataFrame
        processed_df = pd.DataFrame(transformed_X, columns=all_feature_names)
        if y is not None:
            processed_df[target_col] = y
        
        return processed_df
    
    elif mode == 'test':
        # 从状态中读取已拟合的预处理器
        preprocessor = PREPROCESS_STATE['preprocessor']
        cat_cols = PREPROCESS_STATE['cat_cols']
        num_cols = PREPROCESS_STATE['num_cols']
        feature_names = PREPROCESS_STATE['feature_names']
        
        target_col = 'charges'
        if target_col in df.columns:
            y = df[target_col].values
            X = df.drop(columns=[target_col])
        else:
            y = None
            X = df
        
        # 应用预处理器
        transformed_X = preprocessor.transform(X)
        
        # 构建处理后的 DataFrame
        processed_df = pd.DataFrame(transformed_X, columns=feature_names)
        if y is not None:
            processed_df[target_col] = y
        
        return processed_df
    
    return df

def feature_engineering(df):
    """
    特征工程。
    返回特征矩阵 X（必须不含目标列，所有列必须是数值类型）。
    """
    df = df.copy()
    
    # 丢弃目标列
    target_col = 'charges'
    if target_col in df.columns:
        X = df.drop(columns=[target_col])
    else:
        X = df
    
    # 创建交互特征和多项式特征
    # 注意：此时列名已经是编码后的名称（如 smoker_yes, smoker_no）
    # 我们需要找到对应的列
    
    # 检查是否存在 smoker_yes 列（OneHotEncoder 编码后）
    smoker_yes_col = None
    for col in X.columns:
        if col.startswith('smoker_'):
            smoker_yes_col = col
            break
    
    # 创建 bmi * smoker 交互特征
    if smoker_yes_col and 'bmi' in X.columns:
        X['bmi_smoker'] = X['bmi'] * X[smoker_yes_col]
    
    # 创建 age * smoker 交互特征
    if smoker_yes_col and 'age' in X.columns:
        X['age_smoker'] = X['age'] * X[smoker_yes_col]
    
    # 创建 bmi^2 多项式特征
    if 'bmi' in X.columns:
        X['bmi_squared'] = X['bmi'] ** 2
    
    # 创建 age^2 多项式特征
    if 'age' in X.columns:
        X['age_squared'] = X['age'] ** 2
    
    # 创建 children 的平方特征（可选）
    if 'children' in X.columns:
        X['children_squared'] = X['children'] ** 2
    
    return X

def build_model():

    模型构建和超参数设置。

    model = LGBMRegressor(

        max_depth=6,
        learning_rate=0.05,
        subsample=0.8,
        min_child_samples=20,
        reg_alpha=0.1,
        reg_lambda=0.1,
        random_state=42,
        n_jobs=-1
    )
    return model

_label_encoder = None
_target_dtype = str(train[target_col].dtype).lower() if target_col in train.columns else ''
_is_string_target = target_col in train.columns and (

if _is_string_target:

    # 尝试最基本的预测来兜底
