import pandas as pd
import numpy as np
import pickle
import dill
import json
import os
import re
import warnings
from sklearn.metrics import root_mean_squared_error, mean_absolute_error, r2_score
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler, FunctionTransformer
from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from lightgbm import LGBMRegressor
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

warnings.filterwarnings('ignore')

# ========== 全局状态 ==========
PREPROCESS_STATE = {}

# ========== 预处理函数（与训练时一致）==========
def preprocess(df, mode='train'):
    df = df.copy()
    
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    
    cat_cols = ['sex', 'smoker', 'region']
    num_cols = ['age', 'bmi', 'children']
    
    cat_cols = [c for c in cat_cols if c in df.columns]
    num_cols = [c for c in num_cols if c in df.columns]
    
    if mode == 'train':
        preprocessor = ColumnTransformer(
            transformers=[
                ('cat', Pipeline([
                    ('imputer', SimpleImputer(strategy='most_frequent')),
                    ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
                ]), cat_cols),
                ('num', Pipeline([
                    ('imputer', SimpleImputer(strategy='median')),
                    ('scaler', StandardScaler())
                ]), num_cols)
            ],
            remainder='passthrough'
        )
        
        target_col = 'charges'
        if target_col in df.columns:
            y = df[target_col].values
            X = df.drop(columns=[target_col])
        else:
            y = None
            X = df
        
        preprocessor.fit(X)
        
        PREPROCESS_STATE['preprocessor'] = preprocessor
        PREPROCESS_STATE['cat_cols'] = cat_cols
        PREPROCESS_STATE['num_cols'] = num_cols
        
        transformed_X = preprocessor.transform(X)
        
        ohe = preprocessor.named_transformers_['cat'].named_steps['encoder']
        cat_feature_names = []
        for i, col in enumerate(cat_cols):
            categories = ohe.categories_[i]
            for cat in categories:
                cat_feature_names.append(f"{col}_{cat}")
        
        num_feature_names = num_cols
        all_feature_names = cat_feature_names + num_feature_names
        
        PREPROCESS_STATE['feature_names'] = all_feature_names
        
        processed_df = pd.DataFrame(transformed_X, columns=all_feature_names)
        if y is not None:
            processed_df[target_col] = y
        
        return processed_df
    
    elif mode == 'test':
        preprocessor = PREPROCESS_STATE['preprocessor']
        cat_cols = PREPROCESS_STATE['cat_cols']
        num_cols = PREPROCESS_STATE['num_cols']
        feature_names = PREPROCESS_STATE['feature_names']
        
        target_col = 'charges'
        if target_col in df.columns:
            y = df[target_col].values
            X = df.drop(columns=[target_col])
        else:
            y = None
            X = df
        
        transformed_X = preprocessor.transform(X)
        
        processed_df = pd.DataFrame(transformed_X, columns=feature_names)
        if y is not None:
            processed_df[target_col] = y
        
        return processed_df
    
    return df

def feature_engineering(df):
    df = df.copy()
    
    target_col = 'charges'
    if target_col in df.columns:
        X = df.drop(columns=[target_col])
    else:
        X = df
    
    smoker_yes_col = None
    for col in X.columns:
        if col.startswith('smoker_'):
            smoker_yes_col = col
            break
    
    if smoker_yes_col and 'bmi' in X.columns:
        X['bmi_smoker'] = X['bmi'] * X[smoker_yes_col]
    
    if smoker_yes_col and 'age' in X.columns:
        X['age_smoker'] = X['age'] * X[smoker_yes_col]
    
    if 'bmi' in X.columns:
        X['bmi_squared'] = X['bmi'] ** 2
    
    if 'age' in X.columns:
        X['age_squared'] = X['age'] ** 2
    
    if 'children' in X.columns:
        X['children_squared'] = X['children'] ** 2
    
    return X

def build_model():
    model = LGBMRegressor(
        objective='regression',
        n_estimators=500,
        max_depth=6,
        learning_rate=0.05,
        subsample=0.8,
        min_child_samples=20,
        reg_alpha=0.1,
        reg_lambda=0.1,
        random_state=42,
        n_jobs=-1
    )
    return model

os.makedirs('output', exist_ok=True)

if not model_loaded:
    # 预处理训练数据
    train_clean = preprocess(train, mode='train')
    
    if target_col in train_clean.columns:
        y_train = train_clean[target_col]
        X_train = train_clean.drop(columns=[target_col])
    else:
        y_train = train[target_col]
        X_train = train_clean
    
    # 特征工程
    X_train_fe = feature_engineering(X_train)
    if isinstance(X_train_fe, np.ndarray):
        X_train_fe = pd.DataFrame(X_train_fe, index=X_train.index)
    
    # 清洗特征名
    X_train_fe.columns = [re.sub('[^\\w]', '_', str(c)) for c in X_train_fe.columns]
    if X_train_fe.columns.duplicated().any():
        X_train_fe.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X_train_fe.columns)]
    
    # 训练模型
    model = build_model()
    model.fit(X_train_fe, y_train)
    print("Model retrained successfully")

# 保存模型到 output/model.pkl
with open('output/model.pkl', 'wb') as f:

print("Model saved to output/model.pkl")

feature_names = None
if 'feature_names' in PREPROCESS_STATE:
    feature_names = PREPROCESS_STATE['feature_names']

    # 从训练数据重新获取特征名

feat_imp_df.to_csv('output/feature_importance.csv', index=False)

predict_script = '''
import pandas as pd
import numpy as np
import pickle
import re
import os
import warnings
warnings.filterwarnings('ignore')

# 加载模型
with open('model.pkl', 'rb') as f:
    model = pickle.load(f)

def predict(new_data_path, output_path='predictions.csv'):
    """
    对新数据进行预测。
    
    Parameters:
    - new_data_path: str, 新数据 CSV 文件路径
    - output_path: str, 输出预测结果 CSV 文件路径
    
    Returns:
    - pd.DataFrame, 包含 id 和 prediction 列
    """
    df = pd.read_csv(new_data_path)
    
    # 注意：此预测脚本假设输入数据已经过预处理
    # 如果数据是原始格式，需要在此处添加预处理步骤
    
    # 获取特征列（排除 id 和目标列）
    feature_cols = [c for c in df.columns if c not in ['id', 'charges']]
    X = df[feature_cols]
    
    # 预测
    predictions = model.predict(X)
    
    # 构建结果
    id_col = 'id' if 'id' in df.columns else df.columns[0]
    result = pd.DataFrame({
        'id': df[id_col] if id_col in df.columns else range(len(predictions)),
        'prediction': predictions
    })
    
    result.to_csv(output_path, index=False)
    print(f"Predictions saved to {output_path}")
    
    return result

if __name__ == '__main__':
    import sys
    if len(sys.argv) > 1:
        input_file = sys.argv[1]
        output_file = sys.argv[2] if len(sys.argv) > 2 else 'predictions.csv'
        predict(input_file, output_file)
    else:
        print("Usage: python predict.py <input_csv> [output_csv]")
'''

with open('output/predict.py', 'w', encoding='utf-8') as f:
