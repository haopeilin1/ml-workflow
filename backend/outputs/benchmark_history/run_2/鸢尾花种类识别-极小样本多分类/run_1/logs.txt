PREPROCESS_STATE = {

def preprocess(df, mode='train'):
    '''
    数据清洗和预处理。
    mode='train' 时拟合参数，保存到 PREPROCESS_STATE。
    mode='test' 时应用已保存的参数。
    返回处理后的 DataFrame（仍包含目标列）。

    feature_cols = PREPROCESS_STATE['feature_cols']
    
    if mode == 'train':
        # 2. 编码目标列 species 为整数标签 (Must-Do #4)

        PREPROCESS_STATE['scaler'] = scaler
        
    elif mode == 'test':
        # 应用已保存的编码器和标准化器
