# 由于特征工程包含自定义逻辑，这里构建一个包装类
from sklearn.base import BaseEstimator, ClassifierMixin

class FullPipeline(BaseEstimator, ClassifierMixin):
    """完整的预处理+特征工程+模型Pipeline"""
    def __init__(self, preprocessor, cat_cols, num_cols, feature_names, model):
        self.preprocessor = preprocessor
        self.cat_cols = cat_cols
        self.num_cols = num_cols
        self.feature_names = feature_names
        self.model = model
        self._estimator_type = "classifier"
    
    def _preprocess_and_engineer(self, X):
        """对原始数据执行预处理和特征工程"""
        X_copy = X.copy()
        
        # 丢弃ID列
        id_cols = ['product_price', 'discount_percent', 'product_rating', 
                   'past_return_rate', 'delivery_delay_days', 'session_length_minutes', 'order_id']
        X_copy = X_copy.drop(columns=[col for col in id_cols if col in X_copy.columns], errors='ignore')
        
        # 预处理
        X_processed = self.preprocessor.transform(X_copy)
        
        # 获取特征名
        cat_feature_names = []
        if self.cat_cols:
            encoder = self.preprocessor.named_transformers_['cat'].named_steps['encoder']
            cat_feature_names = encoder.get_feature_names_out(self.cat_cols).tolist()
        num_feature_names = self.num_cols
        remainder_cols = [col for col in X_copy.columns if col not in self.cat_cols and col not in self.num_cols]
        all_feature_names = cat_feature_names + num_feature_names + remainder_cols
        
        X_df = pd.DataFrame(X_processed, columns=all_feature_names, index=X_copy.index)
        
        # 特征工程
        num_cols_fe = X_df.select_dtypes(include=[np.number]).columns.tolist()
        
        if 'num_product_views' in num_cols_fe and 'customer_age' in num_cols_fe:
            X_df['views_per_age'] = X_df['num_product_views'] / (X_df['customer_age'] + 1)
        if 'past_purchase_count' in num_cols_fe and 'num_product_views' in num_cols_fe:
            X_df['purchase_to_view_ratio'] = X_df['past_purchase_count'] / (X_df['num_product_views'] + 1)
        if 'customer_age' in num_cols_fe:
            X_df['age_squared'] = X_df['customer_age'] ** 2
            X_df['age_group'] = (X_df['customer_age'] // 10).astype(int)
        if 'num_product_views' in num_cols_fe:
            X_df['log_views'] = np.log1p(X_df['num_product_views'].clip(lower=0))
        if 'past_purchase_count' in num_cols_fe:
            X_df['log_purchases'] = np.log1p(X_df['past_purchase_count'])
        if 'used_coupon' in num_cols_fe:
            X_df['used_coupon'] = X_df['used_coupon'].astype(int)
        
        X_df = X_df.select_dtypes(include=[np.number])
        X_df = X_df.replace([np.inf, -np.inf], np.nan)
        X_df = X_df.fillna(0)
        
        # 清洗特征名
        X_df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_df.columns]
        if X_df.columns.duplicated().any():
            X_df.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X_df.columns)]
        
        # 对齐特征名
        for col in self.feature_names:
            if col not in X_df.columns:
                X_df[col] = 0
        X_df = X_df[self.feature_names]
        
        return X_df
    
    def fit(self, X, y):
        return self
    
    def predict(self, X):
        X_processed = self._preprocess_and_engineer(X)
        return self.model.predict(X_processed)
    
    def predict_proba(self, X):
        X_processed = self._preprocess_and_engineer(X)
        return self.model.predict_proba(X_processed)
    
    @property
    def classes_(self):
        return self.model.classes_
    
    @property
    def feature_importances_(self):
        return self.model.feature_importances_
