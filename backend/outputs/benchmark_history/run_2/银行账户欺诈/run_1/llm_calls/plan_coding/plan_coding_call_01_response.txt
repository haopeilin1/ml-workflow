    # 这里需要重新执行训练逻辑，但根据提示，模型已存在，此分支仅作兜底
    raise RuntimeError("Model loading failed and retraining is not implemented in this script. Please ensure data/best_model.pkl exists.")

if hasattr(model, 'predict_proba'):

    test_preds = model.predict(X_test).astype(int)

importance = None

    importance = model.feature_importances_
elif hasattr(model, 'named_steps'):
    # 如果是Pipeline，尝试从最后一步获取
    final_step = model.steps[-1][1]
    if hasattr(final_step, 'feature_importances_'):
        importance = final_step.feature_importances_
    elif hasattr(final_step, 'coef_'):
        importance = np.abs(final_step.coef_).flatten()

if importance is None:
    # 如果无法获取特征重要性，使用均匀分布
    print("Warning: Could not extract feature importance from model. Using uniform importance.")
    importance = np.ones(len(feature_names)) / len(feature_names)

# 确保特征重要性长度与特征名匹配
if len(importance) != len(feature_names):
    print(f"Warning: Feature importance length ({len(importance)}) does not match feature names length ({len(feature_names)}). Truncating/padding.")
    if len(importance) > len(feature_names):
        importance = importance[:len(feature_names)]
    else:
        importance = np.pad(importance, (0, len(feature_names) - len(importance)), 'constant', constant_values=0)

# 创建特征重要性DataFrame
importance_df = pd.DataFrame({
    'name': feature_names,
    'importance': importance
})

plt.figure(figsize=(10, 8))

plt.yticks(range(top_n), top_features['name'].values)
plt.xlabel('Importance')
plt.title('Top {} Feature Importance'.format(top_n))
plt.gca().invert_yaxis()
plt.tight_layout()
plt.savefig('output/feature_importance.png', dpi=100, bbox_inches='tight')
plt.close()
print("Feature importance plot saved to output/feature_importance.png")

train = pd.read_csv('data/train.csv')

plt.xlabel('Recall')
plt.ylabel('Precision')
plt.title(f'Precision-Recall Curve (AUC={auc_score:.4f}, AP={ap_score:.4f})')
plt.legend(loc='lower left')
plt.grid(True, alpha=0.3)

tn, fp, fn, tp = cm.ravel() if cm.size == 4 else (0, 0, 0, 0)

# ========== 10. 生成独立的预测脚本 ==========
predict_script = '''
import pandas as pd
import pickle
import os
import sys

def load_model(model_path='model.pkl'):
    """加载保存的模型"""
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    return model

def predict(model, data_path):
    """对新数据进行预测"""
    # 加载数据
    df = pd.read_csv(data_path)
    
    # 进行预测
    if hasattr(model, 'predict_proba'):
        probabilities = model.predict_proba(df)[:, 1]
        predictions = (probabilities >= 0.5).astype(int)
    else:
        predictions = model.predict(df).astype(int)
        probabilities = predictions.astype(float)
    
    # 构建结果DataFrame
    result = df.copy()
    result['prediction'] = predictions
    result['probability'] = probabilities
    
    return result

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python predict.py <input_csv_path> [output_csv_path]")
        print("Example: python predict.py new_data.csv predictions.csv")
        sys.exit(1)
    
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else 'predictions.csv'
    
    if not os.path.exists(input_path):
        print(f"Error: Input file '{input_path}' not found.")
        sys.exit(1)
    
    # 加载模型
    model = load_model('model.pkl')
    
    # 进行预测
    result = predict(model, input_path)
    
    # 保存结果
    result.to_csv(output_path, index=False)
    print(f"Predictions saved to {output_path}")
    print(f"Total rows processed: {len(result)}")
    print(f"Fraud predictions: {result['prediction'].sum()}")
    print(f"Fraud rate: {result['prediction'].mean():.4f}")
'''

with open('output/predict.py', 'w', encoding='utf-8') as f:
    f.write(predict_script)
