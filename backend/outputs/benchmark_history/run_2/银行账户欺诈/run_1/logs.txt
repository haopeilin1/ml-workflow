   总结修复方案：
   - 在 `feature_engineering` 中使用 `PREPROCESS_STATE` 保存/加载 OneHotEncoder
   - 在最后强制转换所有列为数值类型
   - 确保所有路径都返回纯数值 DataFrame

import pandas as pd
import numpy as np
from sklearn.compose import ColumnTransformer

# 全局状态：用于在preprocess的train和test模式间传递参数
PREPROCESS_STATE = {}

def preprocess(df, mode='train'):
    '''
    数据清洗和预处理。
    mode='train' 时拟合参数，保存到 PREPROCESS_STATE。
    mode='test' 时应用已保存的参数。
    返回处理后的 DataFrame（仍包含目标列）。
    '''
    global PREPROCESS_STATE
    target_col = 'fraud_bool'
    
    # 1. 丢弃疑似ID列和常数特征
    id_cols = ['id', 'name_email_similarity', 'days_since_request', 
               'intended_balcon_amount', 'velocity_6h', 'velocity_24h', 
               'velocity_4w', 'session_length_in_minutes']
    constant_cols = ['device_fraud_count', 'month']
    drop_cols = id_cols + constant_cols
    
    # 安全丢弃（使用errors='ignore'防止列不存在时报错）
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    # 2. 将-1值识别为缺失标记并替换为NaN
    cols_with_minus_one = ['prev_address_months_count', 'current_address_months_count', 
                           'bank_months_count', 'device_distinct_emails_8w']
    if 'session_length_in_minutes' in df.columns:
        cols_with_minus_one.append('session_length_in_minutes')
    
    for col in cols_with_minus_one:
        if col in df.columns:
            # 创建缺失指示列
            missing_col_name = f'{col}_missing'
            df[missing_col_name] = (df[col] == -1).astype(int)
            # 将-1替换为NaN
            df.loc[df[col] == -1, col] = np.nan
    
    if mode == 'train':
        # 计算中位数用于填充
        PREPROCESS_STATE['medians'] = {}
        for col in cols_with_minus_one:
            if col in df.columns:
                PREPROCESS_STATE['medians'][col] = df[col].median()
        
        # 计算log1p变换所需的偏移量（确保非负）
        PREPROCESS_STATE['log_cols'] = ['prev_address_months_count', 'bank_branch_count_8w']
        if 'session_length_in_minutes' in df.columns:
            PREPROCESS_STATE['log_cols'].append('session_length_in_minutes')
    
    # 填充缺失值
    if mode == 'train':
        for col in cols_with_minus_one:
            if col in df.columns:
                df[col] = df[col].fillna(PREPROCESS_STATE['medians'][col])
    else:
        for col in cols_with_minus_one:
            if col in df.columns and col in PREPROCESS_STATE.get('medians', {}):
                df[col] = df[col].fillna(PREPROCESS_STATE['medians'][col])
    
    # 3. 对高度右偏特征进行log1p变换
    log_cols = PREPROCESS_STATE.get('log_cols', ['prev_address_months_count', 'bank_branch_count_8w'])
    for col in log_cols:
        if col in df.columns:
            # 确保非负（log1p要求输入>=0）
            min_val = df[col].min()
            if min_val < 0:
                df[col] = df[col] - min_val  # 平移使最小值为0
            df[col] = np.log1p(df[col])
    
    # 4. 清洗特征名（移除特殊JSON字符）
    df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
    
    return df


def feature_engineering(df):
    '''
    特征工程。
    返回特征矩阵 X（必须不含目标列，所有列必须是数值类型）。
    使用 PREPROCESS_STATE 保存编码器，确保 train/val/test 使用相同的编码映射。
    '''
    global PREPROCESS_STATE
    target_col = 'fraud_bool'
    
    # 分离特征和目标列（安全处理，因为系统骨架可能已经drop了target_col）
    X = df.drop(columns=[target_col], errors='ignore')
    
    # 1. 创建交叉特征：收入与信用评分的交互
    if 'income' in X.columns and 'credit_risk_score' in X.columns:
        # 确保是数值类型再计算
        income_vals = pd.to_numeric(X['income'], errors='coerce').fillna(0)
        credit_vals = pd.to_numeric(X['credit_risk_score'], errors='coerce').fillna(0)
        X['income_credit_risk_interaction'] = income_vals * credit_vals
    
    # 2. 创建比率特征：地址变更频率
    if 'prev_address_months_count' in X.columns and 'current_address_months_count' in X.columns:
        prev_vals = pd.to_numeric(X['prev_address_months_count'], errors='coerce').fillna(0)
        curr_vals = pd.to_numeric(X['current_address_months_count'], errors='coerce').fillna(0)
        X['address_change_ratio'] = np.where(
            curr_vals > 0,
            prev_vals / (curr_vals + 1),
            0
        )
    
    # 3. 创建聚合特征：设备关联邮箱数与出生日期关联邮箱数的差值
    if 'device_distinct_emails_8w' in X.columns and 'date_of_birth_distinct_emails_4w' in X.columns:
        device_vals = pd.to_numeric(X['device_distinct_emails_8w'], errors='coerce').fillna(0)
        dob_vals = pd.to_numeric(X['date_of_birth_distinct_emails_4w'], errors='coerce').fillna(0)
        X['email_diff_device_dob'] = device_vals - dob_vals
    
    # 4. 识别类别列（object/string/category类型）
    cat_cols = X.select_dtypes(include=['object', 'string', 'category']).columns.tolist()
    num_cols = X.select_dtypes(exclude=['object', 'string', 'category']).columns.tolist()
    
    # 5. 对类别列使用OneHotEncoder
    if cat_cols:
        # 检查 PREPROCESS_STATE 中是否已有编码器
        if 'onehot_encoder' in PREPROCESS_STATE:
            # test 模式：使用已保存的编码器
            encoder = PREPROCESS_STATE['onehot_encoder']
            X_cat_encoded = encoder.transform(X[cat_cols])
        else:
            # train 模式：创建并保存编码器
            encoder = OneHotEncoder(handle_unknown='ignore', sparse_output=False)
            X_cat_encoded = encoder.fit_transform(X[cat_cols])
            PREPROCESS_STATE['onehot_encoder'] = encoder
            PREPROCESS_STATE['cat_cols'] = cat_cols
        
        # 获取编码后的特征名
        encoded_feature_names = encoder.get_feature_names_out(cat_cols)
        # 清洗特征名
        encoded_feature_names = [re.sub(r'[^\w]', '_', str(c)) for c in encoded_feature_names]
        X_cat_df = pd.DataFrame(X_cat_encoded, columns=encoded_feature_names, index=X.index)
        
        # 合并数值列和编码后的类别列
        X_num_df = X[num_cols].copy()
        X = pd.concat([X_num_df.reset_index(drop=True), X_cat_df.reset_index(drop=True)], axis=1)
    
    # 6. 最终清洗所有特征名
    X.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X.columns]
    
    # 7. 【关键】强制确保所有列都是数值类型，杜绝 string/object 残留
    for col in X.columns:
        if X[col].dtype == 'object' or str(X[col].dtype) == 'string':
            X[col] = pd.to_numeric(X[col], errors='coerce')
        elif X[col].dtype == 'bool':
            X[col] = X[col].astype(int)
    
    # 8. 填充可能因类型转换产生的NaN
    X = X.fillna(0)
    
    # 9. 最终安全检查：确保没有任何 object/string 列
    remaining_obj_cols = X.select_dtypes(include=['object', 'string', 'category']).columns.tolist()
    if remaining_obj_cols:
        # 最后的兜底：强制转换所有剩余的非数值列
        for col in remaining_obj_cols:
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    
    return X

def evaluate_model(model, X_val, y_val):
    '''
    自定义验证集评估指标。
    返回一个 dict，键名以 val_ 开头。
    '''
    # 预测概率
    y_pred_proba = model.predict_proba(X_val)[:, 1]
    
    # 计算ROC-AUC
    val_auc = roc_auc_score(y_val, y_pred_proba)
    
    # 计算Average Precision
    val_ap = average_precision_score(y_val, y_pred_proba)
    
    # 搜索最优阈值（在0.01到0.5范围内，步长0.01）
    best_threshold = 0.5
    best_f1 = 0.0
    for threshold in np.arange(0.01, 0.51, 0.01):
        y_pred = (y_pred_proba >= threshold).astype(int)
        f1 = f1_score(y_val, y_pred)
        if f1 > best_f1:
            best_f1 = f1
            best_threshold = threshold
    
    # 使用最优阈值计算指标
    y_pred_optimal = (y_pred_proba >= best_threshold).astype(int)
    val_f1 = f1_score(y_val, y_pred_optimal)
    val_recall = recall_score(y_val, y_pred_optimal)
    val_precision = precision_score(y_val, y_pred_optimal)
    
    # 计算混淆矩阵
    cm = confusion_matrix(y_val, y_pred_optimal)
    tn, fp, fn, tp = cm.ravel() if cm.size == 4 else (0, 0, 0, 0)
    
    return {
        'val_auc': float(val_auc),
        'val_ap': float(val_ap),
        'val_f1': float(val_f1),
        'val_recall': float(val_recall),
        'val_precision': float(val_precision),
        'val_best_threshold': float(best_threshold),
        'val_tp': int(tp),
        'val_fp': int(fp),
        'val_tn': int(tn),
        'val_fn': int(fn)
    }
