import pandas as pd
import numpy as np
import pickle
import dill
import json
import os
import re
import warnings
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.metrics import accuracy_score, f1_score, confusion_matrix, classification_report, log_loss
from sklearn.preprocessing import LabelEncoder

warnings.filterwarnings('ignore')

# 确保输出目录存在
os.makedirs('output', exist_ok=True)

print("=" * 60)
print("产物生成开始")
print("=" * 60)

# ========== 1. 加载已保存的最佳模型 ==========
print("\n[1/7] 加载模型...")
try:
    with open('data/best_model.pkl', 'rb') as f:
        model = dill.load(f)
    print("成功从 data/best_model.pkl 加载模型")
except Exception as e:
    print(f"加载 best_model.pkl 失败: {e}")
    print("尝试重新训练模型...")
    # 如果加载失败，回退到重新训练
    train = pd.read_csv('data/train.csv')

    target_col = 'Status'
    
    # 预处理函数（从最佳代码复制）
    def preprocess(df, mode='train'):
        global PREPROCESS_STATE
        target_col = 'Status'
        df = df.drop(columns=['id'], errors='ignore')
        df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
        
        if target_col in df.columns:
            y = df[target_col].copy()
            X = df.drop(columns=[target_col])
        else:
            y = None
            X = df.copy()
        
        cat_cols = X.select_dtypes(include=['object', 'category']).columns.tolist()
        num_cols = X.select_dtypes(exclude=['object', 'category']).columns.tolist()
        
        right_skewed_features = ['N_Days', 'Bilirubin', 'Cholesterol', 'Copper', 'Alk_Phos', 'Tryglicerides', 'SGOT']
        right_skewed_features = [f for f in right_skewed_features if f in num_cols]
        
        edema_col = 'Edema' if 'Edema' in cat_cols else None
        binary_cat_cols = [c for c in cat_cols if c != edema_col and X[c].nunique() <= 2]
        other_cat_cols = [c for c in cat_cols if c not in binary_cat_cols and c != edema_col]
        
        if mode == 'train':
            from sklearn.compose import ColumnTransformer
            from sklearn.preprocessing import OneHotEncoder, OrdinalEncoder, StandardScaler, FunctionTransformer
            from sklearn.impute import SimpleImputer
            from sklearn.experimental import enable_iterative_imputer
            from sklearn.impute import IterativeImputer
            from sklearn.pipeline import Pipeline
            
            def log_transform_right_skewed(X):
                X = X.copy()
                for col in right_skewed_features:
                    if col in X.columns:
                        min_val = X[col].min()
                        if pd.notna(min_val) and min_val < 0:
                            X[col] = X[col] - min_val
                        X[col] = np.log1p(X[col].clip(lower=0))
                return X
            
            log_transformer = FunctionTransformer(log_transform_right_skewed, validate=False)
            iterative_imputer = IterativeImputer(random_state=42, max_iter=10, verbose=0)
            scaler = StandardScaler()
            
            num_pipeline = Pipeline([
                ('log_transform', log_transformer),
                ('iterative_imputer', iterative_imputer),
                ('scaler', scaler)
            ])
            
            cat_transformers = []
            if edema_col:
                edema_pipeline = Pipeline([
                    ('imputer', SimpleImputer(strategy='most_frequent')),
                    ('encoder', OrdinalEncoder(categories=[['N', 'S', 'Y']], handle_unknown='use_encoded_value', unknown_value=-1))
                ])
                cat_transformers.append(('edema', edema_pipeline, [edema_col]))
            
            if binary_cat_cols:
                binary_pipeline = Pipeline([
                    ('imputer', SimpleImputer(strategy='most_frequent')),
                    ('encoder', OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1))
                ])
                cat_transformers.append(('binary_cat', binary_pipeline, binary_cat_cols))
            
            if other_cat_cols:
                other_pipeline = Pipeline([
                    ('imputer', SimpleImputer(strategy='most_frequent')),
                    ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
                ])
                cat_transformers.append(('other_cat', other_pipeline, other_cat_cols))
            
            transformers = []
            if num_cols:
                transformers.append(('num', num_pipeline, num_cols))
            transformers.extend(cat_transformers)
            
            preprocessor = ColumnTransformer(transformers, remainder='drop')
            X_processed = preprocessor.fit_transform(X)
            
            feature_names = []
            if num_cols:
                feature_names.extend(num_cols)
            if edema_col:
                feature_names.append(edema_col)
            if binary_cat_cols:
                feature_names.extend(binary_cat_cols)
            if other_cat_cols:
                other_encoder = preprocessor.named_transformers_['other_cat'].named_steps['encoder']
                ohe_features = other_encoder.get_feature_names_out(other_cat_cols).tolist()
                feature_names.extend(ohe_features)
            
            PREPROCESS_STATE['preprocessor'] = preprocessor
            PREPROCESS_STATE['feature_names'] = feature_names
            PREPROCESS_STATE['target_col'] = target_col
            PREPROCESS_STATE['num_cols'] = num_cols
            PREPROCESS_STATE['cat_cols'] = cat_cols
            PREPROCESS_STATE['right_skewed_features'] = right_skewed_features
            
            X_processed_df = pd.DataFrame(X_processed, index=X.index, columns=feature_names)
            if y is not None:
                X_processed_df[target_col] = y.values
            return X_processed_df
        else:
            preprocessor = PREPROCESS_STATE.get('preprocessor')
            feature_names = PREPROCESS_STATE.get('feature_names', [])
            X_processed = preprocessor.transform(X)
            X_processed_df = pd.DataFrame(X_processed, index=X.index, columns=feature_names)
            if y is not None:
                X_processed_df[target_col] = y.values
            return X_processed_df
    
    def feature_engineering(df):
        target_col = PREPROCESS_STATE.get('target_col', 'Status')
        if target_col in df.columns:
            X = df.drop(columns=[target_col])
        else:
            X = df.copy()
        
        if 'Stage' in X.columns and 'Bilirubin' in X.columns:
            X['Stage_Bilirubin_interact'] = X['Stage'] * X['Bilirubin']
        if 'N_Days' in X.columns:
            X['N_Days_segment'] = pd.cut(X['N_Days'], bins=[-np.inf, 365, 1825, np.inf], labels=[0, 1, 2]).astype(float)
            if 'Stage' in X.columns:
                X['Stage_N_Days_interact'] = X['Stage'] * X['N_Days']
        if 'Albumin' in X.columns and 'Bilirubin' in X.columns:
            X['Albumin_Bilirubin_ratio'] = X['Albumin'] / (X['Bilirubin'] + 1e-6)
        if 'Age' in X.columns:
            X['Age_years'] = X['Age'] / 365.25
            X['Age_group'] = pd.cut(X['Age_years'], bins=[0, 40, 50, 60, 70, 100], labels=[0, 1, 2, 3, 4]).astype(float)
        
        for col in X.columns:
            if X[col].dtype == 'object' or X[col].dtype.name == 'category':
                X[col] = pd.to_numeric(X[col], errors='coerce')
        X = X.fillna(0)
        return X
    
    from lightgbm import LGBMClassifier
    PREPROCESS_STATE = {}
    
    train_clean = preprocess(train, mode='train')
    val_clean = preprocess(val, mode='test')
    
    if target_col in train_clean.columns:
        y_train = train_clean[target_col]
        X_train = train_clean.drop(columns=[target_col])
    else:
        y_train = train[target_col]
        X_train = train_clean
    
    if target_col in val_clean.columns:
        y_val = val_clean[target_col]
        X_val = val_clean.drop(columns=[target_col])
    else:
        y_val = val[target_col]
        X_val = val_clean
    
    _label_encoder = None
    _target_dtype = str(train[target_col].dtype).lower()
    _is_string_target = train[target_col].dtype == object or _target_dtype == 'category'
    if _is_string_target:
        _label_encoder = LabelEncoder()
        y_train = _label_encoder.fit_transform(train[target_col])
        y_val = _label_encoder.transform(val[target_col])
        PREPROCESS_STATE['label_encoder'] = _label_encoder
    
    X_train_fe = feature_engineering(X_train)
    if isinstance(X_train_fe, np.ndarray):
        X_train_fe = pd.DataFrame(X_train_fe, index=X_train.index)
    X_val_fe = feature_engineering(X_val)
    if isinstance(X_val_fe, np.ndarray):
        X_val_fe = pd.DataFrame(X_val_fe, index=X_val.index)
    
    for _df in [X_train_fe, X_val_fe]:
        _df.columns = [re.sub('[^\\w]', '_', str(c)) for c in _df.columns]
    for _df in [X_train_fe, X_val_fe]:
        if _df.columns.duplicated().any():
            _df.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(_df.columns)]
    
    model = LGBMClassifier(
        objective='multiclass',
        num_class=3,
        boosting_type='gbdt',
        learning_rate=0.05,
        num_leaves=31,
        max_depth=8,
        min_child_samples=20,
        subsample=0.8,
        colsample_bytree=0.8,
        reg_alpha=0.1,
        reg_lambda=0.1,
        n_estimators=500,
        class_weight='balanced',
        random_state=42,
        verbose=-1,
        n_jobs=-1
    )
    try:
        model.fit(X_train_fe, y_train, eval_set=[(X_val_fe, y_val)])
    except Exception:
        model.fit(X_train_fe, y_train)
    
    print("模型重新训练完成")

# ========== 2. 加载测试数据并预处理 ==========
print("\n[2/7] 加载测试数据...")
test = pd.read_csv('data/test.csv')
print(f"测试集形状: {test.shape}")

# 检查是否有预处理状态
if 'PREPROCESS_STATE' in globals() and PREPROCESS_STATE:
    # 使用已有的预处理状态
    target_col = PREPROCESS_STATE.get('target_col', 'Status')
    
    def preprocess_test(df):
        preprocessor = PREPROCESS_STATE.get('preprocessor')
        feature_names = PREPROCESS_STATE.get('feature_names', [])
        df = df.drop(columns=['id'], errors='ignore')
        df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
        if target_col in df.columns:
            X = df.drop(columns=[target_col])
        else:
            X = df.copy()
        X_processed = preprocessor.transform(X)
        X_processed_df = pd.DataFrame(X_processed, index=X.index, columns=feature_names)
        return X_processed_df
    
    def feature_engineering_test(df):
        X = df.copy()
        if 'Stage' in X.columns and 'Bilirubin' in X.columns:
            X['Stage_Bilirubin_interact'] = X['Stage'] * X['Bilirubin']
        if 'N_Days' in X.columns:
            X['N_Days_segment'] = pd.cut(X['N_Days'], bins=[-np.inf, 365, 1825, np.inf], labels=[0, 1, 2]).astype(float)
            if 'Stage' in X.columns:
                X['Stage_N_Days_interact'] = X['Stage'] * X['N_Days']
        if 'Albumin' in X.columns and 'Bilirubin' in X.columns:
            X['Albumin_Bilirubin_ratio'] = X['Albumin'] / (X['Bilirubin'] + 1e-6)
        if 'Age' in X.columns:
            X['Age_years'] = X['Age'] / 365.25
            X['Age_group'] = pd.cut(X['Age_years'], bins=[0, 40, 50, 60, 70, 100], labels=[0, 1, 2, 3, 4]).astype(float)
        for col in X.columns:
            if X[col].dtype == 'object' or X[col].dtype.name == 'category':
                X[col] = pd.to_numeric(X[col], errors='coerce')
        X = X.fillna(0)
        return X
    
    test_clean = preprocess_test(test)
    X_test_fe = feature_engineering_test(test_clean)
    if isinstance(X_test_fe, np.ndarray):
        X_test_fe = pd.DataFrame(X_test_fe, index=test_clean.index)
    X_test_fe.columns = [re.sub('[^\\w]', '_', str(c)) for c in X_test_fe.columns]
    if X_test_fe.columns.duplicated().any():
        X_test_fe.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X_test_fe.columns)]
else:
    # 如果没有预处理状态，尝试直接从模型推断（模型可能是Pipeline）
    target_col = 'Status'
    test_clean = test.drop(columns=['id'], errors='ignore')
    test_clean.columns = [re.sub(r'[^\w]', '_', str(c)) for c in test_clean.columns]
    if target_col in test_clean.columns:
        X_test_fe = test_clean.drop(columns=[target_col])
    else:
        X_test_fe = test_clean.copy()
    for col in X_test_fe.columns:
        if X_test_fe[col].dtype == 'object' or X_test_fe[col].dtype.name == 'category':
            X_test_fe[col] = pd.to_numeric(X_test_fe[col], errors='coerce')
    X_test_fe = X_test_fe.fillna(0)

# 确保特征顺序与训练时一致（如果模型有 feature_names_in_ 属性）
if hasattr(model, 'feature_names_in_'):
    expected_features = model.feature_names_in_
    # 对齐特征
    for feat in expected_features:
        if feat not in X_test_fe.columns:
            X_test_fe[feat] = 0
    X_test_fe = X_test_fe[expected_features]

print(f"预处理后测试集特征形状: {X_test_fe.shape}")

# ========== 3. 测试集预测 ==========
print("\n[3/7] 进行测试集预测...")
test_preds_encoded = model.predict(X_test_fe)
try:
    test_probs_all = model.predict_proba(X_test_fe)
except Exception:
    test_probs_all = None

# 处理标签编码器
label_encoder = None
if 'PREPROCESS_STATE' in globals() and PREPROCESS_STATE:
    label_encoder = PREPROCESS_STATE.get('label_encoder', None)

# 如果没有label_encoder，尝试从模型获取类别
if label_encoder is None and hasattr(model, 'classes_'):
    classes = model.classes_
    # 检查是否是字符串类别
    if len(classes) > 0 and isinstance(classes[0], (str, np.str_)):
        label_encoder = LabelEncoder()
        label_encoder.classes_ = classes

# 转换预测结果
if label_encoder is not None:
    test_preds = label_encoder.inverse_transform(test_preds_encoded)
else:
    test_preds = test_preds_encoded

# 确保预测结果是字符串类型（统一类型，避免混合类型错误）
test_preds = np.array(test_preds).astype(str)

print(f"预测完成，共 {len(test_preds)} 条结果")
print(f"预测类别分布: {pd.Series(test_preds).value_counts().to_dict()}")

# ========== 4. 保存测试预测结果 ==========
print("\n[4/7] 保存测试预测结果...")
id_col = 'id' if 'id' in test.columns else test.columns[0]
result_df = pd.DataFrame({
    'id': test[id_col] if id_col in test.columns else range(len(test_preds)),
    'prediction': test_preds,
})

if test_probs_all is not None:
    for i in range(test_probs_all.shape[1]):
        class_label = label_encoder.inverse_transform([i])[0] if label_encoder is not None else f'class_{i}'
        result_df[f'proba_{class_label}'] = test_probs_all[:, i]

result_df.to_csv('output/test_predictions.csv', index=False)
print("测试预测结果已保存到 output/test_predictions.csv")

# ========== 5. 保存模型 ==========
print("\n[5/7] 保存模型到 output/model.pkl...")
with open('output/model.pkl', 'wb') as f:
    dill.dump(model, f)
print("模型已保存到 output/model.pkl")

# ========== 6. 特征重要性 ==========
print("\n[6/7] 生成特征重要性...")
# 获取特征名
if hasattr(model, 'feature_names_in_'):
    feature_names = list(model.feature_names_in_)
elif hasattr(X_test_fe, 'columns'):
    feature_names = list(X_test_fe.columns)
else:
    feature_names = [f'feature_{i}' for i in range(X_test_fe.shape[1])]

# 获取特征重要性
if hasattr(model, 'feature_importances_'):
    importances = model.feature_importances_
elif hasattr(model, 'coef_'):
    importances = np.abs(model.coef_).mean(axis=0) if model.coef_.ndim > 1 else np.abs(model.coef_)
else:
    importances = np.ones(len(feature_names)) / len(feature_names)

# 确保长度一致
if len(importances) != len(feature_names):
    feature_names = [f'feature_{i}' for i in range(len(importances))]

# 创建特征重要性DataFrame
feature_importance_df = pd.DataFrame({
    'name': feature_names,
    'importance': importances
})
feature_importance_df = feature_importance_df.sort_values('importance', ascending=False)

# 保存CSV
feature_importance_df.to_csv('output/feature_importance.csv', index=False)
print("特征重要性已保存到 output/feature_importance.csv")

# 生成特征重要性图（仅当特征数不超过100时）
if len(feature_importance_df) <= 100:
    plt.figure(figsize=(10, max(6, len(feature_importance_df) * 0.3)))
    top_features = feature_importance_df.head(20)
    plt.barh(range(len(top_features)), top_features['importance'].values)
    plt.yticks(range(len(top_features)), top_features['name'].values)
    plt.xlabel('Importance')
    plt.title('Feature Importance (Top 20)')
    plt.gca().invert_yaxis()
    plt.tight_layout()

    plt.close()
    print("特征重要性图已保存到 output/feature_importance.png")
else:
    print("特征数量过多，跳过特征重要性图生成")

# ========== 7. 生成报告和可视化 ==========
print("\n[7/7] 生成报告和可视化...")

# 加载训练数据用于评估指标计算
train = pd.read_csv('data/train.csv')
target_col = 'Status'

# 对训练集进行预测以计算指标
if 'PREPROCESS_STATE' in globals() and PREPROCESS_STATE:
    train_clean = preprocess_test(train) if 'preprocess_test' in dir() else train.drop(columns=['id'], errors='ignore')
    if isinstance(train_clean, pd.DataFrame):
        X_train_fe_eval = feature_engineering_test(train_clean) if 'feature_engineering_test' in dir() else train_clean
    else:
        X_train_fe_eval = train_clean
    if isinstance(X_train_fe_eval, np.ndarray):
        X_train_fe_eval = pd.DataFrame(X_train_fe_eval)
    X_train_fe_eval.columns = [re.sub('[^\\w]', '_', str(c)) for c in X_train_fe_eval.columns]
    
    # 对齐特征
    if hasattr(model, 'feature_names_in_'):
        for feat in model.feature_names_in_:
            if feat not in X_train_fe_eval.columns:
                X_train_fe_eval[feat] = 0
        X_train_fe_eval = X_train_fe_eval[model.feature_names_in_]
else:
    X_train_fe_eval = X_test_fe  # 回退

# 获取真实标签
if target_col in train.columns:
    y_train_true = train[target_col].values
else:
    y_train_true = None

# 预测训练集
if y_train_true is not None:
    try:
        train_preds_encoded = model.predict(X_train_fe_eval)
        if label_encoder is not None:
            train_preds = label_encoder.inverse_transform(train_preds_encoded)
        else:
            train_preds = train_preds_encoded
        # 统一类型为字符串
        train_preds = np.array(train_preds).astype(str)
        y_train_true_str = np.array(y_train_true).astype(str)
        
        # 计算指标
        acc = accuracy_score(y_train_true_str, train_preds)
        f1_macro = f1_score(y_train_true_str, train_preds, average='macro')
        
        # 混淆矩阵
        cm = confusion_matrix(y_train_true_str, train_preds)
        class_labels = sorted(set(list(y_train_true_str) + list(train_preds)))
        
        # 分类报告
        report_dict = classification_report(y_train_true_str, train_preds, output_dict=True)
        
        # 计算Log Loss
        try:
            train_probs = model.predict_proba(X_train_fe_eval)
            # 需要将字符串标签编码为数字以计算log loss
            le_for_logloss = LabelEncoder()
            y_train_encoded_for_loss = le_for_logloss.fit_transform(y_train_true_str)
            ll = log_loss(y_train_encoded_for_loss, train_probs)
        except Exception:
            ll = None
        
        metrics_summary = {
            'accuracy': round(acc, 4),
            'f1_macro': round(f1_macro, 4),
            'log_loss': round(ll, 4) if ll is not None else 'N/A',
            'num_classes': len(class_labels),
            'classes': class_labels
        }
    except Exception as e:
        print(f"训练集评估出错: {e}")
        metrics_summary = {'error': str(e)}
        cm = None
        report_dict = {}
        class_labels = []
else:
    metrics_summary = {'note': '训练集无目标列，无法计算指标'}
    cm = None
    report_dict = {}
    class_labels = []

    plt.figure(figsize=(max(8, len(class_labels) * 1.2), max(6, len(class_labels) * 1.0)))
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', 
                xticklabels=class_labels, yticklabels=class_labels)
    plt.xlabel('Predicted')

predict_script = '''
import pandas as pd
import numpy as np
import pickle
import dill
import re
import os
import warnings
warnings.filterwarnings('ignore')

def predict(new_data_path, output_path='predictions.csv'):
    """
    加载模型并对新数据进行预测。
    
    参数:
        new_data_path: str, 新数据CSV文件路径
        output_path: str, 预测结果输出路径
    """
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
        model = dill.load(f)
    
    # 加载数据
    df = pd.read_csv(new_data_path)
    original_df = df.copy()
    
    # 预处理
    df = df.drop(columns=['id'], errors='ignore')
    df.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in df.columns]
    
    # 处理目标列（如果存在）
    target_col = 'Status'
    if target_col in df.columns:
        df = df.drop(columns=[target_col])
    
    # 确保所有列都是数值类型
    for col in df.columns:
        if df[col].dtype == 'object' or df[col].dtype.name == 'category':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    df = df.fillna(0)
    
    # 对齐特征
    if hasattr(model, 'feature_names_in_'):
        for feat in model.feature_names_in_:
            if feat not in df.columns:
                df[feat] = 0
        df = df[model.feature_names_in_]
    
    # 预测
    predictions = model.predict(df)
    
    # 尝试获取概率
    try:
        probabilities = model.predict_proba(df)
    except Exception:
        probabilities = None
    
    # 构建结果
    id_col = 'id' if 'id' in original_df.columns else None
    result = pd.DataFrame()
    if id_col:
        result['id'] = original_df[id_col]
    else:
        result['id'] = range(len(predictions))
    
    result['prediction'] = predictions
    
    if probabilities is not None:
        for i in range(probabilities.shape[1]):
            result[f'proba_class_{i}'] = probabilities[:, i]
    
    result.to_csv(output_path, index=False)
    print(f"预测完成，结果已保存到 {output_path}")
    return result

if __name__ == '__main__':
    import sys
    if len(sys.argv) > 1:
        input_file = sys.argv[1]
        output_file = sys.argv[2] if len(sys.argv) > 2 else 'predictions.csv'
        predict(input_file, output_file)
    else:
        print("用法: python predict.py <输入文件路径> [输出文件路径]")
'''
