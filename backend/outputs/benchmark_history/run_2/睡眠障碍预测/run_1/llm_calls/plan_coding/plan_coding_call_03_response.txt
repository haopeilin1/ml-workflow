def preprocess(df, mode='test'):
    df = df.copy()
    
    # 剔除ID特征
    id_cols = [
        'Person ID',
        'Sleep Duration (hours)',
        'Quality of Sleep (scale: 1-10)',
        'Physical Activity Level (minutes/day)',
        'Stress Level (scale: 1-10)',
        'Heart Rate (bpm)',
        'Daily Steps'
    ]
    existing_id_cols = [c for c in id_cols if c in df.columns]
    if existing_id_cols:
        df = df.drop(columns=existing_id_cols)
    
    # 拆解血压特征
    bp_col = PREPROCESS_STATE['bp_col']
    if bp_col in df.columns:
        bp_split = df[bp_col].str.split('/', expand=True)
        df['Systolic BP'] = pd.to_numeric(bp_split[0], errors='coerce')
        df['Diastolic BP'] = pd.to_numeric(bp_split[1], errors='coerce')
        df = df.drop(columns=[bp_col])
    
    # 处理缺失值
    num_cols = df.select_dtypes(include=['int64', 'float64']).columns.tolist()
    cat_cols = df.select_dtypes(include=['object', 'category']).columns.tolist()
    if target_col in num_cols:
        num_cols.remove(target_col)
    if target_col in cat_cols:
        cat_cols.remove(target_col)
    
    for col in num_cols:
        if col in df.columns and df[col].isnull().any():
            df[col] = df[col].fillna(df[col].median())
    for col in cat_cols:
        if col in df.columns and df[col].isnull().any():
            df[col] = df[col].fillna(df[col].mode()[0] if not df[col].mode().empty else 'Unknown')
    
    return df

def feature_engineering(df, preprocessor=None):
    if target_col in df.columns:
        X = df.drop(columns=[target_col])
    else:
        X = df.copy()
    
    cat_cols = X.select_dtypes(include=['object', 'category']).columns.tolist()
    num_cols = X.select_dtypes(exclude=['object', 'category']).columns.tolist()

        X_processed = preprocessor.transform(X)
    else:
        # 如果没有预处理器（独立预测场景），尝试从模型Pipeline中获取
        raise ValueError("Preprocessor is required for feature engineering")
    
    # 获取特征名
    feature_names = []
    for name, _, cols in preprocessor.transformers_:
        if name == 'gender':
            feature_names.append('Gender_encoded')
        elif name == 'cat':
            encoder = preprocessor.named_transformers_['cat'].named_steps['encoder']
            feature_names.extend(encoder.get_feature_names_out().tolist())
        elif name == 'num':
            feature_names.extend(cols)
    
    feature_names = [re.sub(r'[^\w]', '_', str(c)) for c in feature_names]
    X_processed = pd.DataFrame(X_processed, columns=feature_names, index=X.index)
    
    return X_processed

# -*- coding: utf-8 -*-
"""
独立预测脚本
用法: python predict.py <input_csv_path> [output_csv_path]
默认输出到 output/predictions.csv
"""

import pandas as pd
import numpy as np
import pickle
import re
import sys
import os
import warnings
warnings.filterwarnings('ignore')

TARGET_COL = 'Sleep Disorder'
BP_COL = 'Blood Pressure (systolic/diastolic)'
ID_COLS = [
    'Person ID', 'Sleep Duration (hours)', 'Quality of Sleep (scale: 1-10)',
    'Physical Activity Level (minutes/day)', 'Stress Level (scale: 1-10)',
    'Heart Rate (bpm)', 'Daily Steps'
]

def preprocess(df):
    """预处理函数，与训练阶段保持一致"""
    df = df.copy()
    
    # 删除ID列
    existing_id_cols = [c for c in ID_COLS if c in df.columns]
    if existing_id_cols:
        df = df.drop(columns=existing_id_cols)
    
    # 拆解血压
    if BP_COL in df.columns:
        bp_split = df[BP_COL].str.split('/', expand=True)
        df['Systolic BP'] = pd.to_numeric(bp_split[0], errors='coerce')
        df['Diastolic BP'] = pd.to_numeric(bp_split[1], errors='coerce')
        df = df.drop(columns=[BP_COL])
    
    # 处理缺失值
    num_cols = df.select_dtypes(include=['int64', 'float64']).columns.tolist()
    cat_cols = df.select_dtypes(include=['object', 'category']).columns.tolist()
    if TARGET_COL in num_cols:
        num_cols.remove(TARGET_COL)
    if TARGET_COL in cat_cols:
        cat_cols.remove(TARGET_COL)
    
    for col in num_cols:
        if col in df.columns and df[col].isnull().any():
            df[col] = df[col].fillna(df[col].median())
    for col in cat_cols:
        if col in df.columns and df[col].isnull().any():
            df[col] = df[col].fillna(df[col].mode()[0] if not df[col].mode().empty else 'Unknown')
    
    return df

def main():
    if len(sys.argv) < 2:
        print("用法: python predict.py <input_csv_path> [output_csv_path]")
        sys.exit(1)
    
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else 'output/predictions.csv'
    
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    if not os.path.exists(model_path):
        model_path = 'output/model.pkl'
    
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # 加载数据
    df = pd.read_csv(input_path)
    original_df = df.copy()
    
    # 预处理
    df_clean = preprocess(df)
    
    # 提取特征
    if TARGET_COL in df_clean.columns:
        X = df_clean.drop(columns=[TARGET_COL])
    else:
        X = df_clean.copy()
    
    # 如果模型是Pipeline，直接使用
    if hasattr(model, 'predict'):
        predictions = model.predict(X)
        try:
            probabilities = model.predict_proba(X)
        except Exception:
            probabilities = None
    else:
        raise ValueError("Model does not have predict method")
    
    # 构建输出
    result = original_df.copy()
    result['prediction'] = predictions
    
    if probabilities is not None:
        for i in range(probabilities.shape[1]):
            result[f'proba_{i}'] = probabilities[:, i]
    
    os.makedirs(os.path.dirname(output_path) if os.path.dirname(output_path) else '.', exist_ok=True)
    result.to_csv(output_path, index=False)
    print(f"Predictions saved to {output_path}")

if __name__ == '__main__':
    main()
'''
