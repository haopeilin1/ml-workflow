# ========== 预处理函数 ==========
def preprocess(df, mode='train'):
    df = df.copy()
    
    id_cols = [
        'Person ID',
        'Sleep Duration (hours)',
        'Quality of Sleep (scale: 1-10)',
        'Physical Activity Level (minutes/day)',
        'Stress Level (scale: 1-10)',
        'Heart Rate (bpm)',
        'Daily Steps'
    ]
    existing_id_cols = [c for c in id_cols if c in df.columns]
    if existing_id_cols:
        df = df.drop(columns=existing_id_cols)
    
    bp_col = PREPROCESS_STATE['bp_col']
    if bp_col in df.columns:
        bp_split = df[bp_col].str.split('/', expand=True)
        df['Systolic BP'] = pd.to_numeric(bp_split[0], errors='coerce')
        df['Diastolic BP'] = pd.to_numeric(bp_split[1], errors='coerce')
        df = df.drop(columns=[bp_col])
    
    num_cols = df.select_dtypes(include=['int64', 'float64']).columns.tolist()
    cat_cols = df.select_dtypes(include=['object', 'category']).columns.tolist()
    target_col = PREPROCESS_STATE['target_col']
    if target_col in num_cols:
        num_cols.remove(target_col)
    if target_col in cat_cols:
        cat_cols.remove(target_col)
    
    for col in num_cols:
        if df[col].isnull().any():
            df[col] = df[col].fillna(df[col].median())
    for col in cat_cols:
        if df[col].isnull().any():
            df[col] = df[col].fillna(df[col].mode()[0] if not df[col].mode().empty else 'Unknown')
    
    if mode == 'train':
        PREPROCESS_STATE['fitted'] = True
    
    return df


def feature_engineering(df):
    target_col = PREPROCESS_STATE['target_col']
    
    if target_col in df.columns:
        X = df.drop(columns=[target_col])
    else:
        X = df.copy()
    
    cat_cols = X.select_dtypes(include=['object', 'category']).columns.tolist()
    num_cols = X.select_dtypes(exclude=['object', 'category']).columns.tolist()
    
    gender_col = 'Gender' if 'Gender' in cat_cols else None
    other_cat_cols = [c for c in cat_cols if c != gender_col]
    
    preprocessor = PREPROCESS_STATE.get('preprocessor', None)
    
    if preprocessor is None:
        transformers = []
        
        if gender_col:
            transformers.append(
                ('gender', Pipeline([
                    ('imputer', SimpleImputer(strategy='most_frequent')),
                    ('encoder', OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1))
                ]), [gender_col])
            )
        
        if other_cat_cols:
            transformers.append(
                ('cat', Pipeline([
                    ('imputer', SimpleImputer(strategy='most_frequent')),
                    ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
                ]), other_cat_cols)
            )
        
        if num_cols:
            transformers.append(
                ('num', Pipeline([
                    ('imputer', SimpleImputer(strategy='median')),
                    ('scaler', StandardScaler())
                ]), num_cols)
            )
        
        preprocessor = ColumnTransformer(transformers, remainder='drop')
        X_processed = preprocessor.fit_transform(X)
        PREPROCESS_STATE['preprocessor'] = preprocessor
    else:
        X_processed = preprocessor.transform(X)
    
    feature_names = []
    for name, _, cols in preprocessor.transformers_:
        if name == 'gender':
            feature_names.append('Gender_encoded')
        elif name == 'cat':
            encoder = preprocessor.named_transformers_['cat'].named_steps['encoder']
            feature_names.extend(encoder.get_feature_names_out(other_cat_cols).tolist())
        elif name == 'num':
            feature_names.extend(cols)
    
    feature_names = [re.sub(r'[^\w]', '_', str(c)) for c in feature_names]
    X_processed = pd.DataFrame(X_processed, columns=feature_names, index=X.index)
    
    return X_processed


def build_model():
    model = LGBMClassifier(
        objective='multiclass',
        num_class=3,
        metric='multi_logloss',
        is_unbalance=True,
        max_depth=4,
        num_leaves=15,
        learning_rate=0.05,
        n_estimators=500,
        subsample=0.7,
        colsample_bytree=0.7,
        min_child_samples=10,
        min_split_gain=0.01,
        reg_alpha=0.5,
        reg_lambda=0.5,
        random_state=42,
        verbose=-1,
    )
    return model

# ========== 特征工程 ==========

for _df in [X_train_fe, X_test_fe]:
    _df.columns = [re.sub('[^\\w]', '_', str(c)) for c in _df.columns]
for _df in [X_train_fe, X_test_fe]:

# ========== 加载或训练模型 ==========
model = None
model_loaded = False

if model is None:

predict_script = '''#!/usr/bin/env python3
"""独立预测脚本：加载模型对新数据进行预测"""
import pandas as pd
import numpy as np
import pickle
import re
import os
import sys
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, OrdinalEncoder, StandardScaler
from sklearn.impute import SimpleImputer

PREPROCESS_STATE = {
    'bp_col': 'Blood Pressure (systolic/diastolic)',
    'target_col': 'Sleep Disorder',
    'preprocessor': None,
    'label_encoder': None
}

def preprocess(df):
    df = df.copy()
    id_cols = [
        'Person ID', 'Sleep Duration (hours)', 'Quality of Sleep (scale: 1-10)',
        'Physical Activity Level (minutes/day)', 'Stress Level (scale: 1-10)',
        'Heart Rate (bpm)', 'Daily Steps'
    ]
    existing_id_cols = [c for c in id_cols if c in df.columns]
    if existing_id_cols:
        df = df.drop(columns=existing_id_cols)
    
    bp_col = PREPROCESS_STATE['bp_col']
    if bp_col in df.columns:
        bp_split = df[bp_col].str.split('/', expand=True)
        df['Systolic BP'] = pd.to_numeric(bp_split[0], errors='coerce')
        df['Diastolic BP'] = pd.to_numeric(bp_split[1], errors='coerce')
        df = df.drop(columns=[bp_col])
    
    num_cols = df.select_dtypes(include=['int64', 'float64']).columns.tolist()
    cat_cols = df.select_dtypes(include=['object', 'category']).columns.tolist()
    target_col = PREPROCESS_STATE['target_col']
    if target_col in num_cols:
        num_cols.remove(target_col)
    if target_col in cat_cols:
        cat_cols.remove(target_col)
    
    for col in num_cols:
        if df[col].isnull().any():
            df[col] = df[col].fillna(df[col].median())
    for col in cat_cols:
        if df[col].isnull().any():
            df[col] = df[col].fillna(df[col].mode()[0] if not df[col].mode().empty else 'Unknown')
    
    return df


def feature_engineering(df):
    target_col = PREPROCESS_STATE['target_col']
    if target_col in df.columns:
        X = df.drop(columns=[target_col])
    else:
        X = df.copy()
    
    preprocessor = PREPROCESS_STATE.get('preprocessor', None)
    if preprocessor is None:
        raise ValueError("Preprocessor not loaded. Load model.pkl first.")
    
    X_processed = preprocessor.transform(X)
    
    feature_names = []
    for name, _, cols in preprocessor.transformers_:
        if name == 'gender':
            feature_names.append('Gender_encoded')
        elif name == 'cat':
            encoder = preprocessor.named_transformers_['cat'].named_steps['encoder']
            other_cat_cols = [c for c in X.select_dtypes(include=['object', 'category']).columns if c != 'Gender']
            feature_names.extend(encoder.get_feature_names_out(other_cat_cols).tolist())
        elif name == 'num':
            feature_names.extend(cols)
    
    feature_names = [re.sub(r'[^\\w]', '_', str(c)) for c in feature_names]
    X_processed = pd.DataFrame(X_processed, columns=feature_names, index=X.index)
    return X_processed


def load_model(model_path='output/model.pkl'):
    """加载模型和预处理器"""
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    return model


def predict(model, data_path, output_path=None):
    """对新数据进行预测"""
    df = pd.read_csv(data_path)
    df_clean = preprocess(df)
    X = feature_engineering(df_clean)

    preds = model.predict(X)
    probs = model.predict_proba(X)
    
    result = df.copy()
    result['prediction'] = preds
    
    for i in range(probs.shape[1]):
        result[f'proba_class_{i}'] = probs[:, i]
    
    if output_path:
        result.to_csv(output_path, index=False)
        print(f"Predictions saved to {output_path}")
    
    return result


if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python predict.py <input_csv> [output_csv]")
        sys.exit(1)
    
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else 'predictions.csv'
    
    model = load_model()
