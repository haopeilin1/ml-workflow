print("加载测试集...")

# 从最佳代码中复制预处理逻辑
def preprocess_test(df):
    """对测试集进行与训练时相同的预处理"""
    df = df.copy()
    
    # 剔除特征列
    drop_cols = ['instant', 'casual', 'registered']
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    # 从 dteday 提取时间特征
    if 'dteday' in df.columns:
        df['dteday'] = pd.to_datetime(df['dteday'], errors='coerce')
        df['year'] = df['dteday'].dt.year
        df['month'] = df['dteday'].dt.month
        df['day'] = df['dteday'].dt.day
        df['dayofweek'] = df['dteday'].dt.dayofweek
        df['is_weekend'] = (df['dayofweek'] >= 5).astype(int)
        df = df.drop(columns=['dteday'])
    
    return df

def feature_engineering_test(df):
    """对测试集进行特征工程"""
    X = df.copy()
    
    # 确保所有列是数值类型
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce')
    
    # 创建周期性特征
    if 'hr' in X.columns:
        X['hr_sin'] = np.sin(X['hr'] * 2 * np.pi / 24)
        X['hr_cos'] = np.cos(X['hr'] * 2 * np.pi / 24)
    
    # 创建交互特征
    if 'temp' in X.columns and 'hum' in X.columns:
        X['temp_hum'] = X['temp'] * X['hum']
    if 'weathersit' in X.columns and 'windspeed' in X.columns:
        X['weathersit_windspeed'] = X['weathersit'] * X['windspeed']
    
    # 滞后特征（测试集用0填充）
    X['cnt_lag1'] = 0
    X['cnt_lag2'] = 0
    
    # 填充缺失值
    X = X.fillna(0)
    
    # 确保所有列是数值类型
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    
    return X

# 预处理测试集
test_clean = preprocess_test(X_test)

# 清洗特征名
X_test_fe.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_test_fe.columns]
if X_test_fe.columns.duplicated().any():
    X_test_fe.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X_test_fe.columns)]

# ========== 4. 模型预测 ==========
print("进行预测...")
# 检查模型是否是 Pipeline（包含预处理）
if hasattr(model, 'predict'):
    # 尝试直接对原始测试集预测（如果是完整 Pipeline）
    try:
        test_preds = model.predict(X_test_fe)
    except Exception as e1:
        print(f"直接预测失败: {e1}，尝试使用原始特征...")
        try:
            test_preds = model.predict(X_test)
        except Exception as e2:
            raise RuntimeError(f"预测失败: {e2}")
else:
    raise RuntimeError("模型没有 predict 方法")

# 如果模型做了 log1p 变换，需要逆变换
# 检查预测值范围判断是否需要逆变换
if test_preds.max() < 20:  # log1p 变换后的值通常较小
    test_preds = np.expm1(test_preds)

print(f"预测完成，预测值范围: [{test_preds.min():.2f}, {test_preds.max():.2f}]")

# ========== 5. 保存模型到 output/ ==========
print("保存模型到 output/model.pkl...")

# ========== 6. 生成特征重要性 ==========
print("生成特征重要性...")
feature_names = X_test_fe.columns.tolist()

# 尝试从模型中获取特征重要性
importance = None
if hasattr(model, 'feature_importances_'):
    importance = model.feature_importances_
elif hasattr(model, 'named_steps'):
    # 如果是 Pipeline，尝试从最后一步获取
    for step_name in ['model', 'regressor', 'classifier', 'estimator']:
        if step_name in model.named_steps:
            step = model.named_steps[step_name]
            if hasattr(step, 'feature_importances_'):
                importance = step.feature_importances_
                break
    # 如果 Pipeline 中有 ColumnTransformer，尝试获取特征名
    if hasattr(model, 'named_steps'):
        for step_name in model.named_steps:
            step = model.named_steps[step_name]
            if hasattr(step, 'get_feature_names_out'):
                try:
                    feature_names = step.get_feature_names_out().tolist()
                except:
                    pass

if importance is None:
    # 如果无法获取特征重要性，使用均匀分布
    print("警告: 无法从模型获取特征重要性，使用均匀分布")
    importance = np.ones(len(feature_names)) / len(feature_names)

# 确保长度匹配
if len(importance) != len(feature_names):
    print(f"警告: 特征重要性长度({len(importance)})与特征名长度({len(feature_names)})不匹配")
    min_len = min(len(importance), len(feature_names))

fi_df = pd.DataFrame({

predict_script = '''
import pandas as pd
import numpy as np
import pickle
import re
import os
import sys

def preprocess(df):
    """预处理函数"""
    df = df.copy()
    drop_cols = ['instant', 'casual', 'registered']
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    if 'dteday' in df.columns:
        df['dteday'] = pd.to_datetime(df['dteday'], errors='coerce')
        df['year'] = df['dteday'].dt.year
        df['month'] = df['dteday'].dt.month
        df['day'] = df['dteday'].dt.day
        df['dayofweek'] = df['dteday'].dt.dayofweek
        df['is_weekend'] = (df['dayofweek'] >= 5).astype(int)
        df = df.drop(columns=['dteday'])
    return df

def feature_engineering(df):
    """特征工程函数"""
    X = df.copy()
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce')
    if 'hr' in X.columns:
        X['hr_sin'] = np.sin(X['hr'] * 2 * np.pi / 24)
        X['hr_cos'] = np.cos(X['hr'] * 2 * np.pi / 24)
    if 'temp' in X.columns and 'hum' in X.columns:
        X['temp_hum'] = X['temp'] * X['hum']
    if 'weathersit' in X.columns and 'windspeed' in X.columns:
        X['weathersit_windspeed'] = X['weathersit'] * X['windspeed']
    X['cnt_lag1'] = 0
    X['cnt_lag2'] = 0
    X = X.fillna(0)
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    return X

def predict(input_path, output_path=None):
    """
    对新数据进行预测
    input_path: 输入 CSV 文件路径
    output_path: 输出 CSV 文件路径（可选，默认在输入文件同目录下生成 predictions.csv）
    """
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # 加载数据
    df = pd.read_csv(input_path)
    
    # 预处理
    df_clean = preprocess(df)
    X = feature_engineering(df_clean)
    
    # 清洗特征名
    X.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X.columns]
    if X.columns.duplicated().any():
        X.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X.columns)]
    
    # 预测
    preds = model.predict(X)
    if preds.max() < 20:
        preds = np.expm1(preds)
    
    # 保存结果
    result = df.copy()
    result['prediction'] = preds
    
    if output_path is None:
        output_path = os.path.join(os.path.dirname(input_path), 'predictions.csv')
    result.to_csv(output_path, index=False)
    print(f"预测结果已保存到: {output_path}")
    return result

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("用法: python predict.py <input_csv_path> [output_csv_path]")
        sys.exit(1)
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else None
    predict(input_path, output_path)
'''

with open('output/predict.py', 'w', encoding='utf-8') as f:
    f.write(predict_script)
