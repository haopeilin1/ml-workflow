import numpy as np
import pandas as pd
import re
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from lightgbm import LGBMClassifier, early_stopping

# 全局状态：用于在 preprocess 的 train 和 test 模式间共享参数
PREPROCESS_STATE = {
    'log1p_cols': None,  # 需要 log1p 变换的列名列表
    'scaler': None,      # StandardScaler 实例
}

def preprocess(df, mode='train'):
    '''
    数据清洗和预处理。
    mode='train' 时拟合参数，保存到 PREPROCESS_STATE。
    mode='test' 时应用已保存的参数。
    返回处理后的 DataFrame（仍包含目标列）。
    '''
    # 1. 丢弃 'id' 列（must_do #1）
    df = df.drop(columns=['id'], errors='ignore')
    
    # 2. 清洗特征名：替换特殊字符（防止 LightGBM 报 JSON 特殊字符错误）
    df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
    
    # 3. 确定目标列
    target_col = 'spam'
    
    # 4. 获取所有数值特征列（排除目标列）
    feature_cols = [c for c in df.columns if c != target_col]
    
    # 5. 确保所有特征列是数值类型（防御性编程）
    for col in feature_cols:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    if mode == 'train':
        # 训练模式：确定需要 log1p 变换的列，并拟合 scaler
        
        # 对所有特征列应用 log1p 变换（must_do #3：缓解右偏分布）
        # 由于所有特征都是非负的（min >= 0），log1p 是安全的
        PREPROCESS_STATE['log1p_cols'] = feature_cols.copy()
        
        for col in feature_cols:
            # 防御性处理：确保没有负值（log1p 要求 x >= -1）
            min_val = df[col].min()
            if min_val < 0:
                # 如果有负值，先平移使最小值为 0，再 log1p
                df[col] = np.log1p(df[col] - min_val)
            else:
                df[col] = np.log1p(df[col])
        
        # 拟合 StandardScaler（must_do #2：消除特征尺度差异）
        scaler = StandardScaler()
        df[feature_cols] = scaler.fit_transform(df[feature_cols])
        PREPROCESS_STATE['scaler'] = scaler
        
    else:
        # 测试模式：应用训练时保存的参数
        
        log1p_cols = PREPROCESS_STATE.get('log1p_cols', feature_cols)
        scaler = PREPROCESS_STATE.get('scaler', None)

                if min_val < 0:
                    df[col] = np.log1p(df[col] - min_val)
                else:
                    df[col] = np.log1p(df[col])
        
        # 应用 StandardScaler
        if scaler is not None:
            # 只 transform 实际存在的列
            cols_to_scale = [c for c in log1p_cols if c in df.columns]
            if cols_to_scale:
                df[cols_to_scale] = scaler.transform(df[cols_to_scale])
    
    return df


def feature_engineering(df):
    '''
    特征工程。
    返回特征矩阵 X（必须不含目标列，所有列必须是数值类型）。
    
    防御性检查：处理无穷大值和 NaN 值。
    '''
    target_col = 'spam'
    
    # 分离特征和目标
    X = df.drop(columns=[target_col], errors='ignore')
    
    # 防御性编程：处理无穷大值
    X = X.replace([np.inf, -np.inf], np.nan)
    
    # 防御性编程：处理 NaN 值（用 0 填充，因为 log1p 变换后 0 是合理的默认值）
    if X.isnull().any().any():
        X = X.fillna(0)
    
    # 确保所有列都是数值类型
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    
    return X


def build_model():
    '''
    模型构建和超参数设置。
    返回 sklearn 兼容的模型对象（支持 fit/predict/predict_proba）。
    
    使用 LightGBM 作为主模型，带正则化参数防止过拟合。
    '''
    # 数据集 3680 行，不属于极小样本（>= 500），可以使用 LightGBM
    # 但需要设置合理的正则化参数防止过拟合
    
    model = LGBMClassifier(
        objective='binary',
        metric='auc',
        boosting_type='gbdt',
        random_state=42,
        n_estimators=300,           # 适中的树数量
        max_depth=6,                # 限制树深度防止过拟合（must_do 过拟合控制）
        num_leaves=31,              # 默认值，与 max_depth 配合
        learning_rate=0.05,         # 较小的学习率

        min_child_samples=20,       # 叶子节点最小样本数（must_do 过拟合控制）
        reg_alpha=0.1,              # L1 正则化
        reg_lambda=0.1,             # L2 正则化
        verbose=-1,                 # 静默模式
    )
    
    return model
