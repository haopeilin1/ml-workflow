import pandas as pd

# ========== 预处理函数（与最佳代码完全一致）==========
def preprocess(df, mode='train'):
    df = df.drop(columns=['id'], errors='ignore')
    df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
    target_col = 'spam'
    feature_cols = [c for c in df.columns if c != target_col]
    
    for col in feature_cols:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    if mode == 'train':
        PREPROCESS_STATE['log1p_cols'] = feature_cols.copy()
        for col in feature_cols:
            min_val = df[col].min()
            if min_val < 0:
                df[col] = np.log1p(df[col] - min_val)
            else:
                df[col] = np.log1p(df[col])
        from sklearn.preprocessing import StandardScaler
        scaler = StandardScaler()
        df[feature_cols] = scaler.fit_transform(df[feature_cols])
        PREPROCESS_STATE['scaler'] = scaler
    else:
        log1p_cols = PREPROCESS_STATE.get('log1p_cols', feature_cols)
        scaler = PREPROCESS_STATE.get('scaler', None)
        for col in log1p_cols:
            if col in df.columns:
                min_val = df[col].min()
                if min_val < 0:
                    df[col] = np.log1p(df[col] - min_val)
                else:
                    df[col] = np.log1p(df[col])
        if scaler is not None:
            cols_to_scale = [c for c in log1p_cols if c in df.columns]
            if cols_to_scale:
                df[cols_to_scale] = scaler.transform(df[cols_to_scale])
    return df

def feature_engineering(df):
    target_col = 'spam'
    X = df.drop(columns=[target_col], errors='ignore')
    X = X.replace([np.inf, -np.inf], np.nan)
    if X.isnull().any().any():
        X = X.fillna(0)
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    return X

# ========== 加载数据 ==========
train = pd.read_csv('data/train.csv')

target_col = 'spam'

# ========== 尝试加载已有模型 ==========
model = None
model_loaded = False
if os.path.exists('data/best_model.pkl'):
    try:
        with open('data/best_model.pkl', 'rb') as f:
            model = pickle.load(f)
        model_loaded = True
        print("成功从 data/best_model.pkl 加载模型")
    except Exception as e:
        print(f"加载 data/best_model.pkl 失败: {e}，将重新训练")

# ========== 如果加载失败，重新训练 ==========
if not model_loaded:
    print("重新训练模型...")
    from lightgbm import LGBMClassifier, early_stopping
    from sklearn.preprocessing import StandardScaler
    
    # 预处理训练集
    train_clean = preprocess(train, mode='train')
    
    if target_col in train_clean.columns:
        y_train = train_clean[target_col]
        X_train = train_clean.drop(columns=[target_col])
    else:
        y_train = train[target_col]
        X_train = train_clean
    
    # 特征工程
    X_train_fe = feature_engineering(X_train)
    if isinstance(X_train_fe, np.ndarray):
        X_train_fe = pd.DataFrame(X_train_fe, index=X_train.index)
    X_train_fe.columns = [re.sub('[^\\w]', '_', str(c)) for c in X_train_fe.columns]
    
    # 构建并训练模型
    model = LGBMClassifier(
        objective='binary',
        metric='auc',
        boosting_type='gbdt',
        random_state=42,
        n_estimators=300,
        max_depth=6,
        num_leaves=31,
        learning_rate=0.05,
        subsample=0.8,
        colsample_bytree=0.8,
        min_child_samples=20,
        reg_alpha=0.1,
        reg_lambda=0.1,
        verbose=-1,
    )
    model.fit(X_train_fe, y_train)
    print("模型重新训练完成")

# ========== 预处理测试集 ==========
test_clean = preprocess(test, mode='test')

if X_test is test_clean:
    X_test = test_clean.copy()

if isinstance(X_test_fe, np.ndarray):
    X_test_fe = pd.DataFrame(X_test_fe, index=X_test.index)
X_test_fe.columns = [re.sub('[^\\w]', '_', str(c)) for c in X_test_fe.columns]

# ========== 测试集预测 ==========
if hasattr(model, 'predict_proba'):

# ========== 计算验证集指标（用于报告）==========
# 使用训练集的一部分作为验证集来评估指标
train_clean_for_eval = preprocess(train, mode='test')
if target_col in train_clean_for_eval.columns:

else:
    y_eval = train[target_col]
    X_eval = train_clean_for_eval

X_eval_fe = feature_engineering(X_eval)
if isinstance(X_eval_fe, np.ndarray):
    X_eval_fe = pd.DataFrame(X_eval_fe, index=X_eval.index)
X_eval_fe.columns = [re.sub('[^\\w]', '_', str(c)) for c in X_eval_fe.columns]

if hasattr(model, 'predict_proba'):
    eval_probs = model.predict_proba(X_eval_fe)[:, 1]
else:
    eval_probs = model.predict(X_eval_fe).astype(float)
eval_preds = (eval_probs >= 0.5).astype(int)

# 计算指标
eval_auc = float(roc_auc_score(y_eval, eval_probs))

predict_script = '''
import pandas as pd
import numpy as np
import pickle
import re
import os

# 预处理状态（与训练时一致）
PREPROCESS_STATE = {
    'log1p_cols': None,
    'scaler': None,
}

def preprocess(df, mode='test'):
    df = df.drop(columns=['id'], errors='ignore')
    df.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in df.columns]
    target_col = 'spam'
    feature_cols = [c for c in df.columns if c != target_col]
    
    for col in feature_cols:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    log1p_cols = PREPROCESS_STATE.get('log1p_cols', feature_cols)
    scaler = PREPROCESS_STATE.get('scaler', None)
    
    for col in log1p_cols:
        if col in df.columns:
            min_val = df[col].min()
            if min_val < 0:
                df[col] = np.log1p(df[col] - min_val)
            else:
                df[col] = np.log1p(df[col])
    
    if scaler is not None:
        cols_to_scale = [c for c in log1p_cols if c in df.columns]
        if cols_to_scale:
            df[cols_to_scale] = scaler.transform(df[cols_to_scale])
    return df

def feature_engineering(df):
    target_col = 'spam'
    X = df.drop(columns=[target_col], errors='ignore')
    X = X.replace([np.inf, -np.inf], np.nan)
    if X.isnull().any().any():
        X = X.fillna(0)
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    return X

def predict(new_data_path, output_path='predictions.csv'):
    """对新数据进行预测"""
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # 加载数据
    df = pd.read_csv(new_data_path)
    
    # 预处理
    df_clean = preprocess(df, mode='test')
    X = df_clean.drop(columns=['spam'], errors='ignore')
    if X is df_clean:
        X = df_clean.copy()
    
    # 特征工程
    X_fe = feature_engineering(X)
    if isinstance(X_fe, np.ndarray):
        X_fe = pd.DataFrame(X_fe, index=X.index)
    X_fe.columns = [re.sub('[^\\w]', '_', str(c)) for c in X_fe.columns]
    
    # 预测
    if hasattr(model, 'predict_proba'):
        probs = model.predict_proba(X_fe)[:, 1]
    else:
        probs = model.predict(X_fe).astype(float)
    preds = (probs >= 0.5).astype(int)
    
    # 保存结果
    id_col = 'id' if 'id' in df.columns else df.columns[0]
    result = pd.DataFrame({
        'id': df[id_col] if id_col in df.columns else range(len(preds)),
        'prediction': preds,
        'probability': probs
    })
    result.to_csv(output_path, index=False)
    print(f"预测结果已保存到 {output_path}")
    return result

if __name__ == '__main__':
    import sys
    if len(sys.argv) < 2:
        print("用法: python predict.py <新数据文件路径> [输出文件路径]")
        sys.exit(1)
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else 'predictions.csv'
    predict(input_path, output_path)
'''
