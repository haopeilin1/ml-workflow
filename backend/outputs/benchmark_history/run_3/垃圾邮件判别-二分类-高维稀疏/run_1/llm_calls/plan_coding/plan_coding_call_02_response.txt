def preprocess(df, mode='train'):
    df = df.drop(columns=['id'], errors='ignore')
    df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
    
    target_col = 'spam'
    feature_cols = [c for c in df.columns if c != target_col]
    
    for col in feature_cols:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    if mode == 'train':
        PREPROCESS_STATE['log1p_cols'] = feature_cols.copy()
        
        for col in feature_cols:
            min_val = df[col].min()
            if min_val < 0:
                df[col] = np.log1p(df[col] - min_val)
            else:
                df[col] = np.log1p(df[col])
        
        scaler = StandardScaler()
        df[feature_cols] = scaler.fit_transform(df[feature_cols])
        PREPROCESS_STATE['scaler'] = scaler
        
    else:
        log1p_cols = PREPROCESS_STATE.get('log1p_cols')
        scaler = PREPROCESS_STATE.get('scaler')
        
        # 修复：如果 log1p_cols 为 None，使用当前特征列
        if log1p_cols is None:
            log1p_cols = feature_cols
            PREPROCESS_STATE['log1p_cols'] = log1p_cols
        
        for col in log1p_cols:
            if col in df.columns:
                min_val = df[col].min()
                if min_val < 0:
                    df[col] = np.log1p(df[col] - min_val)
                else:
                    df[col] = np.log1p(df[col])
        
        if scaler is not None:
            cols_to_scale = [c for c in log1p_cols if c in df.columns]
            if cols_to_scale:
                df[cols_to_scale] = scaler.transform(df[cols_to_scale])
    
    return df


def feature_engineering(df):
    target_col = 'spam'
    X = df.drop(columns=[target_col], errors='ignore')
    X = X.replace([np.inf, -np.inf], np.nan)
    if X.isnull().any().any():
        X = X.fillna(0)
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    return X

if target_col in train_clean.columns:
    y_train = train_clean[target_col]

_is_string_target = train[target_col].dtype == object
if _is_string_target:
    le = LabelEncoder()
    y_train = le.fit_transform(train[target_col])
    PREPROCESS_STATE['label_encoder'] = le

X_test = test_clean.drop(columns=[target_col], errors='ignore')

    X_test = test_clean.copy()

# 特征工程
X_train_fe = feature_engineering(X_train)

    X_train_fe = pd.DataFrame(X_train_fe, index=X_train.index)

    _df.columns = [re.sub('[^\\w]', '_', str(c)) for c in _df.columns]
for _df in [X_train_fe, X_test_fe]:

import re

# 预处理状态（从训练时保存的参数）
PREPROCESS_STATE = {

def preprocess(df, mode='test'):
    df = df.drop(columns=['id'], errors='ignore')

    for col in log1p_cols:

def feature_engineering(df):
    target_col = 'spam'
    X = df.drop(columns=[target_col], errors='ignore')
    X = X.replace([np.inf, -np.inf], np.nan)
    if X.isnull().any().any():

def predict(input_path, output_path='predictions.csv'):
    """对新数据进行预测"""
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
