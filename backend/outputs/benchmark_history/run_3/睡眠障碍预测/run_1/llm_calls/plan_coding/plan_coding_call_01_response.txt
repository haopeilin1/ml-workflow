PREPROCESS_STATE = {
    'target_encoder': None,

# ========== 预处理函数（与训练时完全一致）==========
def preprocess(df, mode='train'):
    global PREPROCESS_STATE
    target_col = 'Sleep Disorder'
    
    if 'Person ID' in df.columns:
        df = df.drop(columns=['Person ID'])
    
    drop_cols = [
        'Sleep Duration (hours)',
        'Quality of Sleep (scale: 1-10)',
        'Physical Activity Level (minutes/day)',
        'Stress Level (scale: 1-10)',
        'Heart Rate (bpm)',
        'Daily Steps'
    ]
    for col in drop_cols:
        if col in df.columns:
            df = df.drop(columns=[col])
    
    bp_col = 'Blood Pressure (systolic/diastolic)'
    if bp_col in df.columns:
        bp_split = df[bp_col].str.split('/', expand=True)
        if bp_split.shape[1] == 2:
            df['Systolic_BP'] = pd.to_numeric(bp_split[0], errors='coerce')
            df['Diastolic_BP'] = pd.to_numeric(bp_split[1], errors='coerce')
        df = df.drop(columns=[bp_col])
        PREPROCESS_STATE['bp_split_done'] = True
    
    if mode == 'train' and target_col in df.columns:
        label_order = ['No Disorder', 'Insomnia', 'Sleep Apnea']
        unique_labels = sorted(df[target_col].dropna().unique())
        encoding_map = {label: idx for idx, label in enumerate(label_order) if label in unique_labels}
        PREPROCESS_STATE['target_encoder'] = encoding_map
    
    if target_col in df.columns and PREPROCESS_STATE['target_encoder'] is not None:
        df[target_col] = df[target_col].map(PREPROCESS_STATE['target_encoder'])
        if df[target_col].isna().any():
            df[target_col] = df[target_col].fillna(-1).astype(int)
        else:
            df[target_col] = df[target_col].astype(int)
    
    return df


def feature_engineering(df):
    target_col = 'Sleep Disorder'
    X = df.drop(columns=[target_col], errors='ignore')
    
    cat_cols = X.select_dtypes(include=['object', 'category']).columns.tolist()
    num_cols = X.select_dtypes(exclude=['object', 'category']).columns.tolist()
    
    preprocessor = ColumnTransformer([
        ('cat', Pipeline([
            ('encoder', OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1))
        ]), cat_cols),
        ('num', Pipeline([
            ('scaler', StandardScaler())
        ]), num_cols)
    ], remainder='passthrough')
    
    X_processed = preprocessor.fit_transform(X)
    
    feature_names = []
    if cat_cols:
        feature_names.extend(cat_cols)
    if num_cols:
        feature_names.extend(num_cols)
    remainder_cols = [c for c in X.columns if c not in cat_cols and c not in num_cols]
    if remainder_cols:
        feature_names.extend(remainder_cols)
    
    X_processed = pd.DataFrame(X_processed, index=X.index, columns=feature_names)
    X_processed.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_processed.columns]
    
    return X_processed


# ========== 加载模型 ==========
print("Loading model from data/best_model.pkl...")
try:
    with open('data/best_model.pkl', 'rb') as f:
        model = pickle.load(f)
    print("Model loaded successfully.")
except Exception as e:
    print(f"Failed to load model: {e}")
    print("Falling back to retraining...")
    # 回退：重新训练
    from sklearn.linear_model import LogisticRegression
    train = pd.read_csv('data/train.csv')
    train_clean = preprocess(train, mode='train')
    target_col = 'Sleep Disorder'
    y_train = train_clean[target_col]
    X_train = train_clean.drop(columns=[target_col])
    X_train_fe = feature_engineering(X_train)
    model = LogisticRegression(
        solver='lbfgs',
        max_iter=2000,
        C=1.0,
        class_weight='balanced',
        random_state=42,
        n_jobs=-1
    )
    model.fit(X_train_fe, y_train)
    print("Model retrained successfully.")

# ========== 保存模型到 output/ ==========
with open('output/model.pkl', 'wb') as f:
    pickle.dump(model, f)
print("Model saved to output/model.pkl")

    X_test_fe.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X_test_fe.columns)]

# ========== 测试集预测 ==========
test_preds = model.predict(X_test_fe)

# 获取类别标签

# 构建预测结果 DataFrame
result_df = pd.DataFrame({

result_df.to_csv('output/test_predictions.csv', index=False)

# ========== 特征重要性 ==========
# LogisticRegression 的 coef_ 形状为 (n_classes, n_features)
# 取绝对值后按类别平均作为特征重要性
feature_names = X_test_fe.columns.tolist()
if hasattr(model, 'coef_'):
    importances = np.abs(model.coef_).mean(axis=0)
else:
    importances = np.ones(len(feature_names)) / len(feature_names)

# 确保长度一致
if len(importances) != len(feature_names):
    feature_names = [f'feature_{i}' for i in range(len(importances))]

importance_df = pd.DataFrame({

plt.figure(figsize=(10, 6))

    X_val_fe = feature_engineering(X_val)
    X_val_fe.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_val_fe.columns]
    if X_val_fe.columns.duplicated().any():

predict_script = '''
import pandas as pd
import numpy as np
import pickle
import re
import os
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OrdinalEncoder, StandardScaler
from sklearn.pipeline import Pipeline

def preprocess(df):
    """预处理函数 - 与训练时一致"""
    if 'Person ID' in df.columns:
        df = df.drop(columns=['Person ID'])
    
    drop_cols = [
        'Sleep Duration (hours)',
        'Quality of Sleep (scale: 1-10)',
        'Physical Activity Level (minutes/day)',
        'Stress Level (scale: 1-10)',
        'Heart Rate (bpm)',
        'Daily Steps'
    ]
    for col in drop_cols:
        if col in df.columns:
            df = df.drop(columns=[col])
    
    bp_col = 'Blood Pressure (systolic/diastolic)'
    if bp_col in df.columns:
        bp_split = df[bp_col].str.split('/', expand=True)
        if bp_split.shape[1] == 2:
            df['Systolic_BP'] = pd.to_numeric(bp_split[0], errors='coerce')
            df['Diastolic_BP'] = pd.to_numeric(bp_split[1], errors='coerce')
        df = df.drop(columns=[bp_col])
    
    return df

def feature_engineering(df):
    """特征工程函数 - 与训练时一致"""
    cat_cols = df.select_dtypes(include=['object', 'category']).columns.tolist()
    num_cols = df.select_dtypes(exclude=['object', 'category']).columns.tolist()
    
    preprocessor = ColumnTransformer([
        ('cat', Pipeline([
            ('encoder', OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1))
        ]), cat_cols),
        ('num', Pipeline([
            ('scaler', StandardScaler())
        ]), num_cols)
    ], remainder='passthrough')
    
    X_processed = preprocessor.fit_transform(df)
    
    feature_names = []
    if cat_cols:
        feature_names.extend(cat_cols)
    if num_cols:
        feature_names.extend(num_cols)
    remainder_cols = [c for c in df.columns if c not in cat_cols and c not in num_cols]
    if remainder_cols:
        feature_names.extend(remainder_cols)
    
    X_processed = pd.DataFrame(X_processed, index=df.index, columns=feature_names)
    X_processed.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X_processed.columns]
    
    return X_processed

def predict(new_data_path, output_path=None):
    """
    对新数据进行预测
    
    Parameters:
    -----------
    new_data_path : str
        新数据 CSV 文件路径
    output_path : str, optional
        输出文件路径，默认为 'predictions.csv'
    
    Returns:
    --------
    pd.DataFrame : 包含预测结果的 DataFrame
    """
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # 加载数据
    df = pd.read_csv(new_data_path)
    
    # 保存 ID 列
    id_col = 'Person ID' if 'Person ID' in df.columns else df.columns[0]
    ids = df[id_col].copy() if id_col in df.columns else range(len(df))
    
    # 预处理
    df_clean = preprocess(df)
    
    # 特征工程
    X = feature_engineering(df_clean)
    X.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X.columns]
    
    # 预测
    preds = model.predict(X)
    probs = model.predict_proba(X)
    
    # 类别标签
    class_names = ['No Disorder', 'Insomnia', 'Sleep Apnea']
    
    # 构建结果
    result = pd.DataFrame({
        'id': ids,
        'prediction': [class_names[p] for p in preds],
    })
    for i in range(probs.shape[1]):
        result[f'proba_{i}'] = probs[:, i]
    
    # 保存
    if output_path is None:
        output_path = 'predictions.csv'
    result.to_csv(output_path, index=False)
    print(f"Predictions saved to {output_path}")
    
    return result

if __name__ == '__main__':
    import sys
    if len(sys.argv) > 1:
        input_file = sys.argv[1]
        output_file = sys.argv[2] if len(sys.argv) > 2 else None
        predict(input_file, output_file)
    else:
        print("Usage: python predict.py <input_csv> [output_csv]")
'''
