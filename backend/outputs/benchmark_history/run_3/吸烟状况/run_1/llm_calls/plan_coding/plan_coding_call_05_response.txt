import pandas as pd
import numpy as np
import pickle
import json
import os
import re
import warnings

import matplotlib.pyplot as plt
from sklearn.metrics import roc_auc_score, accuracy_score, confusion_matrix, classification_report, precision_recall_curve, auc

# ========== 确保输出目录存在 ==========
os.makedirs('output', exist_ok=True)

# ========== 全局状态 ==========
PREPROCESS_STATE = {}
FEATURE_ENG_STATE = {}

# ========== 预处理函数（与训练阶段一致）==========
def preprocess(df, mode='train'):
    df = df.copy()
    
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    
    df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
    
    target_col = 'smoking'
    if target_col in df.columns:
        y = df[target_col].copy()
        X = df.drop(columns=[target_col])
    else:
        y = None
        X = df.copy()
    
    num_cols = X.columns.tolist()
    
    if mode == 'train':
        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X)
        PREPROCESS_STATE['scaler'] = scaler
        PREPROCESS_STATE['num_cols'] = num_cols
    else:
        scaler = PREPROCESS_STATE.get('scaler')
        if scaler is None:
            raise ValueError("PREPROCESS_STATE 中未找到 scaler，请先运行 train 模式")
        X_scaled = scaler.transform(X)
    
    X_scaled_df = pd.DataFrame(X_scaled, columns=num_cols, index=df.index)
    
    if y is not None:
        X_scaled_df[target_col] = y.values
    
    return X_scaled_df


def feature_engineering(df):
    df = df.copy()
    
    target_col = 'smoking'
    if target_col in df.columns:
        df = df.drop(columns=[target_col])
    
    if 'weight_kg_' in df.columns and 'height_cm_' in df.columns:
        df['bmi'] = df['weight_kg_'] / ((df['height_cm_'] / 100) ** 2)
    
    if 'systolic' in df.columns and 'relaxation' in df.columns:
        df['blood_pressure_diff'] = df['systolic'] - df['relaxation']
    
    if 'triglyceride' in df.columns and 'HDL' in df.columns:
        df['triglyceride_hdl_ratio'] = df['triglyceride'] / (df['HDL'] + 1e-6)
    
    if 'AST' in df.columns and 'ALT' in df.columns:
        df['ast_alt_ratio'] = df['AST'] / (df['ALT'] + 1e-6)
    
    if 'eyesight_left_' in df.columns and 'eyesight_right_' in df.columns:
        df['avg_eyesight'] = (df['eyesight_left_'] + df['eyesight_right_']) / 2
    
    skewed_cols = ['triglyceride', 'Gtp', 'ALT', 'fasting_blood_sugar']
    for col in skewed_cols:
        if col in df.columns:
            min_val = df[col].min()
            shift = 0 if min_val > 0 else abs(min_val) + 1
            df[f'log_{col}'] = np.log1p(df[col] + shift)
    
    for col in df.columns:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    df = df.fillna(0)
    
    return df


def build_model():
    scale_pos_weight = (1 - 0.3655) / 0.3655
    model = LGBMClassifier(
        objective='binary',
        scale_pos_weight=scale_pos_weight,
        num_leaves=31,
        max_depth=6,
        learning_rate=0.05,
        n_estimators=500,
        subsample=0.8,
        colsample_bytree=0.8,
        min_child_samples=20,
        reg_alpha=0.1,
        reg_lambda=0.1,
        random_state=42,
        verbosity=-1
    )
    return model


# ========== 加载数据 ==========
train = pd.read_csv('data/train.csv')
test = pd.read_csv('data/test.csv')

target_col = 'smoking'
id_col = 'id'
if id_col not in test.columns:
    id_col = test.columns[0]

# ========== 尝试加载已有模型 ==========
model_loaded = False
try:
    with open('data/best_model.pkl', 'rb') as f:
        model = pickle.load(f)
    model_loaded = True
    print("成功从 data/best_model.pkl 加载模型")
except Exception as e:
    print(f"加载 best_model.pkl 失败: {e}，将重新训练模型")

# ========== 预处理（train 模式拟合 scaler）==========
train_clean = preprocess(train, mode='train')
test_clean = preprocess(test, mode='test')

# 分离特征和目标
if target_col in train_clean.columns:
    y_train = train_clean[target_col]
    X_train = train_clean.drop(columns=[target_col])
else:
    y_train = train[target_col]
    X_train = train_clean

X_test = test_clean.drop(columns=[target_col], errors='ignore')
if X_test is test_clean:
    X_test = test_clean.copy()

# ========== 特征工程 ==========
X_train_fe = feature_engineering(X_train)
if isinstance(X_train_fe, np.ndarray):
    X_train_fe = pd.DataFrame(X_train_fe, index=X_train.index)

X_test_fe = feature_engineering(X_test)
if isinstance(X_test_fe, np.ndarray):
    X_test_fe = pd.DataFrame(X_test_fe, index=X_test.index)

# 清洗特征名
for _df in [X_train_fe, X_test_fe]:
    _df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in _df.columns]
    if _df.columns.duplicated().any():
        _df.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(_df.columns)]

# ========== 如果模型未加载，重新训练 ==========
if not model_loaded:
    # 加载验证集
    val = pd.read_csv('data/validation.csv')
    val_clean = preprocess(val, mode='test')
    if target_col in val_clean.columns:
        y_val = val_clean[target_col]
        X_val = val_clean.drop(columns=[target_col])
    else:
        y_val = val[target_col]
        X_val = val_clean
    
    X_val_fe = feature_engineering(X_val)
    if isinstance(X_val_fe, np.ndarray):
        X_val_fe = pd.DataFrame(X_val_fe, index=X_val.index)
    X_val_fe.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_val_fe.columns]
    if X_val_fe.columns.duplicated().any():
        X_val_fe.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X_val_fe.columns)]
    
    model = build_model()
    try:
        model.fit(X_train_fe, y_train, eval_set=[(X_val_fe, y_val)])
    except Exception:
        model.fit(X_train_fe, y_train)
    
    # 计算验证集指标
    val_probs = model.predict_proba(X_val_fe)[:, 1]
    val_preds = (val_probs >= 0.5).astype(int)
    val_auc = float(roc_auc_score(y_val, val_probs))
    val_accuracy = float(accuracy_score(y_val, val_preds))
else:
    # 使用训练集计算近似指标（测试集无标签）
    train_probs = model.predict_proba(X_train_fe)[:, 1]
    train_preds = (train_probs >= 0.5).astype(int)
    val_auc = float(roc_auc_score(y_train, train_probs))
    val_accuracy = float(accuracy_score(y_train, train_preds))

# ========== 测试集预测 ==========
test_probs = model.predict_proba(X_test_fe)[:, 1]
test_preds = (test_probs >= 0.5).astype(int)

# 保存测试预测
test_result = pd.DataFrame({
    'id': test[id_col].values if id_col in test.columns else range(len(test_preds)),
    'prediction': test_preds,
    'probability': test_probs
})
test_result.to_csv('output/test_predictions.csv', index=False)

# ========== 保存模型 ==========
with open('output/model.pkl', 'wb') as f:
    pickle.dump(model, f)

# ========== 特征重要性 ==========
feature_names = X_train_fe.columns.tolist()
if hasattr(model, 'feature_importances_'):
    importances = model.feature_importances_
else:
    importances = np.ones(len(feature_names)) / len(feature_names)

# 确保特征重要性排除目标列
feature_importance_df = pd.DataFrame({
    'name': feature_names,

feature_importance_df = feature_importance_df.sort_values('importance', ascending=False)
feature_importance_df.to_csv('output/feature_importance.csv', index=False)

plt.figure(figsize=(10, 8))

top_features = feature_importance_df.head(top_n)
plt.barh(range(top_n), top_features['importance'].values[::-1], color='steelblue')

predict_script = '''import pandas as pd
import numpy as np
import pickle
import re
import os
import sys

def preprocess(df, scaler):
    df = df.copy()
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    df.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in df.columns]
    target_col = 'smoking'
    if target_col in df.columns:
        df = df.drop(columns=[target_col])
    num_cols = df.columns.tolist()
    X_scaled = scaler.transform(df)
    X_scaled_df = pd.DataFrame(X_scaled, columns=num_cols, index=df.index)
    return X_scaled_df

def feature_engineering(df):
    df = df.copy()
    if 'weight_kg_' in df.columns and 'height_cm_' in df.columns:
        df['bmi'] = df['weight_kg_'] / ((df['height_cm_'] / 100) ** 2)
    if 'systolic' in df.columns and 'relaxation' in df.columns:
        df['blood_pressure_diff'] = df['systolic'] - df['relaxation']
    if 'triglyceride' in df.columns and 'HDL' in df.columns:
        df['triglyceride_hdl_ratio'] = df['triglyceride'] / (df['HDL'] + 1e-6)
    if 'AST' in df.columns and 'ALT' in df.columns:
        df['ast_alt_ratio'] = df['AST'] / (df['ALT'] + 1e-6)
    if 'eyesight_left_' in df.columns and 'eyesight_right_' in df.columns:
        df['avg_eyesight'] = (df['eyesight_left_'] + df['eyesight_right_']) / 2
    skewed_cols = ['triglyceride', 'Gtp', 'ALT', 'fasting_blood_sugar']
    for col in skewed_cols:
        if col in df.columns:
            min_val = df[col].min()
            shift = 0 if min_val > 0 else abs(min_val) + 1
            df[f'log_{col}'] = np.log1p(df[col] + shift)
    for col in df.columns:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    df = df.fillna(0)
    return df

def predict(input_path, output_path, model_path='output/model.pkl', scaler_path='output/scaler.pkl'):
    # 加载模型
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # 加载scaler
    with open(scaler_path, 'rb') as f:
        scaler = pickle.load(f)
    
    # 读取数据
    df = pd.read_csv(input_path)
    id_col = 'id' if 'id' in df.columns else df.columns[0]
    ids = df[id_col].values if 'id' in df.columns else range(len(df))
    
    # 预处理
    df_clean = preprocess(df, scaler)
    
    # 特征工程
    X = feature_engineering(df_clean)
    X.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X.columns]
    if X.columns.duplicated().any():
        X.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X.columns)]
    
    # 预测
    probs = model.predict_proba(X)[:, 1]
    preds = (probs >= 0.5).astype(int)
    
    # 保存结果
    result = pd.DataFrame({'id': ids, 'prediction': preds, 'probability': probs})
    result.to_csv(output_path, index=False)
    print(f"预测完成，结果保存至 {output_path}")
    return result

if __name__ == '__main__':
    if len(sys.argv) < 3:
        print("用法: python predict.py <输入CSV路径> <输出CSV路径>")
        sys.exit(1)
    predict(sys.argv[1], sys.argv[2])
'''

    ]
}
