产物生成计划：
1. 加载已存在的 data/best_model.pkl 模型
2. 加载测试集 data/test.csv 并进行与训练时相同的预处理
3. 对测试集进行预测，生成 output/test_predictions.csv（含 prediction 和 probability 列）
4. 从模型提取特征重要性，生成 output/feature_importance.csv 和 output/feature_importance.png
5. 生成 output/report.html 可视化报告
6. 保存模型到 output/model.pkl
7. 生成 output/predict.py 独立预测脚本
8. 输出摘要 JSON

关键修复：报错显示 paymentMethod 和 Category 列在测试集预测时仍是字符串类型，需要在预处理中确保这些列被正确编码。由于加载的是已保存的模型（不是 Pipeline），需要单独处理预处理步骤。

import pandas as pd
import numpy as np
import pickle
import json
import os
import re
import warnings
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, roc_auc_score, confusion_matrix, classification_report
from sklearn.preprocessing import OrdinalEncoder, LabelEncoder

warnings.filterwarnings('ignore')

# 创建输出目录
os.makedirs('output', exist_ok=True)

# ========== 全局预处理状态（与训练时保持一致）==========
PREPROCESS_STATE = {
    'isWeekend_median': None,
    'ordinal_encoder': None,
    'cat_cols': ['paymentMethod', 'Category'],
    'log_transform_cols': ['numItems', 'localTime', 'paymentMethodAgeDays'],
    'drop_cols': ['id'],
    'target_col': 'label',
    'best_threshold': 0.5,
    'label_encoder': None
}

# ========== 预处理函数（与训练时一致）==========
def preprocess(df, mode='test'):
    global PREPROCESS_STATE
    target_col = PREPROCESS_STATE['target_col']
    drop_cols = PREPROCESS_STATE['drop_cols']
    cat_cols = PREPROCESS_STATE['cat_cols']
    log_transform_cols = PREPROCESS_STATE['log_transform_cols']
    
    df = df.copy()
    
    # 1. 丢弃ID列
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    # 2. 处理Category缺失值
    if 'Category' in df.columns:
        df['Category'] = df['Category'].fillna('Unknown')
    
    # 3. 处理isWeekend缺失值
    if 'isWeekend' in df.columns:
        if mode == 'train':
            PREPROCESS_STATE['isWeekend_median'] = df['isWeekend'].median()
        df['isWeekend'] = df['isWeekend'].fillna(PREPROCESS_STATE.get('isWeekend_median', 0))
    
    # 4. log1p变换
    for col in log_transform_cols:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
            min_val = df[col].min()
            if pd.notna(min_val) and min_val < -1:
                df[col] = df[col] - min_val
            df[col] = np.log1p(df[col].clip(lower=-1))
    
    # 5. 编码类别特征
    if mode == 'train':
        encoder = OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1)
        existing_cat_cols = [c for c in cat_cols if c in df.columns]
        if existing_cat_cols:
            df[existing_cat_cols] = encoder.fit_transform(df[existing_cat_cols].astype(str))
            PREPROCESS_STATE['ordinal_encoder'] = encoder
            PREPROCESS_STATE['existing_cat_cols'] = existing_cat_cols
    else:
        encoder = PREPROCESS_STATE.get('ordinal_encoder')
        existing_cat_cols = PREPROCESS_STATE.get('existing_cat_cols', cat_cols)
        if encoder is not None:
            existing_cat_cols_in_df = [c for c in existing_cat_cols if c in df.columns]
            if existing_cat_cols_in_df:
                df[existing_cat_cols_in_df] = encoder.transform(df[existing_cat_cols_in_df].astype(str))
    
    # 6. 确保所有列数值类型
    for col in df.columns:
        if col != target_col and df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    return df


def feature_engineering(df):
    target_col = PREPROCESS_STATE['target_col']
    X = df.drop(columns=[target_col], errors='ignore')
    
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce')
    
    X = X.fillna(0)
    return X


# ========== 加载训练数据并拟合预处理参数 ==========
print("加载训练数据以拟合预处理参数...")
train = pd.read_csv('data/train.csv')
target_col = 'label'

# 拟合预处理参数
train_clean = preprocess(train, mode='train')

# 处理目标列编码
_label_encoder = None
if train[target_col].dtype == object:
    _label_encoder = LabelEncoder()
    y_train_encoded = _label_encoder.fit_transform(train[target_col])
    PREPROCESS_STATE['label_encoder'] = _label_encoder
else:
    y_train_encoded = train[target_col].values

# 特征工程
X_train_fe = feature_engineering(train_clean)
X_train_fe.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_train_fe.columns]

# ========== 加载已保存的模型 ==========
print("加载已保存的模型...")
try:
    with open('data/best_model.pkl', 'rb') as f:
        model = pickle.load(f)
    print("模型加载成功！")
except Exception as e:
    print(f"加载模型失败: {e}")
    print("尝试使用 dill 加载...")
    import dill
    with open('data/best_model.pkl', 'rb') as f:
        model = dill.load(f)
    print("模型加载成功（dill）！")

# ========== 加载测试集并预测 ==========
print("加载测试集...")
test = pd.read_csv('data/test.csv')
id_col = 'id' if 'id' in test.columns else test.columns[0]

# 预处理测试集
test_clean = preprocess(test, mode='test')
X_test_fe = feature_engineering(test_clean)
X_test_fe.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_test_fe.columns]

# 确保测试集列与训练集一致
train_cols = X_train_fe.columns.tolist()
for col in train_cols:
    if col not in X_test_fe.columns:
        X_test_fe[col] = 0
X_test_fe = X_test_fe[train_cols]

# 预测
print("进行测试集预测...")
if hasattr(model, 'predict_proba'):
    test_probs = model.predict_proba(X_test_fe)[:, 1]
else:
    test_probs = model.predict(X_test_fe).astype(float)

best_threshold = PREPROCESS_STATE.get('best_threshold', 0.5)
test_preds = (test_probs >= best_threshold).astype(int)

# 保存测试预测结果
result_df = pd.DataFrame({
    'id': test[id_col] if id_col in test.columns else range(len(test_preds)),
    'prediction': test_preds,
    'probability': test_probs
})
result_df.to_csv('output/test_predictions.csv', index=False)
print("测试集预测结果已保存到 output/test_predictions.csv")

# ========== 保存模型到 output/ ==========
with open('output/model.pkl', 'wb') as f:
    pickle.dump(model, f)
print("模型已保存到 output/model.pkl")

# ========== 特征重要性 ==========
print("提取特征重要性...")
feature_names = X_train_fe.columns.tolist()

if hasattr(model, 'feature_importances_'):
    importances = model.feature_importances_
elif hasattr(model, 'coef_'):
    importances = np.abs(model.coef_).flatten()
else:
    importances = np.ones(len(feature_names)) / len(feature_names)

# 创建特征重要性 DataFrame
fi_df = pd.DataFrame({
    'name': feature_names,
    'importance': importances
})
fi_df = fi_df.sort_values('importance', ascending=False).reset_index(drop=True)
fi_df.to_csv('output/feature_importance.csv', index=False)
print("特征重要性已保存到 output/feature_importance.csv")

# 特征重要性图
plt.figure(figsize=(10, 6))
top_n = min(20, len(fi_df))
top_features = fi_df.head(top_n)
plt.barh(range(top_n), top_features['importance'].values[::-1])
plt.yticks(range(top_n), top_features['name'].values[::-1])
plt.xlabel('Importance')
plt.title('Feature Importance (Top {})'.format(top_n))
plt.tight_layout()
plt.savefig('output/feature_importance.png', dpi=100, bbox_inches='tight')
plt.close()
print("特征重要性图已保存到 output/feature_importance.png")

# ========== 计算验证集指标（用于报告）==========
print("计算验证集指标...")
val = pd.read_csv('data/validation.csv')
val_clean = preprocess(val, mode='test')

if target_col in val_clean.columns:
    y_val = val_clean[target_col]
    X_val = val_clean.drop(columns=[target_col])
else:
    y_val = val[target_col]
    X_val = val_clean

# 处理验证集目标编码
if _label_encoder is not None:
    try:
        y_val = _label_encoder.transform(val[target_col])
    except ValueError:
        _val_labels = val[target_col].astype(str).str.strip().str.rstrip('.')
        _train_labels = pd.Series(train[target_col]).astype(str).str.strip().str.rstrip('.')
        _label_encoder.fit(_train_labels)
        y_val = _label_encoder.transform(_val_labels)

X_val_fe = feature_engineering(X_val)
X_val_fe.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_val_fe.columns]
for col in train_cols:
    if col not in X_val_fe.columns:
        X_val_fe[col] = 0
X_val_fe = X_val_fe[train_cols]

if hasattr(model, 'predict_proba'):
    val_probs = model.predict_proba(X_val_fe)[:, 1]
else:
    val_probs = model.predict(X_val_fe).astype(float)

val_preds = (val_probs >= best_threshold).astype(int)

# 计算指标
metrics = {
    'accuracy': float(accuracy_score(y_val, val_preds)),
    'precision': float(precision_score(y_val, val_preds, zero_division=0)),
    'recall': float(recall_score(y_val, val_preds, zero_division=0)),
    'f1': float(f1_score(y_val, val_preds, zero_division=0)),
    'auc': float(roc_auc_score(y_val, val_probs)),
    'best_threshold': float(best_threshold)
}

# 混淆矩阵
cm = confusion_matrix(y_val, val_preds)
cm_list = cm.tolist()

# 分类报告
cr_dict = classification_report(y_val, val_preds, output_dict=True, zero_division=0)

print("验证集指标计算完成")

# ========== 生成 ROC 曲线图 ==========
from sklearn.metrics import roc_curve
fpr, tpr, _ = roc_curve(y_val, val_probs)
plt.figure(figsize=(8, 6))
plt.plot(fpr, tpr, label=f'ROC (AUC = {metrics["auc"]:.4f})')
plt.plot([0, 1], [0, 1], 'k--', label='Random')
plt.xlabel('False Positive Rate')
plt.ylabel('True Positive Rate')
plt.title('ROC Curve')
plt.legend(loc='lower right')
plt.tight_layout()
plt.savefig('output/report_fig.png', dpi=100, bbox_inches='tight')
plt.close()
print("ROC曲线已保存到 output/report_fig.png")

# ========== 生成 HTML 报告 ==========
print("生成 HTML 报告...")

# 构建特征重要性表格 HTML
fi_html_rows = ""
for _, row in fi_df.head(20).iterrows():

test_preview_html = ""

# ========== 生成独立预测脚本 ==========
predict_script = '''import pandas as pd
import numpy as np
import pickle
import re
import os
import sys

def load_model(model_path='model.pkl'):
    """加载保存的模型"""
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    return model

def predict(new_data_path, model_path='model.pkl', output_path='predictions.csv'):
    """
    对新数据进行预测
    new_data_path: 新数据CSV文件路径
    model_path: 模型文件路径
    output_path: 输出预测结果路径
    """
    # 加载模型
    model = load_model(model_path)
    
    # 加载数据
    df = pd.read_csv(new_data_path)
    
    # 预处理（与训练时一致）
    drop_cols = ['id']
    cat_cols = ['paymentMethod', 'Category']
    log_transform_cols = ['numItems', 'localTime', 'paymentMethodAgeDays']
    
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    if 'Category' in df.columns:
        df['Category'] = df['Category'].fillna('Unknown')
    
    if 'isWeekend' in df.columns:
        df['isWeekend'] = df['isWeekend'].fillna(0)
    
    for col in log_transform_cols:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
            df[col] = np.log1p(df[col].clip(lower=-1))
    
    # 注意：这里需要加载训练时保存的编码器
    # 简化处理：假设类别列已经是数值或使用默认编码
    for col in cat_cols:
        if col in df.columns and df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    # 确保所有列数值类型
    for col in df.columns:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    X = df.fillna(0)
    X.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X.columns]
    
    # 预测
    if hasattr(model, 'predict_proba'):
        probs = model.predict_proba(X)[:, 1]
    else:
        probs = model.predict(X).astype(float)
    
    preds = (probs >= 0.5).astype(int)
    
    # 保存结果
    result = pd.DataFrame({
        'prediction': preds,
        'probability': probs
    })
    result.to_csv(output_path, index=False)
    print(f"预测结果已保存到 {output_path}")
    
    return result

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("用法: python predict.py <新数据文件路径> [输出文件路径]")
        sys.exit(1)
    
    data_path = sys.argv[1]
    out_path = sys.argv[2] if len(sys.argv) > 2 else 'predictions.csv'
    predict(data_path, output_path=out_path)
'''
