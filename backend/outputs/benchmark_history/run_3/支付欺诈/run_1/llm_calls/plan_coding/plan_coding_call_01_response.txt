6. 保存模型到 output/model.pkl
7. 生成配套预测脚本 output/predict.py
8. 输出摘要 JSON
</plan>

    PREPROCESS_STATE = {

    def preprocess(df, mode='train'):

        target_col = PREPROCESS_STATE['target_col']

        df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')

        if 'Category' in df.columns:
            df['Category'] = df['Category'].fillna('Unknown')

        if 'isWeekend' in df.columns:
            if mode == 'train':
                PREPROCESS_STATE['isWeekend_median'] = df['isWeekend'].median()
            df['isWeekend'] = df['isWeekend'].fillna(PREPROCESS_STATE['isWeekend_median'])

        for col in log_transform_cols:
            if col in df.columns:
                df[col] = pd.to_numeric(df[col], errors='coerce')
                min_val = df[col].min()
                if min_val < -1:
                    df[col] = df[col] - min_val
                df[col] = np.log1p(df[col].clip(lower=-1))

        if mode == 'train':
            encoder = OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1)
            existing_cat_cols = [c for c in cat_cols if c in df.columns]
            if existing_cat_cols:
                df[existing_cat_cols] = encoder.fit_transform(df[existing_cat_cols].astype(str))
                PREPROCESS_STATE['ordinal_encoder'] = encoder
                PREPROCESS_STATE['existing_cat_cols'] = existing_cat_cols
        else:
            encoder = PREPROCESS_STATE.get('ordinal_encoder')
            existing_cat_cols = PREPROCESS_STATE.get('existing_cat_cols', [])
            if encoder is not None and existing_cat_cols:
                existing_cat_cols_in_df = [c for c in existing_cat_cols if c in df.columns]
                if existing_cat_cols_in_df:
                    df[existing_cat_cols_in_df] = encoder.transform(df[existing_cat_cols_in_df].astype(str))

        for col in df.columns:
            if col != target_col and df[col].dtype == 'object':
                df[col] = pd.to_numeric(df[col], errors='coerce')

    X_train_fe = feature_engineering(X_train)

# 预处理测试集

predict_script = '''#!/usr/bin/env python3
"""Standalone prediction script for binary classification model."""
import pandas as pd
import numpy as np
import pickle
import re
import os
import sys

def load_model(model_path='model.pkl'):
    """Load the trained model from pickle file."""
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    return model

def predict(model, data_path, output_path='predictions.csv', threshold=0.5):
    """Make predictions on new data."""
    # Load data
    df = pd.read_csv(data_path)
    
    # Identify ID column
    id_col = 'id' if 'id' in df.columns else df.columns[0]
    
    # Prepare features (drop ID and any target column if present)
    target_col = 'label'
    drop_cols = ['id', target_col]
    X = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    # Ensure numeric
    X = X.select_dtypes(include=[np.number]).fillna(0)
    
    # Clean column names
    X.columns = [re.sub('[^\\\\w]', '_', str(c)) for c in X.columns]
    
    # Predict
    if hasattr(model, 'predict_proba'):
        probs = model.predict_proba(X)[:, 1]
    else:
        probs = model.predict(X).astype(float)
    
    preds = (probs >= threshold).astype(int)
    
    # Save results
    result = pd.DataFrame({
        id_col: df[id_col] if id_col in df.columns else range(len(preds)),
        'prediction': preds,
        'probability': probs
    })
    result.to_csv(output_path, index=False)
    print(f"Predictions saved to {output_path}")
    return result

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python predict.py <input_csv> [output_csv] [threshold]")
        sys.exit(1)
    
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else 'predictions.csv'
    threshold = float(sys.argv[3]) if len(sys.argv) > 3 else 0.5
    
    model = load_model()
    predict(model, input_path, output_path, threshold)
'''

with open('output/predict.py', 'w', encoding='utf-8') as f:
