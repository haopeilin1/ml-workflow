def preprocess(df, mode='test'):
    global PREPROCESS_STATE
    target_col = PREPROCESS_STATE['target_col']
    drop_cols = PREPROCESS_STATE['drop_cols']
    cat_cols = PREPROCESS_STATE['cat_cols']
    log_transform_cols = PREPROCESS_STATE['log_transform_cols']
    
    df = df.copy()
    
    # 1. 丢弃ID列
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    # 2. 处理Category缺失值
    if 'Category' in df.columns:
        df['Category'] = df['Category'].fillna('Unknown')
    
    # 3. 处理isWeekend缺失值
    if 'isWeekend' in df.columns:
        if mode == 'train':
            PREPROCESS_STATE['isWeekend_median'] = df['isWeekend'].median()
        median_val = PREPROCESS_STATE.get('isWeekend_median', 0)
        if median_val is None:
            median_val = 0
        df['isWeekend'] = df['isWeekend'].fillna(median_val)
    
    # 4. log1p变换
    for col in log_transform_cols:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
            min_val = df[col].min()
            if pd.notna(min_val) and min_val < -1:
                df[col] = df[col] - min_val
            df[col] = np.log1p(df[col].clip(lower=-1))
    
    # 5. 编码类别特征
    if mode == 'train':
        encoder = OrdinalEncoder(handle_unknown='use_encoded_value', unknown_value=-1)
        existing_cat_cols = [c for c in cat_cols if c in df.columns]
        if existing_cat_cols:
            df[existing_cat_cols] = encoder.fit_transform(df[existing_cat_cols].astype(str))
            PREPROCESS_STATE['ordinal_encoder'] = encoder
            PREPROCESS_STATE['existing_cat_cols'] = existing_cat_cols
    else:
        encoder = PREPROCESS_STATE.get('ordinal_encoder')
        existing_cat_cols = PREPROCESS_STATE.get('existing_cat_cols', cat_cols)
        if encoder is not None:
            existing_cat_cols_in_df = [c for c in existing_cat_cols if c in df.columns]
            if existing_cat_cols_in_df:
                df[existing_cat_cols_in_df] = encoder.transform(df[existing_cat_cols_in_df].astype(str))
    
    # 6. 确保所有非目标列是数值类型
    for col in df.columns:
        if col != target_col and df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    return df


def feature_engineering(df):
    target_col = PREPROCESS_STATE['target_col']
    X = df.drop(columns=[target_col], errors='ignore')
    
    # 确保所有列都是数值类型（修复报错关键）
    for col in X.columns:

    return X

import pandas as pd
import numpy as np
import pickle
import re
import os
from sklearn.preprocessing import OrdinalEncoder

# 预处理状态（与训练时一致）
PREPROCESS_STATE = {
    'isWeekend_median': None,
    'ordinal_encoder': None,
    'cat_cols': ['paymentMethod', 'Category'],
    'log_transform_cols': ['numItems', 'localTime', 'paymentMethodAgeDays'],
    'drop_cols': ['id'],
    'target_col': 'label',
    'existing_cat_cols': [],
    'best_threshold': 0.5,
    'label_encoder': None
}

def preprocess(df, mode='test'):
    global PREPROCESS_STATE
    target_col = PREPROCESS_STATE['target_col']
    drop_cols = PREPROCESS_STATE['drop_cols']
    cat_cols = PREPROCESS_STATE['cat_cols']
    log_transform_cols = PREPROCESS_STATE['log_transform_cols']
    
    df = df.copy()
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    if 'Category' in df.columns:
        df['Category'] = df['Category'].fillna('Unknown')
    
    if 'isWeekend' in df.columns:
        median_val = PREPROCESS_STATE.get('isWeekend_median', 0)
        if median_val is None:
            median_val = 0
        df['isWeekend'] = df['isWeekend'].fillna(median_val)
    
    for col in log_transform_cols:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
            min_val = df[col].min()
            if pd.notna(min_val) and min_val < -1:
                df[col] = df[col] - min_val
            df[col] = np.log1p(df[col].clip(lower=-1))
    
    encoder = PREPROCESS_STATE.get('ordinal_encoder')
    existing_cat_cols = PREPROCESS_STATE.get('existing_cat_cols', cat_cols)
    if encoder is not None:
        existing_cat_cols_in_df = [c for c in existing_cat_cols if c in df.columns]
        if existing_cat_cols_in_df:
            df[existing_cat_cols_in_df] = encoder.transform(df[existing_cat_cols_in_df].astype(str))
    
    for col in df.columns:
        if col != target_col and df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    return df

def feature_engineering(df):
    target_col = PREPROCESS_STATE['target_col']
    X = df.drop(columns=[target_col], errors='ignore')
    
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce')
    
    X = X.fillna(0)
    X.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X.columns]
    if X.columns.duplicated().any():
        X.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X.columns)]
    
    return X

def predict(new_data_path, output_path=None):
    """
    对新数据进行预测
    
    Parameters:
    - new_data_path: str, 新数据CSV文件路径
    - output_path: str, 输出文件路径（可选），默认保存到同目录下 predictions.csv
    
    Returns:
    - DataFrame with predictions
    """
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # 加载数据
    df = pd.read_csv(new_data_path)
    original_df = df.copy()
    
    # 预处理
    df_clean = preprocess(df, mode='test')
    X = feature_engineering(df_clean)
    
    # 对齐特征列
    if hasattr(model, 'feature_name_'):
        expected_features = model.feature_name_
        missing_cols = set(expected_features) - set(X.columns)
        for col in missing_cols:
            X[col] = 0
        X = X[expected_features]
    
    # 预测
    if hasattr(model, 'predict_proba'):
        probs = model.predict_proba(X)[:, 1]
    else:
        probs = model.predict(X).astype(float)
    
    best_threshold = PREPROCESS_STATE.get('best_threshold', 0.5)
