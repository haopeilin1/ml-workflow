import pandas as pd
import numpy as np
import pickle
import json
import os
import re
import warnings

# 确保输出目录存在
os.makedirs('output', exist_ok=True)

print("Loading model from data/best_model.pkl...")
try:
    with open('data/best_model.pkl', 'rb') as f:
        model = pickle.load(f)
    print("Model loaded successfully.")

    print(f"Failed to load model: {e}")
    raise

# ========== 2. 加载数据 ==========
train = pd.read_csv('data/train.csv')
test = pd.read_csv('data/test.csv')

target_col = 'Status'
id_col = 'id'
if id_col not in test.columns:
    id_col = test.columns[0]

# ========== 3. 预处理函数（与训练时一致） ==========
def preprocess(df, mode='test'):
    df = df.drop(columns=['id'], errors='ignore')
    df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
    
    categorical_cols = ['Drug', 'Sex', 'Ascites', 'Hepatomegaly', 'Spiders', 'Edema']
    categorical_cols = [c for c in categorical_cols if c in df.columns]
    
    for col in categorical_cols:
        if col in df.columns:
            df[col] = df[col].fillna('Missing')
            df[col] = df[col].astype(str)
    
    feature_cols = [c for c in df.columns if c != target_col]
    num_cols = [c for c in feature_cols if c not in categorical_cols]
    
    for col in num_cols:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    return df

def feature_engineering(df):
    target_col = 'Status'
    X = df.drop(columns=[target_col], errors='ignore')
    
    if 'Bilirubin' in X.columns and 'Albumin' in X.columns:
        X['Bilirubin_Albumin_ratio'] = X['Bilirubin'] / (X['Albumin'] + 1e-6)
    if 'Stage' in X.columns and 'N_Days' in X.columns:
        X['Stage_N_Days_interact'] = X['Stage'] * X['N_Days']
    
    return X

# ========== 4. 处理训练集（用于获取特征名和标签编码） ==========
train_clean = preprocess(train, mode='test')
if target_col in train_clean.columns:
    y_train_original = train_clean[target_col].copy()
    X_train_raw = train_clean.drop(columns=[target_col])
else:
    y_train_original = train[target_col].copy()
    X_train_raw = train_clean

# 标签编码
label_encoder = LabelEncoder()
y_train_encoded = label_encoder.fit_transform(y_train_original)
label_mapping = {i: label for i, label in enumerate(label_encoder.classes_)}

# 特征工程
X_train_fe = feature_engineering(X_train_raw)

# ========== 5. 处理测试集 ==========
test_clean = preprocess(test, mode='test')
X_test_raw = test_clean.drop(columns=[target_col], errors='ignore')
if X_test_raw is test_clean:
    X_test_raw = test_clean.copy()

X_test_fe = feature_engineering(X_test_raw)

# 清洗特征名
for _df in [X_train_fe, X_test_fe]:
    _df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in _df.columns]
    if _df.columns.duplicated().any():
        _df.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(_df.columns)]

# 对齐特征（确保测试集特征与训练集一致）
common_cols = [c for c in X_train_fe.columns if c in X_test_fe.columns]
X_train_fe = X_train_fe[common_cols]
X_test_fe = X_test_fe[common_cols]

# ========== 6. 测试集预测 ==========
print("Making predictions on test set...")
test_preds_encoded = model.predict(X_test_fe)
test_probs_all = model.predict_proba(X_test_fe)

# 反编码预测标签
test_preds_labels = label_encoder.inverse_transform(test_preds_encoded)

# 保存测试集预测结果
result_df = pd.DataFrame({
    'id': test[id_col] if id_col in test.columns else range(len(test_preds_labels)),
    'prediction': test_preds_labels,
})
for i in range(test_probs_all.shape[1]):
    result_df[f'proba_{i}'] = test_probs_all[:, i]
result_df.to_csv('output/test_predictions.csv', index=False)
print("Test predictions saved to output/test_predictions.csv")

# ========== 7. 训练集预测（用于评估指标） ==========
train_preds_encoded = model.predict(X_train_fe)
train_probs = model.predict_proba(X_train_fe)

# 计算指标
accuracy = float(accuracy_score(y_train_encoded, train_preds_encoded))
f1_macro = float(f1_score(y_train_encoded, train_preds_encoded, average='macro'))
try:
    multi_logloss = float(log_loss(y_train_encoded, train_probs))
except Exception:
    multi_logloss = None

# 分类报告
report_dict = classification_report(y_train_encoded, train_preds_encoded, 
                                     target_names=[str(label_mapping[i]) for i in range(len(label_mapping))],
                                     output_dict=True)

# 混淆矩阵
cm = confusion_matrix(y_train_encoded, train_preds_encoded)

# ========== 8. 保存模型到 output/model.pkl ==========
with open('output/model.pkl', 'wb') as f:
    pickle.dump(model, f)
print("Model saved to output/model.pkl")

# ========== 9. 特征重要性 ==========
print("Generating feature importance...")
feature_names = list(X_train_fe.columns)

# 获取特征重要性
if hasattr(model, 'feature_importances_'):
    importances = model.feature_importances_
elif hasattr(model, 'coef_'):
    importances = np.abs(model.coef_).flatten()
else:
    importances = np.ones(len(feature_names)) / len(feature_names)

# 确保长度匹配
if len(importances) != len(feature_names):
    importances = importances[:len(feature_names)]

# 保存特征重要性 CSV
importance_df = pd.DataFrame({
    'name': feature_names,
    'importance': importances
})
importance_df = importance_df.sort_values('importance', ascending=False)
importance_df.to_csv('output/feature_importance.csv', index=False)
print("Feature importance saved to output/feature_importance.csv")

# 生成特征重要性图
plt.figure(figsize=(10, 8))
top_n = min(20, len(importance_df))
top_features = importance_df.head(top_n)
plt.barh(range(top_n), top_features['importance'].values[::-1])
plt.yticks(range(top_n), top_features['name'].values[::-1])
plt.xlabel('Importance')
plt.title('Feature Importance (Top {})'.format(top_n))
plt.tight_layout()
plt.savefig('output/feature_importance.png', dpi=100, bbox_inches='tight')
plt.close()
print("Feature importance plot saved to output/feature_importance.png")

# ========== 10. 混淆矩阵热力图 ==========
print("Generating confusion matrix heatmap...")
plt.figure(figsize=(8, 6))
class_names = [str(label_mapping[i]) for i in range(len(label_mapping))]
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', 
            xticklabels=class_names, yticklabels=class_names)
plt.xlabel('Predicted')
plt.ylabel('True')
plt.title('Confusion Matrix (Training Set)')
plt.tight_layout()

report_rows = []
for class_name, metrics in report_dict.items():

predict_script = '''
import pandas as pd
import pickle
import re
import os
import sys

def preprocess(df):
    """预处理函数"""
    df = df.drop(columns=['id'], errors='ignore')
    df.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in df.columns]
    
    categorical_cols = ['Drug', 'Sex', 'Ascites', 'Hepatomegaly', 'Spiders', 'Edema']
    categorical_cols = [c for c in categorical_cols if c in df.columns]
    
    for col in categorical_cols:
        if col in df.columns:
            df[col] = df[col].fillna('Missing')
            df[col] = df[col].astype(str)
    
    feature_cols = [c for c in df.columns if c != 'Status']
    num_cols = [c for c in feature_cols if c not in categorical_cols]
    
    for col in num_cols:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    return df

def feature_engineering(df):
    """特征工程"""
    X = df.drop(columns=['Status'], errors='ignore')
    
    if 'Bilirubin' in X.columns and 'Albumin' in X.columns:
        X['Bilirubin_Albumin_ratio'] = X['Bilirubin'] / (X['Albumin'] + 1e-6)
    if 'Stage' in X.columns and 'N_Days' in X.columns:
        X['Stage_N_Days_interact'] = X['Stage'] * X['N_Days']
    
    return X

def predict(input_file, output_file=None):
    """
    对新数据进行预测。
    
    Parameters:
    -----------
    input_file : str
        输入 CSV 文件路径，必须包含与训练数据相同的特征列
    output_file : str, optional
        输出 CSV 文件路径，默认输出到同目录下 predictions.csv
    
    Returns:
    --------
    pd.DataFrame : 包含 id, prediction 和各类概率的预测结果
    """
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # 加载数据
    df = pd.read_csv(input_file)
    
    # 保存 id 列
    id_col = 'id'
    if id_col not in df.columns:
        id_col = df.columns[0]
    ids = df[id_col].copy() if id_col in df.columns else range(len(df))
    
    # 预处理
    df_clean = preprocess(df)
    X = feature_engineering(df_clean)
    
    # 清洗特征名
    X.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X.columns]
    
    # 预测
    predictions = model.predict(X)
    probabilities = model.predict_proba(X)
    
    # 构建结果
    result = pd.DataFrame({
        'id': ids,
        'prediction': predictions,
    })
    for i in range(probabilities.shape[1]):
        result[f'proba_{i}'] = probabilities[:, i]
    
    # 保存结果
    if output_file is None:
        output_file = os.path.join(os.path.dirname(input_file), 'predictions.csv')
    result.to_csv(output_file, index=False)
    print(f"Predictions saved to {output_file}")
    
    return result

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python predict.py <input_file.csv> [output_file.csv]")
        sys.exit(1)
    
    input_file = sys.argv[1]
    output_file = sys.argv[2] if len(sys.argv) > 2 else None
    predict(input_file, output_file)
'''

with open('output/predict.py', 'w', encoding='utf-8') as f:

print("Prediction script saved to output/predict.py")
