        model = pickle.load(f)
    print("成功从 data/best_model.pkl 加载模型")
except Exception as e:
    print(f"加载模型失败: {e}")
    print("将尝试重新训练模型...")
    # 如果加载失败，回退到重新训练
    raise SystemExit("模型加载失败，请检查 data/best_model.pkl 是否存在且有效")

train = pd.read_csv('data/train.csv')

# 从最佳代码中提取预处理逻辑
cat_cols = ['gender', 'Partner', 'Dependents', 'PhoneService', 'MultipleLines',
            'InternetService', 'OnlineSecurity', 'OnlineBackup', 'DeviceProtection',
            'TechSupport', 'StreamingTV', 'StreamingMovies', 'Contract',
            'PaperlessBilling', 'PaymentMethod']

def preprocess_data(df, is_train=False):
    """预处理数据，返回处理后的DataFrame"""
    df = df.copy()
    
    # 丢弃id列
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    
    # 处理目标列
    if target_col in df.columns:
        from sklearn.preprocessing import LabelEncoder
        le = LabelEncoder()
        df[target_col] = le.fit_transform(df[target_col])
    
    # 只保留实际存在的类别列
    existing_cat_cols = [c for c in cat_cols if c in df.columns]
    
    # 数值列
    exclude_cols = set(existing_cat_cols + [target_col])
    num_cols = [c for c in df.columns if c not in exclude_cols and df[c].dtype in ['int64', 'float64']]
    
    return df, existing_cat_cols, num_cols

# ========== 4. 构建完整的Pipeline（用于保存）==========
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler, LabelEncoder
from sklearn.impute import SimpleImputer
from sklearn.pipeline import Pipeline

# 预处理训练数据以获取列信息
train_processed, train_cat_cols, train_num_cols = preprocess_data(train, is_train=True)

# 构建预处理器
preprocessor = ColumnTransformer([
    ('cat', Pipeline([
        ('imputer', SimpleImputer(strategy='most_frequent')),
        ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
    ]), train_cat_cols),
    ('num', Pipeline([
        ('imputer', SimpleImputer(strategy='median')),
        ('scaler', StandardScaler())
    ]), train_num_cols)
], remainder='passthrough')

# 拟合预处理器
X_train_raw = train_processed.drop(columns=[target_col], errors='ignore')
preprocessor.fit(X_train_raw)

# 获取特征名称
try:
    feature_names = preprocessor.get_feature_names_out()
except AttributeError:
    ohe = preprocessor.named_transformers_['cat'].named_steps['encoder']
    cat_feature_names = []
    for i, col in enumerate(train_cat_cols):
        categories = ohe.categories_[i]
        cat_feature_names.extend([f"{col}_{cat}" for cat in categories])
    feature_names = np.array(cat_feature_names + train_num_cols)

# 清洗特征名
feature_names_clean = [re.sub(r'[^\w]', '_', str(c)) for c in feature_names]

# ========== 5. 构建完整Pipeline并保存 ==========
full_pipeline = Pipeline([
    ('preprocessor', preprocessor),
    ('model', model)
])

# 保存完整Pipeline
with open('output/model.pkl', 'wb') as f:
    pickle.dump(full_pipeline, f)
print("模型已保存到 output/model.pkl")

# ========== 6. 特征重要性（直接从模型获取）==========
print("正在提取特征重要性...")
if hasattr(model, 'feature_importances_'):
    importances = model.feature_importances_
    
    # 确保特征名称和重要性长度一致
    if len(feature_names_clean) == len(importances):
        feature_importance_df = pd.DataFrame({
            'name': feature_names_clean,
            'importance': importances
        })
    else:
        # 如果长度不匹配，使用索引
        feature_importance_df = pd.DataFrame({
            'name': [f'feature_{i}' for i in range(len(importances))],
            'importance': importances
        })
    
    # 按重要性降序排序
    feature_importance_df = feature_importance_df.sort_values('importance', ascending=False)
    feature_importance_df.to_csv('output/feature_importance.csv', index=False)
    print(f"特征重要性已保存到 output/feature_importance.csv (共 {len(feature_importance_df)} 个特征)")
else:
    # 如果模型没有feature_importances_属性
    feature_importance_df = pd.DataFrame({'name': ['N/A'], 'importance': [0]})
    feature_importance_df.to_csv('output/feature_importance.csv', index=False)
    print("警告: 模型没有 feature_importances_ 属性")

test_processed, _, _ = preprocess_data(test, is_train=False)

if target_col in test_processed.columns:

# 预测
if hasattr(model, 'predict_proba'):

else:
    test_probs = model.predict(preprocessor.transform(X_test)).astype(float)

test_preds = (test_probs >= 0.5).astype(int)

# 保存预测结果
result_df = pd.DataFrame({

result_df.to_csv('output/test_predictions.csv', index=False)

predict_script = '''#!/usr/bin/env python3
"""
独立预测脚本
用法: python predict.py <输入CSV文件路径> [输出CSV文件路径]
示例: python predict.py new_data.csv predictions.csv
"""
import pandas as pd
import pickle
import sys
import os

def load_model(model_path='output/model.pkl'):
    """加载保存的模型"""
    if not os.path.exists(model_path):
        # 尝试从当前目录的output子目录加载
        model_path = os.path.join(os.path.dirname(__file__), 'output', 'model.pkl')
    
    if not os.path.exists(model_path):
        raise FileNotFoundError(f"模型文件未找到: {model_path}")
    
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    return model

def predict(model, data_path, output_path=None):
    """对数据进行预测"""
    # 加载数据
    df = pd.read_csv(data_path)
    print(f"加载数据: {df.shape[0]} 行 × {df.shape[1]} 列")
    
    # 保存ID列（如果存在）
    id_col = None
    if 'id' in df.columns:
        id_col = df['id'].copy()
    
    # 预测
    if hasattr(model, 'predict_proba'):
        probabilities = model.predict_proba(df)[:, 1]
        predictions = (probabilities >= 0.5).astype(int)
    else:
        predictions = model.predict(df)
        probabilities = predictions.astype(float)
    
    # 构建结果
    result = pd.DataFrame()
    if id_col is not None:
        result['id'] = id_col
    else:
        result['id'] = range(len(predictions))
    
    result['prediction'] = predictions
    result['probability'] = probabilities
    
    # 保存结果
    if output_path is None:
        output_path = 'predictions.csv'
    
    result.to_csv(output_path, index=False)
    print(f"预测结果已保存到: {output_path}")
    print(f"预测统计: 正类={predictions.sum()}, 负类={(1-predictions).sum()}")
    
    return result

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("用法: python predict.py <输入CSV文件路径> [输出CSV文件路径]")
        sys.exit(1)
    
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else None
    
    model = load_model()
    predict(model, input_path, output_path)
'''
