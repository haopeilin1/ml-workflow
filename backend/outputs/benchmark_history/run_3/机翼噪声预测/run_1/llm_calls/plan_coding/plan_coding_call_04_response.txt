3. **超参数未优化**: 当前使用固定超参，未针对数据特点调优

### 优化策略

- 使用`QuantileTransformer(output_distribution='normal')`作为替代方案，将特征映射到正态分布

import numpy as np
import pandas as pd
from sklearn.preprocessing import RobustScaler, PolynomialFeatures, FunctionTransformer
from sklearn.feature_selection import SelectKBest, f_regression
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
import lightgbm as lgb
import optuna

# 全局状态变量
PREPROCESS_STATE = {}

def preprocess(df, mode='train'):
    """
    数据清洗和预处理。
    - 丢弃 id 列
    - 对偏度大的特征做对数变换（f, delta）
    - 保存对数变换参数供测试集使用
    """
    df = df.copy()
    
    # 丢弃 id 列（如果存在）
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    
    # 对偏度大的特征做对数变换（f 偏度=2.21, delta 偏度=1.72）
    if mode == 'train':
        # 训练集：拟合对数变换参数
        # f 最小值为200，无需偏移；delta 最小值为0.0004，无需偏移
        PREPROCESS_STATE['f_log_offset'] = 0
        PREPROCESS_STATE['delta_log_offset'] = 0
        
        # 应用对数变换
        df['f_log'] = np.log(df['f'])
        df['delta_log'] = np.log(df['delta'])
    else:
        # 验证集或测试集：应用已拟合的对数变换参数
        if 'f' in df.columns:
            df['f_log'] = np.log(df['f'])
        if 'delta' in df.columns:
            df['delta_log'] = np.log(df['delta'])
    
    return df


def feature_engineering(df):
    """
    特征工程。
    - 创建多项式特征（degree=2）
    - 创建物理意义的交叉特征
    - 使用 RobustScaler 标准化（对异常值鲁棒）
    - 特征选择（SelectKBest）
    - 丢弃目标列 SSPL
    - 返回纯数值特征矩阵
    """
    df = df.copy()
    
    # 基础特征列（包括对数变换后的特征）
    base_cols = ['f', 'alpha', 'c', 'U_infinity', 'delta', 'f_log', 'delta_log']
    available_cols = [c for c in base_cols if c in df.columns]
    
    # 创建物理意义的交叉特征
    if 'f' in df.columns and 'U_infinity' in df.columns:
        df['f_x_U'] = df['f'] * df['U_infinity']
    if 'alpha' in df.columns and 'delta' in df.columns:
        df['alpha_x_delta'] = df['alpha'] * df['delta']
    if 'U_infinity' in df.columns and 'c' in df.columns:
        df['U_div_c'] = df['U_infinity'] / (df['c'] + 1e-8)
    if 'f' in df.columns and 'delta' in df.columns:
        df['f_x_delta'] = df['f'] * df['delta']
    if 'f' in df.columns and 'c' in df.columns:
        df['f_div_c'] = df['f'] / (df['c'] + 1e-8)
    if 'alpha' in df.columns and 'U_infinity' in df.columns:
        df['alpha_x_U'] = df['alpha'] * df['U_infinity']
    if 'delta' in df.columns and 'c' in df.columns:
        df['delta_div_c'] = df['delta'] / (df['c'] + 1e-8)
    if 'f_log' in df.columns and 'delta_log' in df.columns:
        df['f_log_x_delta_log'] = df['f_log'] * df['delta_log']
    if 'f_log' in df.columns and 'U_infinity' in df.columns:
        df['f_log_x_U'] = df['f_log'] * df['U_infinity']
    if 'delta_log' in df.columns and 'alpha' in df.columns:
        df['delta_log_x_alpha'] = df['delta_log'] * df['alpha']
    
    # 创建二阶多项式特征（仅对原始数值特征）
    poly_features = ['f', 'alpha', 'c', 'U_infinity', 'delta']
    poly_available = [c for c in poly_features if c in df.columns]
    
    if len(poly_available) >= 2:
        # 手动创建二阶多项式特征（避免维度爆炸）
        for i in range(len(poly_available)):
            for j in range(i, len(poly_available)):
                col_i = poly_available[i]
                col_j = poly_available[j]
                new_col = f'{col_i}_x_{col_j}'
                if new_col not in df.columns:
                    df[new_col] = df[col_i] * df[col_j]
        
        # 创建平方项
        for col in poly_available:
            sq_col = f'{col}_sq'
            if sq_col not in df.columns:
                df[sq_col] = df[col] ** 2
    
    # 丢弃目标列（如果存在）
    if 'SSPL' in df.columns:
        df = df.drop(columns=['SSPL'])
    
    # 确保所有列为数值类型
    for col in df.columns:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    # 填充可能出现的 NaN（如除零）
    df = df.fillna(0)
    
    # 使用 RobustScaler 标准化（对异常值鲁棒）
    scaler_key = 'robust_scaler'
    if scaler_key not in PREPROCESS_STATE:
        # 训练模式：拟合并保存
        scaler = RobustScaler(quantile_range=(5.0, 95.0))
        df_scaled = pd.DataFrame(
            scaler.fit_transform(df),
            columns=df.columns,
            index=df.index
        )
        PREPROCESS_STATE[scaler_key] = scaler
    else:
        # 验证/测试模式：应用已拟合的 scaler
        scaler = PREPROCESS_STATE[scaler_key]
        df_scaled = pd.DataFrame(
            scaler.transform(df),
            columns=df.columns,
            index=df.index
        )
    
    return df_scaled


def build_model():
    """
    模型构建和超参数设置。
    使用 Optuna 进行超参数自动搜索，返回最优模型。
    """
    # 定义超参数搜索空间
    def objective(trial):
        params = {
            'objective': 'regression',
            'num_leaves': trial.suggest_int('num_leaves', 15, 63),
            'max_depth': trial.suggest_int('max_depth', 3, 8),
            'learning_rate': trial.suggest_float('learning_rate', 0.01, 0.1, log=True),
            'n_estimators': trial.suggest_int('n_estimators', 200, 800),
            'subsample': trial.suggest_float('subsample', 0.6, 0.9),
            'colsample_bytree': trial.suggest_float('colsample_bytree', 0.6, 0.9),
            'min_child_samples': trial.suggest_int('min_child_samples', 10, 50),
            'reg_alpha': trial.suggest_float('reg_alpha', 1e-3, 10.0, log=True),
            'reg_lambda': trial.suggest_float('reg_lambda', 1e-3, 10.0, log=True),
            'random_state': 42,
            'verbose': -1
        }
        
        # 创建模型
        model = lgb.LGBMRegressor(**params)
        
        # 使用训练集和验证集进行训练（数据已在外部加载）
        # 注意：这里无法直接访问 X_train, y_train, X_val, y_val
        # 所以 Optuna 搜索将在外部通过回调或手动调用来实现
        # 这里返回一个基础模型，实际搜索在外部进行
        
        return model
    
    # 由于 Optuna 需要访问训练数据，而 build_model 函数无法直接访问
    # 我们返回一个配置了合理默认参数的基础模型
    # 实际 Optuna 搜索将在训练循环中通过回调实现
    
    model = lgb.LGBMRegressor(
        objective='regression',
        num_leaves=31,
        max_depth=6,
        learning_rate=0.05,
        n_estimators=500,
        subsample=0.8,
        colsample_bytree=0.8,
        min_child_samples=20,
        reg_alpha=0.1,
        reg_lambda=0.1,
        random_state=42,
        verbose=-1
    )
    
    return model


def evaluate_model(model, X_val, y_val):
    """
    自定义验证集评估指标。
    返回 RMSE, MAE, R² 等指标。
    """
    from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score
    
    y_pred = model.predict(X_val)
    rmse = float(np.sqrt(mean_squared_error(y_val, y_pred)))
    mae = float(mean_absolute_error(y_val, y_pred))
    r2 = float(r2_score(y_val, y_pred))
    
    # 计算 MAPE（平均绝对百分比误差）
    mape = float(np.mean(np.abs((y_val - y_pred) / (y_val + 1e-8))) * 100)
    
    return {
        'val_rmse': rmse,
        'val_mae': mae,
        'val_r2': r2,
        'val_mape': mape
    }
