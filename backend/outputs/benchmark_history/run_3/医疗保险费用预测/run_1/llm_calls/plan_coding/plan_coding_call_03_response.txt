### 当前代码问题
1. **preprocess 函数存在严重问题**：在 `mode='train'` 时，`preprocessor.fit(X)` 后，`X_processed = preprocessor.transform(X)` 返回的是 numpy array，但后续 `feature_names` 的获取方式有误——`ohe.get_feature_names_out(cat_cols)` 传入的是原始列名列表，但 `OneHotEncoder` 的 `get_feature_names_out` 应该传入原始列名列表，这里写法正确但逻辑上存在隐患（列名数量可能不匹配）。
2. **feature_engineering 函数过于简单**：没有做任何特征工程，只是返回了预处理后的数值特征。
3. **build_model 已经使用了较好的参数**：max_depth=6, learning_rate=0.05, n_estimators=500，但缺少 early_stopping 回调。

### 优化策略
1. **修复 preprocess 函数**：确保 `get_feature_names_out` 正确获取列名，避免列名不匹配问题。
2. **增强特征工程**：
   - 创建 BMI 分类特征（肥胖/正常/偏瘦）
   - 创建年龄分组特征
   - 创建交互特征：`age * bmi`、`smoker * age`、`smoker * bmi`
   - 创建 `children` 的平方项（非线性关系）
3. **优化模型**：
   - 保持 LightGBM 但增加 early_stopping 回调

4. **添加 evaluate_model 函数**：提供自定义评估指标（RMSE、MAE、R²）

import numpy as np
import pandas as pd
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler, FunctionTransformer
from sklearn.impute import SimpleImputer
from sklearn.pipeline import Pipeline
import lightgbm as lgb

# 全局状态变量，用于保存预处理参数
PREPROCESS_STATE = {}

def preprocess(df, mode='train'):
    """
    数据清洗和预处理。
    mode='train' 时拟合参数，保存到 PREPROCESS_STATE。
    mode='test' 时应用已保存的参数。
    返回处理后的 DataFrame（仍包含目标列）。
    """
    global PREPROCESS_STATE
    
    # 复制数据，避免修改原始数据
    df = df.copy()
    
    # 丢弃ID列（所有模式下都丢弃）
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    
    if mode == 'train':
        # 定义类别列和数值列
        cat_cols = ['sex', 'smoker', 'region']
        num_cols = ['age', 'bmi', 'children']
        
        # 构建预处理 Pipeline
        preprocessor = ColumnTransformer([
            ('cat', Pipeline([
                ('imputer', SimpleImputer(strategy='most_frequent')),
                ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
            ]), cat_cols),
            ('num', Pipeline([
                ('imputer', SimpleImputer(strategy='median')),
                ('scaler', StandardScaler())
            ]), num_cols)
        ], remainder='passthrough')
        
        # 分离特征和目标
        if 'charges' in df.columns:
            X = df.drop(columns=['charges'])
            y = df['charges']
        else:
            X = df
            y = None
        
        # 拟合预处理器
        preprocessor.fit(X)
        
        # 保存状态
        PREPROCESS_STATE['preprocessor'] = preprocessor
        PREPROCESS_STATE['cat_cols'] = cat_cols
        PREPROCESS_STATE['num_cols'] = num_cols
        
        # 获取特征名（用于后续特征工程）
        ohe = preprocessor.named_transformers_['cat'].named_steps['encoder']
        cat_feature_names = ohe.get_feature_names_out(cat_cols).tolist()
        num_feature_names = num_cols
        PREPROCESS_STATE['feature_names'] = cat_feature_names + num_feature_names
        
        # 保存原始数值列用于特征工程
        PREPROCESS_STATE['raw_num_cols'] = num_cols
        
    elif mode == 'test':
        # 从状态中读取已拟合的预处理器
        if 'preprocessor' not in PREPROCESS_STATE:
            raise ValueError("PREPROCESS_STATE 中没有预处理器，请先运行 mode='train'")
        
        preprocessor = PREPROCESS_STATE['preprocessor']
        
        # 分离特征和目标
        if 'charges' in df.columns:
            X = df.drop(columns=['charges'])
            y = df['charges']
        else:
            X = df
            y = None
    
    # 应用预处理
    X_processed = preprocessor.transform(X)
    
    # 将 numpy 数组转回 DataFrame
    feature_names = PREPROCESS_STATE.get('feature_names', [f'feature_{i}' for i in range(X_processed.shape[1])])
    
    # 确保列名数量匹配
    if len(feature_names) != X_processed.shape[1]:
        feature_names = [f'feature_{i}' for i in range(X_processed.shape[1])]
    
    X_processed_df = pd.DataFrame(X_processed, columns=feature_names, index=df.index)
    
    # 如果目标列存在，合并回去
    if y is not None:
        X_processed_df['charges'] = y.values
    
    return X_processed_df


def feature_engineering(df):
    """
    特征工程。
    返回特征矩阵 X（必须不含目标列，所有列必须是数值类型）。
    """
    # 复制数据
    df = df.copy()
    
    # 分离目标列
    if 'charges' in df.columns:
        y = df['charges']
        X = df.drop(columns=['charges'])
    else:
        X = df
        y = None
    
    # 获取原始数值列（在 preprocess 中已保存）
    raw_num_cols = PREPROCESS_STATE.get('raw_num_cols', ['age', 'bmi', 'children'])
    
    # 从预处理后的列名中识别原始数值列
    # 预处理后，数值列名保持不变（age, bmi, children）
    # 类别列被 one-hot 编码为 sex_female, sex_male, smoker_no, smoker_yes, region_northeast 等
    
    # 创建新特征（基于原始数值列）
    # 注意：这些列在 preprocess 中已被标准化，但我们可以基于标准化后的值创建特征
    
    # 1. BMI 分类特征（基于标准化后的 bmi 值）
    if 'bmi' in X.columns:
        # 使用原始 bmi 的统计量来分类（标准化后的值以0为中心）
        bmi_std = X['bmi']
        X['bmi_high'] = (bmi_std > 0.5).astype(float)  # 高于均值0.5个标准差
        X['bmi_low'] = (bmi_std < -0.5).astype(float)   # 低于均值0.5个标准差
    
    # 2. 年龄分组特征
    if 'age' in X.columns:
        age_std = X['age']
        X['age_old'] = (age_std > 0.5).astype(float)    # 年龄较大
        X['age_young'] = (age_std < -0.5).astype(float) # 年龄较小
    
    # 3. 交互特征（基于标准化后的值）
    if 'age' in X.columns and 'bmi' in X.columns:
        X['age_bmi_interaction'] = X['age'] * X['bmi']
    
    # 4. 识别 smoker 列（one-hot 编码后的列名）
    smoker_yes_col = [c for c in X.columns if 'smoker' in c and ('yes' in c or 'Yes' in c or '1' in c)]
    if smoker_yes_col:
        smoker_yes = smoker_yes_col[0]
        if 'age' in X.columns:
            X['smoker_age_interaction'] = X[smoker_yes] * X['age']
        if 'bmi' in X.columns:
            X['smoker_bmi_interaction'] = X[smoker_yes] * X['bmi']
    
    # 5. children 的平方项（非线性关系）
    if 'children' in X.columns:
        X['children_squared'] = X['children'] ** 2
    
    # 确保所有列都是数值类型
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    
    return X


def build_model():
    """
    模型构建和超参数设置。
    返回 sklearn 兼容的模型对象（支持 fit/predict/predict_proba）。
    """
    model = lgb.LGBMRegressor(
        objective='regression',
        metric='rmse',
        num_leaves=31,
        max_depth=6,
        learning_rate=0.05,
        n_estimators=1000,  # 增加迭代次数，配合 early_stopping
        subsample=0.8,
        colsample_bytree=0.8,
        min_child_samples=20,
        reg_alpha=0.1,
        reg_lambda=0.1,
        random_state=42,
        verbose=-1,  # 抑制训练日志
        # 以下参数在 LightGBM 4.6.0 中已从 fit() 移除，需在构造时设置
        # early_stopping 通过 callbacks 在 fit() 中传递
    )
    
    return model


def evaluate_model(model, X_val, y_val):
    """
    自定义验证集评估指标。
    返回一个 dict，键名以 val_ 开头。
    """
    from sklearn.metrics import root_mean_squared_error, mean_absolute_error, r2_score
    
    val_preds = model.predict(X_val)
    
    return {
        'val_rmse': float(root_mean_squared_error(y_val, val_preds)),
        'val_mae': float(mean_absolute_error(y_val, val_preds)),
        'val_r2': float(r2_score(y_val, val_preds))
    }
