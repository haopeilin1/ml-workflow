import pandas as pd
import numpy as np

import warnings

from sklearn.metrics import root_mean_squared_error, mean_absolute_error, r2_score

from sklearn.impute import SimpleImputer

# ========== 全局状态 ==========
PREPROCESS_STATE = {}

# ========== 预处理函数（与训练阶段一致）==========
def preprocess(df, mode='train'):
    global PREPROCESS_STATE
    
    df = df.copy()
    
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    
    if mode == 'train':
        cat_cols = ['sex', 'smoker', 'region']
        num_cols = ['age', 'bmi', 'children']
        
        preprocessor = ColumnTransformer([
            ('cat', Pipeline([
                ('imputer', SimpleImputer(strategy='most_frequent')),
                ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
            ]), cat_cols),
            ('num', Pipeline([
                ('imputer', SimpleImputer(strategy='median')),
                ('scaler', StandardScaler())
            ]), num_cols)
        ], remainder='passthrough')
        
        if 'charges' in df.columns:
            X = df.drop(columns=['charges'])
            y = df['charges']
        else:
            X = df
            y = None
        
        preprocessor.fit(X)
        
        PREPROCESS_STATE['preprocessor'] = preprocessor
        PREPROCESS_STATE['cat_cols'] = cat_cols
        PREPROCESS_STATE['num_cols'] = num_cols
        
        ohe = preprocessor.named_transformers_['cat'].named_steps['encoder']
        cat_feature_names = ohe.get_feature_names_out(cat_cols)
        num_feature_names = num_cols
        PREPROCESS_STATE['feature_names'] = list(cat_feature_names) + list(num_feature_names)
        
    elif mode == 'test':
        if 'preprocessor' not in PREPROCESS_STATE:
            raise ValueError("PREPROCESS_STATE 中没有预处理器，请先运行 mode='train'")
        
        preprocessor = PREPROCESS_STATE['preprocessor']
        
        if 'charges' in df.columns:
            X = df.drop(columns=['charges'])
            y = df['charges']
        else:
            X = df
            y = None
    
    X_processed = preprocessor.transform(X)
    
    feature_names = PREPROCESS_STATE.get('feature_names', [f'feature_{i}' for i in range(X_processed.shape[1])])
    X_processed_df = pd.DataFrame(X_processed, columns=feature_names, index=df.index)
    
    if mode == 'test' and y is not None:
        X_processed_df['charges'] = y.values
    
    return X_processed_df


def feature_engineering(df):
    df = df.copy()
    
    if 'charges' in df.columns:
        y = df['charges']
        X = df.drop(columns=['charges'])
    else:
        X = df
        y = None
    
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce').fillna(0)
    
    return X


# ========== 加载数据 ==========
train = pd.read_csv('data/train.csv')
test = pd.read_csv('data/test.csv')

target_col = 'charges'
id_col = 'id'
if id_col not in test.columns:
    id_col = test.columns[0]

# ========== 预处理训练集（获取预处理器参数）==========
train_clean = preprocess(train, mode='train')

# ========== 预处理测试集 ==========
test_clean = preprocess(test, mode='test')

# ========== 分离特征和目标 ==========
if target_col in train_clean.columns:
    y_train = train_clean[target_col]
    X_train = train_clean.drop(columns=[target_col])
else:
    y_train = train[target_col]
    X_train = train_clean

X_test = test_clean.drop(columns=[target_col], errors='ignore')
if X_test is test_clean:
    X_test = test_clean.copy()

# ========== 特征工程 ==========
X_train_fe = feature_engineering(X_train)
if isinstance(X_train_fe, np.ndarray):
    X_train_fe = pd.DataFrame(X_train_fe, index=X_train.index)

X_test_fe = feature_engineering(X_test)
if isinstance(X_test_fe, np.ndarray):
    X_test_fe = pd.DataFrame(X_test_fe, index=X_test.index)

# ========== 清洗特征名 ==========
for _df in [X_train_fe, X_test_fe]:
    _df.columns = [re.sub('[^\\w]', '_', str(c)) for c in _df.columns]
for _df in [X_train_fe, X_test_fe]:
    if _df.columns.duplicated().any():
        _df.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(_df.columns)]

# ========== 加载最佳模型 ==========
print("正在加载 data/best_model.pkl ...")
try:
    with open('data/best_model.pkl', 'rb') as f:
        model = pickle.load(f)
    print("模型加载成功！")
except Exception as e:
    print(f"加载模型失败: {e}，将重新训练...")
    model = lgb.LGBMRegressor(
        objective='regression',
        metric='rmse',
        num_leaves=31,
        max_depth=6,
        learning_rate=0.05,
        n_estimators=500,
        subsample=0.8,
        colsample_bytree=0.8,
        min_child_samples=20,
        reg_alpha=0.1,
        reg_lambda=0.1,
        random_state=42,
        verbose=-1
    )
    model.fit(X_train_fe, y_train)
    print("模型重新训练完成！")

importance_df.to_csv('output/feature_importance.csv', index=False)

test_ids = test[id_col] if id_col in test.columns else range(len(test_preds))

train_preds = model.predict(X_train_fe)

predict_script = '''import pandas as pd
import numpy as np
import pickle
import re
import os
import warnings
warnings.filterwarnings('ignore')

from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.pipeline import Pipeline

# 预处理参数（与训练时一致）
CAT_COLS = ['sex', 'smoker', 'region']
NUM_COLS = ['age', 'bmi', 'children']
FEATURE_NAMES = None  # 将在加载模型后从特征重要性推断

def load_model(model_path='output/model.pkl'):
    """加载已训练的模型"""
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    return model

def preprocess_input(df):
    """对输入数据进行预处理"""
    df = df.copy()
    
    # 丢弃ID列
    if 'id' in df.columns:
        df = df.drop(columns=['id'])
    
    # 构建预处理器
    preprocessor = ColumnTransformer([
        ('cat', Pipeline([
            ('imputer', SimpleImputer(strategy='most_frequent')),
            ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
        ]), CAT_COLS),
        ('num', Pipeline([
            ('imputer', SimpleImputer(strategy='median')),
            ('scaler', StandardScaler())
        ]), NUM_COLS)
    ], remainder='passthrough')
    
    # 注意：这里需要拟合预处理器，实际使用时应该加载已拟合的预处理器
    # 简化处理：直接对原始数据做转换
    X = df.copy()
    if 'charges' in X.columns:
        X = X.drop(columns=['charges'])
    
    X_processed = preprocessor.fit_transform(X)
    
    # 生成特征名
    ohe = preprocessor.named_transformers_['cat'].named_steps['encoder']
    cat_names = ohe.get_feature_names_out(CAT_COLS)
    feature_names = list(cat_names) + NUM_COLS
    
    X_df = pd.DataFrame(X_processed, columns=feature_names)
    
    # 清洗特征名
    X_df.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X_df.columns]
    
    return X_df

def predict(new_data_path, model_path='output/model.pkl'):
    """对新数据进行预测"""
    model = load_model(model_path)
    df = pd.read_csv(new_data_path)
    
    # 预处理
    X = preprocess_input(df)
    
    # 预测
    predictions = model.predict(X)
    
    # 构建结果
    result = pd.DataFrame({
        'prediction': predictions
    })
    
    if 'id' in df.columns:
        result.insert(0, 'id', df['id'])
    
    return result

if __name__ == '__main__':
    import sys
    if len(sys.argv) > 1:
        input_file = sys.argv[1]
    else:
        input_file = 'data/test.csv'
    
    if not os.path.exists(input_file):
        print(f"错误：输入文件 {input_file} 不存在")
        sys.exit(1)
    
    result = predict(input_file)
    output_file = 'predictions_output.csv'
    result.to_csv(output_file, index=False)
    print(f"预测完成！结果已保存到 {output_file}")
    print(f"共 {len(result)} 条预测记录")
    print(result.head(10))
'''

with open('output/predict.py', 'w', encoding='utf-8') as f:
    f.write(predict_script)
