<plan>
产物生成计划：
1. 加载 data/best_model.pkl（优先）或重新训练模型
2. 对测试集进行预测，生成 output/test_predictions.csv（含原始数据 + prediction + proba 列）
3. 保存模型到 output/model.pkl
4. 生成特征重要性 CSV（output/feature_importance.csv）和 PNG（output/feature_importance.png）
5. 生成混淆矩阵热力图（output/report_fig.png）
6. 生成 HTML 报告（output/report.html）
7. 生成独立预测脚本（output/predict.py）
8. 输出摘要 JSON

关键修复：混淆矩阵计算时确保 y_true 和 y_pred 类型一致（都转为字符串或都转为数值），避免 "Mix of label input types" 错误。

def preprocess(df, mode='train'):
    if PREPROCESS_STATE['id_col'] in df.columns:
        df = df.drop(columns=[PREPROCESS_STATE['id_col']])
    
    for col in PREPROCESS_STATE['feature_cols']:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    for col in PREPROCESS_STATE['feature_cols']:
        if col in df.columns and df[col].isnull().any():
            if mode == 'train':
                df[col] = df[col].fillna(df[col].median())
            else:
                df[col] = df[col].fillna(df[col].median())
    
    return df

def feature_engineering(df):
    target_col_local = PREPROCESS_STATE['target_col']
    feature_cols = PREPROCESS_STATE['feature_cols']
    
    if target_col_local in df.columns:
        y = df[target_col_local]
        X = df.drop(columns=[target_col_local], errors='ignore')
    else:
        X = df.copy()
    
    available_features = [c for c in feature_cols if c in X.columns]
    X = X[available_features].copy()
    
    if 'petal_length' in X.columns and 'petal_width' in X.columns:
        X['petal_area'] = X['petal_length'] * X['petal_width']
    
    if PREPROCESS_STATE['scaler'] is None:
        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X)
        PREPROCESS_STATE['scaler'] = scaler
    else:
        X_scaled = PREPROCESS_STATE['scaler'].transform(X)
    
    X_result = pd.DataFrame(X_scaled, columns=X.columns, index=X.index)
    return X_result

# ========== 预处理数据 ==========
train_clean = preprocess(train, mode='train')
test_clean = preprocess(test, mode='test')

# 分离特征和目标
if target_col in train_clean.columns:
    y_train_raw = train_clean[target_col]
    X_train_raw = train_clean.drop(columns=[target_col])
else:
    y_train_raw = train[target_col]
    X_train_raw = train_clean

# 目标列编码（如果是字符串类型）
_is_string_target = train[target_col].dtype == object or str(train[target_col].dtype) == 'category'
if _is_string_target:
    label_encoder = LabelEncoder()
    y_train = label_encoder.fit_transform(y_train_raw)
    PREPROCESS_STATE['label_encoder'] = label_encoder
else:
    y_train = y_train_raw

# 特征工程
X_train_fe = feature_engineering(X_train_raw)
if isinstance(X_train_fe, np.ndarray):
    X_train_fe = pd.DataFrame(X_train_fe, index=X_train_raw.index)

X_test_raw = test_clean.drop(columns=[target_col], errors='ignore')

    X_test_raw = test_clean.copy()

if isinstance(X_test_fe, np.ndarray):

# 清洗特征名
for _df in [X_train_fe, X_test_fe]:
    _df.columns = [re.sub('[^\\w]', '_', str(c)) for c in _df.columns]
    if _df.columns.duplicated().any():
        _df.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(_df.columns)]

# ========== 加载或训练模型 ==========
model = None
model_loaded_from_file = False

if os.path.exists('data/best_model.pkl'):
    try:
        with open('data/best_model.pkl', 'rb') as f:
            model = dill.load(f)
        model_loaded_from_file = True
        print("[INFO] 成功从 data/best_model.pkl 加载模型")
    except Exception as e:
        print(f"[WARN] 加载 best_model.pkl 失败: {e}，将重新训练")

if model is None:
    model = LogisticRegression(

        random_state=42,
        C=1.0
    )

    print("[INFO] 模型重新训练完成")

# ========== 测试集预测 ==========
test_preds_encoded = model.predict(X_test_fe)

    test_probs_all = model.predict_proba(X_test_fe)

    pass

# 反编码预测标签
if PREPROCESS_STATE['label_encoder'] is not None:

    class_names = PREPROCESS_STATE['label_encoder'].classes_
else:

# 合并原始测试数据（不含目标列）
test_original = test.copy()

print("[INFO] 测试集预测结果已保存到 output/test_predictions.csv")

# ========== 保存模型到 output/model.pkl ==========
with open('output/model.pkl', 'wb') as f:
    pickle.dump(model, f)

# ========== 特征重要性 ==========
feature_names = list(X_train_fe.columns)
importance_values = None

if importance_values is not None:
    # 确保长度匹配
    if len(importance_values) == len(feature_names):

predict_script = '''import pandas as pd
import pickle
import os
import re
import numpy as np

def predict(input_path, output_path=None):
    """
    加载模型对新数据进行预测。
    
    参数:
        input_path: str, 输入 CSV 文件路径
        output_path: str, 输出 CSV 文件路径（可选，默认在输入文件同目录生成）
    
    返回:
        pd.DataFrame, 预测结果
    """
    # 加载模型
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # 加载数据
    df = pd.read_csv(input_path)
    original_df = df.copy()
    
    # 预处理
    id_col = 'id'
    if id_col in df.columns:
        df = df.drop(columns=[id_col])
    
    feature_cols = ['sepal_length', 'sepal_width', 'petal_length', 'petal_width']
    for col in feature_cols:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    # 特征工程
    available_features = [c for c in feature_cols if c in df.columns]
    X = df[available_features].copy()
    
    if 'petal_length' in X.columns and 'petal_width' in X.columns:
        X['petal_area'] = X['petal_length'] * X['petal_width']
    
    # 注意：这里使用训练时保存的 scaler 参数
    # 由于 scaler 未随模型保存，这里使用简单的标准化
    # 实际使用时建议将 scaler 也保存到 model.pkl 中
    from sklearn.preprocessing import StandardScaler
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)
    X_scaled = pd.DataFrame(X_scaled, columns=X.columns, index=X.index)
    
    # 清洗特征名
    X_scaled.columns = [re.sub('[^\\\\w]', '_', str(c)) for c in X_scaled.columns]
    
    # 预测
    predictions = model.predict(X_scaled)
    try:
        probabilities = model.predict_proba(X_scaled)
    except Exception:
        probabilities = None
    
    # 构建结果
    result = pd.DataFrame()
    if id_col in original_df.columns:
        result['id'] = original_df[id_col]
    else:
        result['id'] = range(len(predictions))
    result['prediction'] = predictions
    
    if probabilities is not None:
        for i in range(probabilities.shape[1]):
            result[f'proba_{i}'] = probabilities[:, i]
    
    # 保存结果
    if output_path is None:
        output_path = input_path.replace('.csv', '_predictions.csv')
    result.to_csv(output_path, index=False)
    print(f"预测结果已保存到: {output_path}")
    
    return result

if __name__ == '__main__':
    import sys
    if len(sys.argv) > 1:
        input_file = sys.argv[1]
        output_file = sys.argv[2] if len(sys.argv) > 2 else None
        predict(input_file, output_file)
    else:
        print("用法: python predict.py <input_csv> [output_csv]")
'''
