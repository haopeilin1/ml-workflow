产物生成计划：
1. 加载已保存的最佳模型（data/best_model.pkl），如果加载失败则回退到重新训练
2. 对测试集进行预测，生成 output/test_predictions.csv（含原始数据 + prediction + 各类概率）
3. 生成特征重要性 CSV 和 PNG（从模型属性获取，排除目标列）
4. 生成混淆矩阵热力图 output/report_fig.png
5. 生成 HTML 报告 output/report.html（含指标摘要、特征重要性表格、混淆矩阵、分类报告、测试集预测预览）
6. 保存模型到 output/model.pkl
7. 生成配套预测脚本 output/predict.py
</plan>

PREPROCESS_STATE = {

    'target_col': 'species',

    'feature_cols': ['sepal_length', 'sepal_width', 'petal_length', 'petal_width'],
    'label_encoder': None
}

# ========== 预处理函数 ==========
def preprocess(df, mode='train'):
    if PREPROCESS_STATE['id_col'] in df.columns:
        df = df.drop(columns=[PREPROCESS_STATE['id_col']])
    
    for col in PREPROCESS_STATE['feature_cols']:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
    
    for col in PREPROCESS_STATE['feature_cols']:
        if col in df.columns and df[col].isnull().any():
            if mode == 'train':
                df[col] = df[col].fillna(df[col].median())
            else:
                df[col] = df[col].fillna(df[col].median())
    
    return df


def feature_engineering(df):
    target_col = PREPROCESS_STATE['target_col']
    feature_cols = PREPROCESS_STATE['feature_cols']
    
    if target_col in df.columns:
        y = df[target_col]
        X = df.drop(columns=[target_col], errors='ignore')
    else:
        X = df.copy()
    
    available_features = [c for c in feature_cols if c in X.columns]
    X = X[available_features].copy()
    
    if 'petal_length' in X.columns and 'petal_width' in X.columns:
        X['petal_area'] = X['petal_length'] * X['petal_width']
    
    if PREPROCESS_STATE['scaler'] is None:
        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X)
        PREPROCESS_STATE['scaler'] = scaler
    else:
        X_scaled = PREPROCESS_STATE['scaler'].transform(X)
    
    X_result = pd.DataFrame(X_scaled, columns=X.columns, index=X.index)
    return X_result


def build_model():
    model = LogisticRegression(
        solver='lbfgs',
        max_iter=1000,
        random_state=42,
        C=1.0
    )
    return model

    else:
        # 从训练数据重新拟合预处理

        le = LabelEncoder()
        le.fit(train[target_col])
        PREPROCESS_STATE['label_encoder'] = le
        label_encoder = le
        print("[INFO] Fitted label encoder from training data")
    
except Exception as e:
    print(f"[WARN] Failed to load model: {e}, will retrain")
    model = None

# ========== 如果加载失败，重新训练 ==========
if model is None:

    val = pd.read_csv('data/validation.csv')
    
    train_clean = preprocess(train, mode='train')
    val_clean = preprocess(val, mode='test')
    
    if target_col in train_clean.columns:
        y_train = train_clean[target_col]
        X_train = train_clean.drop(columns=[target_col])
    else:
        y_train = train[target_col]
        X_train = train_clean
    
    if target_col in val_clean.columns:
        y_val = val_clean[target_col]

    # 处理字符串目标列

predict_script = '''#!/usr/bin/env python3
"""独立预测脚本 - 加载模型对新数据进行预测"""
import pandas as pd
import numpy as np
import pickle
import dill
import re
import sys
import os
from sklearn.preprocessing import StandardScaler

# 预处理状态（从训练时保存）
PREPROCESS_STATE = {
    'scaler': None,
    'target_col': 'species',
    'id_col': 'id',
    'feature_cols': ['sepal_length', 'sepal_width', 'petal_length', 'petal_width'],
    'label_encoder': None
}

def preprocess(df):
    if PREPROCESS_STATE['id_col'] in df.columns:
        df = df.drop(columns=[PREPROCESS_STATE['id_col']])
    for col in PREPROCESS_STATE['feature_cols']:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
    for col in PREPROCESS_STATE['feature_cols']:
        if col in df.columns and df[col].isnull().any():
            df[col] = df[col].fillna(df[col].median())
    return df

def feature_engineering(df):
    feature_cols = PREPROCESS_STATE['feature_cols']
    X = df.copy()
    available_features = [c for c in feature_cols if c in X.columns]
    X = X[available_features].copy()
    if 'petal_length' in X.columns and 'petal_width' in X.columns:
        X['petal_area'] = X['petal_length'] * X['petal_width']
    if PREPROCESS_STATE['scaler'] is not None:
        X_scaled = PREPROCESS_STATE['scaler'].transform(X)
    else:
        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X)
        PREPROCESS_STATE['scaler'] = scaler
    X_result = pd.DataFrame(X_scaled, columns=X.columns, index=X.index)
    return X_result

def load_model(model_path='output/model.pkl'):
    with open(model_path, 'rb') as f:
        model = dill.load(f)
    return model

def predict(input_path, output_path=None, model_path='output/model.pkl'):
    model = load_model(model_path)
    df = pd.read_csv(input_path)
    original_df = df.copy()
    df_clean = preprocess(df)
    X = feature_engineering(df_clean)
    X.columns = [re.sub('[^\\\\w]', '_', str(c)) for c in X.columns]
    
    predictions = model.predict(X)
    result_df = original_df.copy()
    result_df['prediction'] = predictions
    
    try:
        probas = model.predict_proba(X)
        for i in range(probas.shape[1]):
            result_df[f'proba_{i}'] = probas[:, i]
    except Exception:
        pass
    
    if output_path is None:
        output_path = input_path.replace('.csv', '_predictions.csv')
    result_df.to_csv(output_path, index=False)
    print(f"Predictions saved to {output_path}")
    return result_df

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: python predict.py <input_csv> [output_csv]")
        sys.exit(1)
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else None
    predict(input_path, output_path)
'''
