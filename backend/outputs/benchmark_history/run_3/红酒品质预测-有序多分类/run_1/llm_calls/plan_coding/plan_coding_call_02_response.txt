warnings.filterwarnings('ignore')

# ========== 全局状态 ==========
PREPROCESS_STATE = {
    'quality_mapping': None,
    'num_cols': None,
    'cat_cols': None,
    'high_skew_cols': ['chlorides', 'sulphates', 'residual sugar', 'free sulfur dioxide'],
    'interaction_cols': ['alcohol', 'volatile acidity', 'residual sugar', 'sulphates'],
    'wine_type_encoder': None,
    'robust_scaler': None,
    'label_encoder': None,
    'fitted': False
}

# ========== 辅助函数 ==========
def _clean_feature_names(df):
    df = df.copy()
    df.columns = [re.sub(r'[^\w]', '_', str(c)) for c in df.columns]
    return df

def _merge_rare_classes(y, min_samples=10):
    value_counts = y.value_counts().sort_index()
    rare_classes = value_counts[value_counts < min_samples].index.tolist()
    if len(rare_classes) == 0:
        return y, None
    y_mapped = y.copy()
    mapping = {}
    for cls in rare_classes:
        all_classes = sorted(value_counts.index.tolist())
        idx = all_classes.index(cls)
        if idx == 0:
            target = all_classes[idx + 1]
        elif idx == len(all_classes) - 1:
            target = all_classes[idx - 1]
        else:
            left_count = value_counts.get(all_classes[idx - 1], 0)
            right_count = value_counts.get(all_classes[idx + 1], 0)
            target = all_classes[idx - 1] if left_count >= right_count else all_classes[idx + 1]
        mapping[cls] = target
    y_mapped = y_mapped.replace(mapping)
    return y_mapped, mapping

def preprocess(df, mode='train'):
    global PREPROCESS_STATE
    df = df.copy()
    df = _clean_feature_names(df)
    df = df.drop(columns=['id'], errors='ignore')
    
    # wine_type 有序编码
    if 'wine_type' in df.columns:
        if mode == 'train':
            encoder = OrdinalEncoder(categories=[['red', 'white']], handle_unknown='use_encoded_value', unknown_value=-1)
            df['wine_type'] = encoder.fit_transform(df[['wine_type']]).astype(int)
            PREPROCESS_STATE['wine_type_encoder'] = encoder
        else:
            encoder = PREPROCESS_STATE.get('wine_type_encoder')
            if encoder is not None:
                df['wine_type'] = encoder.transform(df[['wine_type']]).astype(int)
    
    # log1p 变换
    high_skew_cols = PREPROCESS_STATE['high_skew_cols']
    for col in high_skew_cols:
        if col in df.columns:
            min_val = df[col].min()
            if min_val < 0:
                df[col] = df[col] - min_val + 1e-6
            df[col] = np.log1p(df[col])
    
    # RobustScaler
    target_col = 'quality'
    num_cols = df.select_dtypes(include=[np.number]).columns.tolist()
    num_cols = [c for c in num_cols if c not in [target_col, 'wine_type']]
    
    if mode == 'train':
        PREPROCESS_STATE['num_cols'] = num_cols
        scaler = RobustScaler()
        df[num_cols] = scaler.fit_transform(df[num_cols])
        PREPROCESS_STATE['robust_scaler'] = scaler
        PREPROCESS_STATE['fitted'] = True
    else:
        scaler = PREPROCESS_STATE.get('robust_scaler')
        if scaler is not None and len(num_cols) > 0:
            df[num_cols] = scaler.transform(df[num_cols])
    
    # 合并稀有类别
    if mode == 'train' and target_col in df.columns:
        y = df[target_col].copy()
        y_mapped, mapping = _merge_rare_classes(y, min_samples=10)
        if mapping is not None:
            df[target_col] = y_mapped
            PREPROCESS_STATE['quality_mapping'] = mapping
    elif mode == 'test' and target_col in df.columns:
        mapping = PREPROCESS_STATE.get('quality_mapping')
        if mapping is not None:
            df[target_col] = df[target_col].replace(mapping)
    
    return df

def feature_engineering(df):
    global PREPROCESS_STATE
    df = df.copy()
    target_col = 'quality'
    
    y = df[target_col].copy() if target_col in df.columns else None
    X = df.drop(columns=[target_col], errors='ignore')
    
    # 确保 wine_type 是数值类型（修复报错）
    if 'wine_type' in X.columns:
        X['wine_type'] = pd.to_numeric(X['wine_type'], errors='coerce').fillna(0).astype(int)
    
    # 交互特征
    if 'wine_type' in X.columns:
        if 'alcohol' in X.columns:
            X['wine_type_x_alcohol'] = X['wine_type'].astype(float) * X['alcohol'].astype(float)
        if 'volatile_acidity' in X.columns:
            X['wine_type_x_volatile_acidity'] = X['wine_type'].astype(float) * X['volatile_acidity'].astype(float)
    
    # 比值特征
    if 'fixed_acidity' in X.columns and 'volatile_acidity' in X.columns:
        X['fixed_to_volatile_acidity'] = X['fixed_acidity'].astype(float) / (X['volatile_acidity'].astype(float) + 1e-6)
    if 'free_sulfur_dioxide' in X.columns and 'total_sulfur_dioxide' in X.columns:
        X['free_to_total_so2'] = X['free_sulfur_dioxide'].astype(float) / (X['total_sulfur_dioxide'].astype(float) + 1e-6)
    
    # 酒精与糖分交互
    if 'alcohol' in X.columns and 'residual_sugar' in X.columns:
        X['alcohol_x_sugar'] = X['alcohol'].astype(float) * X['residual_sugar'].astype(float)
    
    # 确保所有列都是数值类型
    for col in X.columns:
        if X[col].dtype == 'object':
            X[col] = pd.to_numeric(X[col], errors='coerce')
    
    X = X.fillna(0)
    return X

def build_model():
    lgbm = LGBMClassifier(
        objective='multiclass',
        boosting_type='gbdt',
        n_estimators=300,
        learning_rate=0.03,
        num_leaves=31,
        max_depth=7,
        min_child_samples=50,
        min_split_gain=0.01,
        subsample=0.8,
        colsample_bytree=0.8,
        reg_alpha=0.5,
        reg_lambda=0.5,
        is_unbalance=True,
        random_state=42,
        verbose=-1,
        n_jobs=-1
    )
    return lgbm

os.makedirs('output', exist_ok=True)

# 如果模型加载失败，重新训练
if model is None:

    train_clean = preprocess(train, mode='train')
    
    # 处理目标列编码
    if target_col in train.columns and train[target_col].dtype == object:
        _label_encoder = LabelEncoder()
        y_train = _label_encoder.fit_transform(train[target_col])
        PREPROCESS_STATE['label_encoder'] = _label_encoder
    else:
        y_train = train_clean[target_col] if target_col in train_clean.columns else train[target_col]
    
    X_train = train_clean.drop(columns=[target_col], errors='ignore')
    X_train_fe = feature_engineering(X_train)
    if isinstance(X_train_fe, np.ndarray):
        X_train_fe = pd.DataFrame(X_train_fe, index=X_train.index)
    
    # 清洗特征名
    X_train_fe.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_train_fe.columns]
    if X_train_fe.columns.duplicated().any():
        X_train_fe.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X_train_fe.columns)]
    
    model = build_model()
    model.fit(X_train_fe, y_train)
    
    # 保存模型
    with open('data/best_model.pkl', 'wb') as f:
        pickle.dump(model, f)
    print("[INFO] Model retrained and saved to data/best_model.pkl")

# 预处理测试集

import pandas as pd
import pickle
import re
import numpy as np
import os

def predict(new_data_path, output_path='predictions.csv'):
    """加载模型并对新数据进行预测"""
    # 加载模型
    with open('output/model.pkl', 'rb') as f:
        model = pickle.load(f)
    
    # 加载数据
    df = pd.read_csv(new_data_path)
    
    # 预处理（简化版，实际使用时需要完整的预处理流程）

    # 请根据实际情况调整
    
    # 预测
    preds = model.predict(df)
    probs = model.predict_proba(df)
    
    # 保存结果
    result = pd.DataFrame({'prediction': preds})
    for i in range(probs.shape[1]):
        result[f'proba_{i}'] = probs[:, i]
    result.to_csv(output_path, index=False)
    print(f"Predictions saved to {output_path}")
    return result

if __name__ == '__main__':
    import sys
    if len(sys.argv) > 1:
        predict(sys.argv[1])
