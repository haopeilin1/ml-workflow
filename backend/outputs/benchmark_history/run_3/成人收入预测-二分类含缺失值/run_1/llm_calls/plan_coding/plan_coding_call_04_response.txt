def preprocess(df, mode='train'):
    """
    数据清洗和预处理。
    mode='train' 时拟合参数，保存到 PREPROCESS_STATE。
    mode='test' 时应用已保存的参数。
    返回处理后的 DataFrame（仍包含目标列）。
    """
    global PREPROCESS_STATE
    
    df = df.copy()
    
    # 丢弃 id 列（唯一值太多，会导致过拟合）
    df = df.drop(columns=['id'], errors='ignore')
    
    # 处理 native_country：尝试转为数值，失败则保留为字符串
    if 'native_country' in df.columns:
        # 先尝试 pd.to_numeric，但它是字符串，会失败，所以用 errors='coerce'
        # 实际上 native_country 是分类变量，转为数值会全部变成 NaN
        # 所以这里我们保留为字符串，后续用 OrdinalEncoder 处理
        pass  # 保留为字符串，在 feature_engineering 中编码
    
    # 处理目标列 income（仅在训练和验证时存在）
    if 'income' in df.columns:
        # 将 income 从字符串转为 0/1
        if mode == 'train':
            income_map = {'<=50K': 0, '>50K': 1}
            PREPROCESS_STATE['income_map'] = income_map
        df['income'] = df['income'].map(PREPROCESS_STATE.get('income_map', {'<=50K': 0, '>50K': 1}))
    
    # 缺失值填充（对类别列用 'Unknown'，对数值列用中位数）
    cat_cols = ['workclass', 'occupation', 'native_country']
    for col in cat_cols:
        if col in df.columns:
            if mode == 'train':
                PREPROCESS_STATE[f'{col}_fill'] = 'Unknown'
            df[col] = df[col].fillna(PREPROCESS_STATE.get(f'{col}_fill', 'Unknown'))
    
    # 数值列缺失值填充（fnlwgt 没有缺失值，但以防万一）
    num_cols = ['age', 'fnlwgt', 'education_num', 'capital_gain', 'capital_loss', 'hours_per_week']
    for col in num_cols:
        if col in df.columns and df[col].isna().any():
            if mode == 'train':
                PREPROCESS_STATE[f'{col}_fill'] = df[col].median()
            df[col] = df[col].fillna(PREPROCESS_STATE.get(f'{col}_fill', 0))
    
    return df


def feature_engineering(df):
    """
    特征工程。
    返回特征矩阵 X（必须不含目标列，所有列必须是数值类型）。
    """
    global PREPROCESS_STATE
    
    df = df.copy()
    
    # 分离目标列（如果有）
    y = None
    if 'income' in df.columns:
        y = df['income']
        df = df.drop(columns=['income'])
    
    # 识别类别列和数值列
    cat_cols = df.select_dtypes(include=['object', 'category']).columns.tolist()
    num_cols = df.select_dtypes(exclude=['object', 'category']).columns.tolist()
    
    # 使用 ColumnTransformer 进行编码和缩放
    from sklearn.compose import ColumnTransformer
    from sklearn.preprocessing import OneHotEncoder, StandardScaler
    from sklearn.impute import SimpleImputer
    from sklearn.pipeline import Pipeline
    
    preprocessor = ColumnTransformer([
        ('cat', Pipeline([
            ('imputer', SimpleImputer(strategy='constant', fill_value='Unknown')),
            ('encoder', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
        ]), cat_cols),
        ('num', Pipeline([
            ('imputer', SimpleImputer(strategy='median')),
            ('scaler', StandardScaler())
        ]), num_cols)
    ], remainder='passthrough')
    
    # 拟合或转换
    if not hasattr(PREPROCESS_STATE, 'feature_pipeline') or PREPROCESS_STATE.get('feature_pipeline') is None:
        # 训练模式：拟合
        X_processed = preprocessor.fit_transform(df)
        PREPROCESS_STATE['feature_pipeline'] = preprocessor
        # 获取特征名称
        feature_names = []
        for name, transformer, cols in preprocessor.transformers_:
            if name == 'cat':
                encoder = transformer.named_steps['encoder']
                if hasattr(encoder, 'get_feature_names_out'):
                    feature_names.extend(encoder.get_feature_names_out(cols))
                else:
                    feature_names.extend(cols)
            elif name == 'num':
                feature_names.extend(cols)
            elif name == 'remainder' and transformer == 'passthrough':
                feature_names.extend(cols)
        PREPROCESS_STATE['feature_names'] = feature_names
    else:
        # 测试模式：应用已拟合的转换器
        preprocessor = PREPROCESS_STATE['feature_pipeline']
        X_processed = preprocessor.transform(df)
        feature_names = PREPROCESS_STATE.get('feature_names', [])
    
    # 转换为 DataFrame
    if len(feature_names) == X_processed.shape[1]:
        X_result = df.__class__(X_processed, columns=feature_names)
    else:
        X_result = df.__class__(X_processed)
    
    # 添加一些交叉特征（在编码后的数值特征上）
    # 注意：这里我们直接在原始数值列上创建特征，因为编码后的列名不确定
    if 'age' in df.columns and 'hours_per_week' in df.columns:
        X_result['age_hours_interaction'] = df['age'] * df['hours_per_week']
    
    if 'capital_gain' in df.columns and 'capital_loss' in df.columns:
        X_result['capital_diff'] = df['capital_gain'] - df['capital_loss']
    
    # 确保所有列都是数值类型
    for col in X_result.columns:
        if X_result[col].dtype == 'object':
            X_result[col] = X_result[col].astype('category').cat.codes
    
    return X_result


def build_model():
    """
    模型构建和超参数设置。
    返回 sklearn 兼容的模型对象（支持 fit/predict/predict_proba）。
    """
    import lightgbm as lgb
    
    model = lgb.LGBMClassifier(
        objective='binary',
        num_leaves=31,
        max_depth=6,
        learning_rate=0.05,
        n_estimators=300,
        subsample=0.8,
        colsample_bytree=0.8,
        min_child_samples=20,
        reg_alpha=0.1,
        reg_lambda=0.1,
        random_state=42,
        verbose=-1
    )
    
    return model
