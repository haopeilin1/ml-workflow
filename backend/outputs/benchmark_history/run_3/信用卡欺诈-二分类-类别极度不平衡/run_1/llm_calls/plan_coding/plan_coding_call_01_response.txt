产物生成计划：加载已保存的最佳模型，对测试集进行预测，生成二分类任务所需的所有产物文件。

步骤：
1. 加载 data/best_model.pkl 模型
2. 加载测试集 data/test.csv 并进行与训练时相同的预处理
3. 对测试集进行预测，生成 prediction 和 probability 列
4. 保存模型到 output/model.pkl
5. 提取特征重要性（从 LightGBM 模型属性获取），保存 CSV 和 PNG
6. 生成 report.html（包含指标摘要、特征重要性表格、测试集预测预览）
7. 生成 output/test_predictions.csv
8. 输出摘要 JSON

import warnings

# 确保输出目录存在
os.makedirs('output', exist_ok=True)

        model = pickle.load(f)

print("Loading test data...")
test = pd.read_csv('data/test.csv')

target_col = 'IsFraud'
id_col = 'id' if 'id' in test.columns else test.columns[0]

# 预处理函数（与训练时保持一致）
def preprocess_test(df):
    df = df.copy()
    
    # 丢弃 id 列和所有 feat1-feat28 列
    drop_cols = ['id'] + [f'feat{i}' for i in range(1, 29)]
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    # Transaction_Amount 做 log1p 变换
    if 'Transaction_Amount' in df.columns:
        df['Transaction_Amount_log'] = np.log1p(df['Transaction_Amount'])
        df = df.drop(columns=['Transaction_Amount'])
    
    # 从 Time 列提取周期性特征
    if 'Time' in df.columns:
        df['Time_hour'] = (df['Time'] // 3600) % 24
        df['Time_day_of_week'] = (df['Time'] // (3600 * 24)) % 7
        df = df.drop(columns=['Time'])
    
    # 特征名清洗
    df.columns = [str(c).replace(' ', '_').replace('-', '_') for c in df.columns]
    
    # 处理缺失值
    for col in df.columns:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
        if df[col].isna().any():
            if df[col].dtype in ['float64', 'int64']:
                df[col] = df[col].fillna(df[col].median())
            else:
                df[col] = df[col].fillna(0)
    
    # 移除目标列（如果存在）
    if target_col in df.columns:
        df = df.drop(columns=[target_col])
    
    return df

# 预处理测试集
X_test_raw = preprocess_test(test)

# 确保所有列为数值类型

    if X_test_raw[col].dtype == 'object':
        X_test_raw[col] = pd.to_numeric(X_test_raw[col], errors='coerce')
    if X_test_raw[col].isna().any():
        X_test_raw[col] = X_test_raw[col].fillna(X_test_raw[col].median() if X_test_raw[col].dtype in ['float64', 'int64'] else 0)

# 清洗特征名（与训练时一致）
X_test_raw.columns = [re.sub(r'[^\w]', '_', str(c)) for c in X_test_raw.columns]
if X_test_raw.columns.duplicated().any():
    X_test_raw.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X_test_raw.columns)]

print(f"Test features shape: {X_test_raw.shape}")

# ========== 3. 测试集预测 ==========
print("Making predictions on test set...")
if hasattr(model, 'predict_proba'):

# 使用默认阈值 0.5 进行预测
test_preds = (test_probs >= 0.5).astype(int)

# ========== 4. 保存模型到 output/model.pkl ==========

# ========== 5. 提取特征重要性 ==========
print("Extracting feature importance...")
feature_names = X_test_raw.columns.tolist()

# 尝试从 Pipeline 中提取特征重要性
try:
    # 如果是 Pipeline，获取最终模型
    if hasattr(model, 'named_steps') and 'model' in model.named_steps:
        final_model = model.named_steps['model']
    else:
        final_model = model
    
    # 获取特征重要性
    if hasattr(final_model, 'feature_importances_'):
        importances = final_model.feature_importances_
    elif hasattr(final_model, 'coef_'):
        importances = np.abs(final_model.coef_).flatten()
    else:
        importances = np.ones(len(feature_names)) / len(feature_names)
    
    # 确保长度匹配
    if len(importances) != len(feature_names):
        # 如果 Pipeline 中有特征选择步骤，尝试获取支持的特征
        if hasattr(model, 'named_steps'):
            for step_name, step in model.named_steps.items():
                if hasattr(step, 'get_support'):
                    mask = step.get_support()
                    feature_names = [f for f, m in zip(feature_names, mask) if m]
                    break
        # 截断或填充以匹配
        if len(importances) > len(feature_names):
            importances = importances[:len(feature_names)]
        elif len(importances) < len(feature_names):
            importances = np.pad(importances, (0, len(feature_names) - len(importances)), 'constant', constant_values=0)
    
    # 创建特征重要性 DataFrame
    fi_df = pd.DataFrame({
        'name': feature_names,
        'importance': importances
    })
    fi_df = fi_df.sort_values('importance', ascending=False).reset_index(drop=True)
    
except Exception as e:
    print(f"Warning: Could not extract feature importance: {e}")
    fi_df = pd.DataFrame({

    })

# ========== 8. 生成预测脚本 output/predict.py ==========
predict_script = '''#!/usr/bin/env python3
"""
Independent prediction script.
Usage: python predict.py <input_csv_path> [output_csv_path]
Loads output/model.pkl and makes predictions on new data.
"""

import pandas as pd
import numpy as np
import pickle
import sys
import os
import re
import warnings
warnings.filterwarnings('ignore')

def preprocess(df):
    """Preprocess input data (same as training preprocessing)."""
    df = df.copy()
    
    # Drop id and feat columns
    drop_cols = ['id'] + [f'feat{i}' for i in range(1, 29)]
    df = df.drop(columns=[c for c in drop_cols if c in df.columns], errors='ignore')
    
    # Transaction_Amount log1p transform
    if 'Transaction_Amount' in df.columns:
        df['Transaction_Amount_log'] = np.log1p(df['Transaction_Amount'])
        df = df.drop(columns=['Transaction_Amount'])
    
    # Time features
    if 'Time' in df.columns:
        df['Time_hour'] = (df['Time'] // 3600) % 24
        df['Time_day_of_week'] = (df['Time'] // (3600 * 24)) % 7
        df = df.drop(columns=['Time'])
    
    # Clean column names
    df.columns = [str(c).replace(' ', '_').replace('-', '_') for c in df.columns]
    
    # Handle missing values
    for col in df.columns:
        if df[col].dtype == 'object':
            df[col] = pd.to_numeric(df[col], errors='coerce')
        if df[col].isna().any():
            if df[col].dtype in ['float64', 'int64']:
                df[col] = df[col].fillna(df[col].median())
            else:
                df[col] = df[col].fillna(0)
    
    # Remove target column if present
    if 'IsFraud' in df.columns:
        df = df.drop(columns=['IsFraud'])
    
    return df

def main():
    if len(sys.argv) < 2:
        print("Usage: python predict.py <input_csv_path> [output_csv_path]")
        sys.exit(1)
    
    input_path = sys.argv[1]
    output_path = sys.argv[2] if len(sys.argv) > 2 else 'predictions.csv'
    
    # Load model
    model_path = os.path.join(os.path.dirname(__file__), 'model.pkl')
    with open(model_path, 'rb') as f:
        model = pickle.load(f)
    
    # Load and preprocess data
    df = pd.read_csv(input_path)
    id_col = 'id' if 'id' in df.columns else df.columns[0]
    ids = df[id_col] if id_col in df.columns else range(len(df))
    
    X = preprocess(df)
    
    # Clean feature names
    X.columns = [re.sub(r'[^\\w]', '_', str(c)) for c in X.columns]
    if X.columns.duplicated().any():
        X.columns = [f"{c}_{i}" if i > 0 else str(c) for i, c in enumerate(X.columns)]
    
    # Predict
    if hasattr(model, 'predict_proba'):
        probs = model.predict_proba(X)[:, 1]
    else:
        probs = model.predict(X).astype(float)
    
    preds = (probs >= 0.5).astype(int)
    
    # Save results
    result = pd.DataFrame({
        'id': ids,
        'prediction': preds,
        'probability': probs
    })
    result.to_csv(output_path, index=False)
    print(f"Predictions saved to {output_path}")
    print(f"Total samples: {len(result)}")
    print(f"Predicted fraud: {preds.sum()} ({preds.mean()*100:.2f}%)")

if __name__ == '__main__':
    main()
'''
